10    (tmp_path / "a.txt").write_text("old")

16        )

18    assert blocked.reason == "fresh_read_required"
19    read = await runtime.step(ToolCall(name="read_file", arguments={"path": "a.txt"}))
20    assert read.status == "success"
21    changed = await runtime.step(
22        ToolCall(

24        )
25    )
