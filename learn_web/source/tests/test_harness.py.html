133@pytest.mark.asyncio

136async def test_partial_test_success_cannot_erase_an_unfixed_regression(tmp_path, native, subtests):
137    (tmp_path / "app.py").write_text("VALUE = 0\n")
138    (tmp_path / "test_app.py").write_text(
139        "from app import VALUE\n"
140        "def test_regression(): assert VALUE == 1\n"
141        "def test_control(): assert True\n"
142    )
143    full = "python -m pytest -q test_app.py"
144    control, regression = full + "::test_control", full + "::test_regression"
145    if native:
146        check = "    def test_regression(self): self.assertEqual(VALUE, 1)\n"
147        if subtests:
148            check = ("    def test_regression(self):\n"
149                     "        with self.subTest(case='value'): self.assertEqual(VALUE, 1)\n")
150        (tmp_path / "test_app.py").write_text(
151            "import unittest\nfrom app import VALUE\n"
152            "class Check(unittest.TestCase):\n"
153            + check +
154            "    def test_control(self): self.assertTrue(True)\n"
155        )
156        (tmp_path / "runtests.py").write_text(
157            "import sys, unittest\n"
158            "suite = unittest.defaultTestLoader.loadTestsFromName(sys.argv[1])\n"
159            "result = unittest.TextTestRunner().run(suite)\n"
160            "raise SystemExit(not result.wasSuccessful())\n"
161        )
162        full = "python runtests.py test_app.Check"
163        control, regression = full + ".test_control", full + ".test_regression"
164    runtime = Runtime(tmp_path, Settings(approval="auto", memory_enabled=False))
165    runtime.start(TaskSpec(prompt="Verify the behavior", require_verification=True))
166    failed = await runtime.step(ToolCall(name="run_tests", arguments={
167        "command": full,
168    }))
169    assert failed.reason == "tests_failed"
170    passed = await runtime.step(ToolCall(name="run_tests", arguments={
171        "command": control,
172    }))
173    assert passed.status == "success"
174    assert not runtime.verifier.final(runtime.state).passed
175    runtime.finalize("The selected test passed; done.")
176    assert runtime.state.phase != "completed"
177    assert "test_regression" in runtime.state.feedback
178
179    await runtime.step(ToolCall(name="read_file", arguments={"path": "app.py"}))
180    await runtime.step(ToolCall(name="patch_file", arguments={
181        "path": "app.py", "old_text": "VALUE = 0", "new_text": "VALUE = 1",
182    }))
183    await runtime.step(ToolCall(name="run_tests", arguments={
184        "command": control,
185    }))
186    assert not runtime.verifier.final(runtime.state).passed
187    repaired = await runtime.step(ToolCall(name="run_tests", arguments={
188        "command": regression,
189    }))
190    assert repaired.status == "success"
191    assert runtime.verifier.final(runtime.state).passed

323async def test_injected_hooks_run_and_fail_closed(tmp_path):
324    class Plugin:
325        def __init__(self):
326            self.seen = []
327
328        def before_tool(self, _state, call):
329            self.seen.append(("before", call.name))
330            return HookResult()
331
332        def after_tool(self, _state, call, _outcome):
333            self.seen.append(("after", call.name))
334            return HookResult(feedback="hook note")
335
336    plugin = Plugin()
337    runtime = Runtime(
338        tmp_path,
339        Settings(approval="auto"),
340        model=FakeModel([]),
341        hooks=Hooks([plugin]),
342    )
343    runtime.start(TaskSpec(prompt="list files"))
344    outcome = await runtime.step(ToolCall(name="list_files", arguments={}))
345    assert outcome.status == "success"
346    assert plugin.seen == [("before", "list_files"), ("after", "list_files")]
347    assert any(m["role"] == "user" and m["content"] == "hook note" for m in runtime.state.messages)
