14def last_output(runtime):
15    item = next(x for x in reversed(runtime._structured_history) if x.get("type") == "function_call_output")
16    return json.loads(item["output"])

19@pytest.mark.asyncio
20async def test_successful_read_omits_empty_fields_and_full_hashes(tmp_path):
21    (tmp_path / "a.txt").write_text("alpha\n")
22    runtime = Runtime(tmp_path, Settings(approval="auto", memory_enabled=False))
23    runtime.start(TaskSpec(prompt="inspect a.txt"))
24    runtime._in_turn = False
25    result = await runtime.step(ToolCall(name="read_file", arguments={"path": "a.txt"}))
26    payload = last_output(runtime)
27    assert payload == {"status": "success", "evidence_id": result.evidence_id[:12],
28                       "content": "1: alpha", "path": "a.txt", "start": 1, "end": 1}

31@pytest.mark.asyncio
32async def test_blocked_and_edit_outputs_keep_reason_and_changed_paths(tmp_path):
33    (tmp_path / "a.txt").write_text("alpha\n")
34    runtime = Runtime(tmp_path, Settings(approval="auto", memory_enabled=False))
35    runtime.start(TaskSpec(prompt="edit a.txt"))
36    await runtime.step(ToolCall(name="patch_file", arguments={"path": "a.txt", "old_text": "alpha", "new_text": "beta"}))
37    assert last_output(runtime)["reason"] == "fresh_read_required"
38    await runtime.step(ToolCall(name="read_file", arguments={"path": "a.txt"}))
39    await runtime.step(ToolCall(name="patch_file", arguments={"path": "a.txt", "old_text": "alpha", "new_text": "beta"}))
40    assert last_output(runtime)["changed_paths"] == ["a.txt"]

43@pytest.mark.asyncio
44async def test_memory_store_accepts_the_visible_evidence_prefix(tmp_path):
45    (tmp_path / "settings.py").write_text("RETRIES = 4\n")
46    runtime = Runtime(tmp_path, Settings(approval="auto", completion="off", memory_enabled=True,
47                                         memory_evolution=True))
48    runtime.start(TaskSpec(prompt="Inspect retry settings"))
49    read = await runtime.step(ToolCall(name="read_file", arguments={"path": "settings.py"}))
50    saved = await runtime.step(ToolCall(name="memory_store", arguments={
51        "text": "Retry at most four times.", "kind": "fact", "when": "Configuring retries",
52        "evidence_id": read.evidence_id[:12]}))
53    assert saved.status == "success", saved
54    assert saved.data["provenance"]["evidence_id"] == read.evidence_id
55    too_short = await runtime.step(ToolCall(name="memory_store", arguments={
56        "text": "Retry at most four times.", "kind": "fact", "when": "Configuring retries",
57        "evidence_id": read.evidence_id[:8]}))
58    assert too_short.reason == "memory_provenance"
