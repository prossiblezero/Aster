168def test_complete_outputs_carry_no_archive_footer_and_ids_are_shortened():
169    from aster.context.view import render_output
170    full = "a" * 64
171    record = {"status": "success", "evidence_id": "e" * 12, "content": "ok", "observation": {"id": full, "characters": 2}}
172    assert render_output(json.dumps(record), "run_shell") == "ok"
173    clipped = {**record, "content": "part\n[excerpt; observation_read for archived content]"}
174    assert render_output(json.dumps(clipped), "run_shell").endswith(f"[full output: observation_read observation_id={full[:16]}]")
175    masked = {**record, "content": f"[archived: 900 chars; observation_read observation_id={full}]"}
176    assert render_output(json.dumps(masked), "run_shell") == f"[archived: 900 chars; observation_read observation_id={full[:16]}]"
177    listing = {"status": "success", "content": json.dumps({"observations": [{"id": full, "tool": "run_shell"}]})}
178    assert f'"id": "{full[:16]}"' in render_output(json.dumps(listing), "observation_read")
179    # Other 64-hex text (a digest a command printed) is data and stays intact.
180    printed = {**record, "content": f"sha256 {full}"}
181    assert render_output(json.dumps(printed), "run_shell") == f"sha256 {full}"

184def test_observation_read_resolves_unique_prefixes(tmp_path):
185    from aster.safety.redaction import Redactor
186    from aster.storage.evidence import EvidenceStore
187    from aster.storage.workspace import BoundaryError
188    store = EvidenceStore(tmp_path, Redactor([]))
189    ref = store.archive_observation("run1", "c1", "run_shell", "first output")
190    assert store.read_observation("run1", ref["id"][:16])["content"] == "first output"
191    with pytest.raises(BoundaryError):
192        store.read_observation("run1", ref["id"][:8])  # below the minimum prefix
193    with pytest.raises(BoundaryError):
194        store.read_observation("run1", "f" * 16)  # no such archive
