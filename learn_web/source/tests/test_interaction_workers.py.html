20class CallThenFinish:
21    def __init__(self, name, arguments):
22        self.call = ToolCall(name=name, arguments=arguments)
23        self.requests = []
24
25    async def complete(self, messages, tools):
26        self.requests.append(messages)
27        if len(self.requests) == 1:
28            call = self.call
29            return ModelReply(
30                calls=[call],
31                raw=[
32                    {
33                        "type": "function_call",
34                        "name": call.name,
35                        "call_id": call.call_id,
36                        "arguments": json.dumps(call.arguments),
37                    }
38                ],
39            )
40        return ModelReply(text="Finished.")

43class PausedModel(CallThenFinish):
44    def __init__(self):
45        super().__init__("list_files", {})
46        self.entered = asyncio.Event()
47        self.release = asyncio.Event()
48
49    async def complete(self, messages, tools):
50        if not self.requests:
51            self.entered.set()
52            await self.release.wait()
53        return await super().complete(messages, tools)
