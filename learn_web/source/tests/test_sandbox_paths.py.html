15def test_map_root_only_rewrites_whole_path_components():
16    source, target = Path("/app"), Path("/tmp/copy")
17    assert map_root("cd /app && python3 /app/x.py '/app'", source, target) == (
18        "cd /tmp/copy && python3 /tmp/copy/x.py '/tmp/copy'")
19    unchanged = "ls /apps /app-old /x/app ./app ~/app /app.bak"
20    assert map_root(unchanged, source, target) == unchanged

23@pytest.mark.asyncio
24async def test_absolute_workspace_paths_read_and_write_through_the_sandbox(tmp_path):
25    root = tmp_path.resolve()
26    (root / "interp.py").write_text("print('hello')\n")
27    runtime = Runtime(root, Settings(approval="auto", memory_enabled=False, sandbox="required"),
28                      approval_callback=lambda *_: True)
29    runtime.start(TaskSpec(prompt="run interp.py"))
30    result = await runtime.step(ToolCall(name="run_shell", arguments={
31        "command": f"cd {root} && python3 {root}/interp.py > {root}/out.txt && pwd"}))
32    assert result.status == "success", result
33    assert result.content.strip() == str(root)
34    assert result.changed_paths == ["out.txt"]
35    assert (root / "out.txt").read_text() == "hello\n"

38@pytest.mark.asyncio
39async def test_absolute_paths_in_test_commands_target_the_test_copy(tmp_path):
40    root = tmp_path.resolve()
41    (root / "test_ok.py").write_text("def test_ok():\n    assert True\n")
42    runtime = Runtime(root, Settings(approval="auto", memory_enabled=False, sandbox="required"))
43    runtime.start(TaskSpec(prompt="verify", require_verification=True))
44    result = await runtime.step(ToolCall(name="run_tests", arguments={
45        "command": f"python -m pytest -q {root}/test_ok.py"}))
46    assert result.status == "success", result
47    assert result.data["test_results"] == {"pytest: test_ok::test_ok": "passed"}
