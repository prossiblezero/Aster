13ENV_TEST = "def test_env():\n    raise RuntimeError('cftime version mismatch')\n"

17def setup(tmp_path):
18    (tmp_path / "app.py").write_text("VALUE = 0\n")
19    (tmp_path / "test_env.py").write_text(ENV_TEST)
20    (tmp_path / "test_value.py").write_text(TARGET)
21    runtime = Runtime(tmp_path, Settings(approval="auto", memory_enabled=False, sandbox="required",
22                                         completion="strict"))
23    runtime.start(TaskSpec(prompt="Fix VALUE", require_verification=True))
24    return runtime

31async def edit(runtime, new):
32    await runtime.step(ToolCall(name="read_file", arguments={"path": "app.py"}))
33    await runtime.step(ToolCall(name="patch_file", arguments={
34        "path": "app.py", "old_text": runtime.workspace.read("app.py").strip(), "new_text": new}))

46    runtime = setup(tmp_path)

115async def test_each_failing_test_is_checked_on_the_original_files_once(tmp_path, monkeypatch):
116    runtime = setup(tmp_path)
117    calls = []
118    original = testing._run_tests

120    async def counted(workspace, *args, **kwargs):
121        calls.append(workspace.root)
122        return await original(workspace, *args, **kwargs)
123
124    monkeypatch.setattr(testing, "_run_tests", counted)
125    await run(runtime)

131async def test_original_files_check_sees_the_real_workspace_path(tmp_path):
132    # Editable installs and fixtures address the workspace absolutely; the rerun on the

172def test_signature_ignores_temporary_paths_relative_to_the_copy():
173    a = ("AssertionError: 'Deleted /tmp/aster-x1/.tmp/django_pwt6bh_r/tmpjtmk4_ra/m/0002.py' not found in "
174         "\"Migrations for 'migrations':\\n  .tmp/django_pwt6bh_r/tmpjtmk4_ra/m/0002.py\"")
175    b = ("AssertionError: 'Deleted /tmp/aster-y9/.tmp/django_k2j1aa_q/tmp0v1dkj0z/m/0002.py' not found in "
176         "\"Migrations for 'migrations':\\n  .tmp/django_k2j1aa_q/tmp0v1dkj0z/m/0002.py\"")
177    assert failure_signature(a) == failure_signature(b)
178    # Ordinary relative paths and real values still distinguish failures.
179    assert failure_signature("E   AssertionError: src/a.py != src/b.py") != failure_signature("E   AssertionError: src/a.py != src/c.py")
