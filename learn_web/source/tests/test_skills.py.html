8@pytest.mark.parametrize("link_location", ["file", "directory", "parent", "internal_file"])
9def test_skills_reject_symlinks(tmp_path, link_location):
10    workspace = tmp_path / "repo"
11    workspace.mkdir()
12    target = workspace / "target" if link_location == "internal_file" else tmp_path / "outside"
13    target.mkdir()
14    (target / "hidden.md").write_text("must not load", encoding="utf-8")
15    if link_location == "directory":
16        (workspace / ".skills").symlink_to(target, target_is_directory=True)
17    elif link_location == "parent":
18        (target / "skills").mkdir()
19        (target / "skills" / "hidden.md").write_text("must not load", encoding="utf-8")
20        (workspace / ".aster").symlink_to(target, target_is_directory=True)
21    else:
22        (workspace / ".skills").mkdir()
23        (workspace / ".skills" / "hidden.md").symlink_to(target / "hidden.md")
24
25    skills = Skills(workspace)
26    assert skills.list() == []
27    with pytest.raises(FileNotFoundError):
28        skills.load("hidden")

31def test_skills_catalog_matches_regular_loadable_files(tmp_path):
32    root = tmp_path / ".skills"
33    root.mkdir()
34    (root / "valid.md").write_text("local instructions", encoding="utf-8")
35    (root / "directory.md").mkdir()

38    protected = tmp_path / ".aster" / "skills"
39    protected.mkdir(parents=True)
