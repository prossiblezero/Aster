1"""Requirements guessed from a request: an edit is demanded only when one is clearly asked for.
2
3Wrongly demanding an edit is the costly direction: the completion check would push the agent to
4change code the user only asked about. Negations count only when they forbid touching the
5workspace itself; constraints on the solution ("without changing existing values") do not make
6a request read-only. Every prompt here is a real phrasing class seen in use or in evaluations.
7"""
8
9import pytest
10
11from aster.core.verifier import Verifier
12
13pytestmark = pytest.mark.core
14
15QUESTIONS = [
16    "Explain how the dispatch table in router.py works.",  # "patch" inside a word
17    "Explain the suffix array implementation.",  # "fix" inside a word; a noun, not a request
18    "Explain the pytest fixture in conftest.py.",
19    "Explain why the cache does not update.",  # a negated verb is not a request
20    "解释一下这个函数是怎么实现的",  # "how was it implemented" is a question
21]
22
23READ_ONLY = [
24    "Inspect config.txt and tell me the mode. Do not modify files.",
25    "Review src/app.py without modifying any files.",
26    "Explain the bug; don't change anything, just tell me.",
27    "Make no changes. Explain why the test fails.",
28    "Read-only: summarize how the parser works.",
29    "请分析这个模块，不要修改。",
30    "不要对代码做任何修改，只解释原因",
31    "无需修改代码即可回答：这个参数是什么意思",
32    "只读分析这个仓库",
33]
34
35EDITS = [
36    # (request, verification expected)
37    ("Calculate the cumulative sum without changing existing values to NaN. "
38     "Implement task_func in solution.py.", False),  # BigCodeBench/966, inferred read-only before
39    ("Fix the dtype handling for transformers that do not modify the input values.", False),  # sklearn-25102
40    ("Write a function that sorts a list without modifying it.", False),
41    ("Implement task_func; do not modify the input DataFrame. Add tests.", True),
42    ("Fix the bug without changing the public API and run the tests.", True),
43    ("Fix the parser. Do not modify files other than parser.py.", False),
44    ("Fix the flaky test without touching the public API.", True),
45    ("Why does it not update? Fix it and run pytest.", True),
46    ("Rewrite the loader to stream files.", False),
47    ("实现这个函数，不要修改原始数据，并写测试", True),
48    ("修复这个 bug，不要修改测试文件", True),
49]

52@pytest.mark.parametrize("prompt", QUESTIONS + READ_ONLY)
53def test_questions_and_read_only_requests_never_demand_an_edit(prompt):
54    inferred = Verifier.infer(prompt)
55    assert not inferred["require_edit"] and not inferred["require_verification"]
56
57
58@pytest.mark.parametrize("prompt,verification", EDITS)
59def test_edit_requests_with_solution_constraints_still_demand_an_edit(prompt, verification):
60    inferred = Verifier.infer(prompt)
61    assert inferred["require_edit"] and inferred["require_observation"]
62    assert inferred["require_verification"] == verification
