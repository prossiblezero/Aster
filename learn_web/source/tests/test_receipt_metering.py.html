14def anchored():
15    state = RunState(task=TaskSpec(prompt="Inspect the code."))
16    builder = ContextBuilder(16000, 512)
17    prompt, _ = builder.build(state, state.task.prompt, model="test")
18    payload = request_payload(prompt, [], model="test", output_tokens=512,
19                              prompt_cache_key=state.session_id)
20    reply = ModelReply(usage={"usage_requests": 1, "input_tokens": 5000, "output_tokens": 80})
21    builder.calculator.observe(payload, reply, prompt)
22    return state, builder, prompt, payload

25def test_receipt_charges_only_new_observations_and_appended_context_notes():
26    _, builder, prompt, payload = anchored()
27    calculator = builder.calculator
28    additions = [
29        {"type": "reasoning", "encrypted_content": "opaque" * 100},
30        {"type": "function_call", "call_id": "c", "name": "read_file", "arguments": "{}"},
31        {"type": "function_call_output", "call_id": "c", "output": "source code"},
32        {"role": "assistant", "content": "Here is what I found."},
33        {"role": "user", "content": "Also check the callers."},
34    ]
35    tail = {"role": "user", "content": "new workflow state " * 100}
36    current = [*prompt, *additions, tail]
37    expected = (5080 + calculator.count(additions[2]) + calculator.count(additions[4])
38                + calculator.count(tail))
39    assert calculator.estimate(current, {**payload, "input": current}) == (expected, "receipt")
40    current[-1] = {"role": "user", "content": "short"}
41    assert calculator.estimate(current, {**payload, "input": current}) == (
42        5080 + calculator.count(additions[2]) + calculator.count(additions[4])
43        + calculator.count(current[-1]), "receipt")

46@pytest.mark.parametrize("change", ["replace_last", "remove_last"])
47def test_receipt_anchor_covers_the_final_input_item(change):
48    _, builder, prompt, payload = anchored()
49    current = deepcopy(prompt)
50    if change == "replace_last":
51        current[-1]["content"] = "replaced state"
52    else:
53        current.pop()
54    payload = {**payload, "input": current}
55    assert builder.calculator._first_changed_item(current) == len(prompt) - 1
56    assert builder.calculator.estimate(current, payload) == (
57        builder.calculator.request_tokens(payload), "local")

60def test_history_drift_is_located_before_falling_back_to_local_count():
61    state, builder, prompt, payload = anchored()
62    changed = deepcopy(prompt)
63    changed[1]["content"] = "A different task"
64    payload = {**payload, "input": changed}
65    assert builder.calculator.estimate(changed, payload) == (
66        builder.calculator.request_tokens(payload), "local")
67    _, report = builder.build(state, "A different task", model="test")
68    assert report.prefix_drift and report.first_changed_item == 1

93async def test_continuation_keeps_anchor_but_new_session_or_model_clears_it(tmp_path):
94    class ReceiptModel:
95        async def complete(self, messages, tools):
96            return ModelReply(text="Inspected.", usage={
97                "usage_requests": 1, "input_tokens": 5000, "output_tokens": 80})
98
99    runtime = Runtime(tmp_path, Settings(completion="off", memory_enabled=False), model=ReceiptModel())
100    await runtime.run(TaskSpec(prompt="Inspect the code."))
101    calculator = runtime.context.calculator
102    runtime.continue_task("Inspect the callers too.")
103    _, _, report = runtime.prepare_request()
104    assert report.estimate_basis == "receipt" and not report.prefix_drift
105    runtime.settings.model = "other"
106    assert runtime.prepare_request()[2].estimate_basis == "local"
107    assert calculator._anchor is None
108    await runtime.run()
109    runtime.start(TaskSpec(prompt="A separate task."))
110    assert runtime.prepare_request()[2].estimate_basis == "local"
111    assert calculator._anchor is None
