1"""Harness bookkeeping stays proportional to what changed, not to repository size.
2
3Counts are deterministic, so this guards the v12 overhead fixes in CI without timing noise:
4before them, 16 scripted steps on this repository hashed 136 MB and called fsync 137 times."""

21def test_scripted_run_hashes_and_syncs_within_budget(tmp_path, monkeypatch):
22    spec = importlib.util.spec_from_file_location("bench_harness_overhead", BENCH)
23    bench = importlib.util.module_from_spec(spec)
24    spec.loader.exec_module(bench)
25    counts = {"hashed": 0, "fsync": 0}
26
27    class CountingSha256:
28        def __init__(self, data=b""):
29            self.inner = hashlib.sha256()
30            self.update(data)
31
32        def update(self, data):
33            counts["hashed"] += len(data)
34            self.inner.update(data)
35
36        def hexdigest(self):
37            return self.inner.hexdigest()
38
39    real_fsync = os.fsync
40
41    def counting_fsync(fd):
42        counts["fsync"] += 1
43        return real_fsync(fd)
44
45    # Files just written are inside the racy window and would always be rehashed.
46    monkeypatch.setattr(workspace_module, "RACY_WINDOW_NS", 0, raising=False)
47    monkeypatch.setattr(workspace_module, "hashlib", type("Hashlib", (), {"sha256": CountingSha256}))
48    monkeypatch.setattr(os, "fsync", counting_fsync)
49    bench.build_repo(tmp_path, 200, 4)
50    size = sum(p.stat().st_size for p in tmp_path.rglob("*") if p.is_file())
51    _, state = asyncio.run(bench.run(tmp_path, 16, "off"))
52    assert state.phase == "completed" and state.turn == 17
53    assert counts["hashed"] < 6 * size, counts  # about 4x today: initial snapshots only
54    assert counts["fsync"] < 4 * state.turn, counts  # group commit; about 3 per turn today
