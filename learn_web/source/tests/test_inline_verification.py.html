10@pytest.mark.parametrize("command,code,status,reason", [
11    ("pytest -q", "from subject import answer\ndef test_answer(): assert answer() == 42\n", "success", ""),
12    ("python -m pytest -q", "from subject import answer\ndef test_answer(): assert answer() == 43\n", "error", "tests_failed"),
13    ("python -m unittest -q", "import unittest\nfrom subject import answer\nclass Checks(unittest.TestCase):\n    def test_answer(self): self.assertEqual(answer(), 42)\n", "success", ""),
14    ("pytest -q", "print('10 tests passed')\n", "error", "no_tests_collected"),
15    ("pytest -q", "import pytest\n@pytest.mark.skip\ndef test_skipped(): pass\n", "blocked", "no_tests_executed"),
16    ("pytest -q", "from pathlib import Path\ndef test_modify_source():\n    Path('subject.py').write_text('tampered')\n", "blocked", "read_only"),
17    ("python -m unittest discover", "pass", "blocked", "inline_test_command_not_supported"),
18])
19async def test_inline_checks_execute_against_project_without_publishing_files(tmp_path, command, code, status, reason):
20    (tmp_path / "subject.py").write_text("def answer(): return 42\n")
21    runtime = Runtime(tmp_path, Settings(approval="auto", memory_enabled=False))
22    runtime.start(TaskSpec(prompt="Check subject.py"))
23    before = runtime.workspace.snapshot()
24    outcome = await runtime.step(ToolCall(name="run_tests", arguments={"command": command, "code": code}))
25    assert (outcome.status, outcome.reason) == (status, reason), outcome.content
26    assert (runtime.state.verification.state == "passed") == (status == "success")
27    assert runtime.workspace.snapshot() == before and not outcome.changed_paths
28    if status == "success":
29        assert outcome.data["passed_tests"] == 1
30        assert outcome.data["tested_workspace_hash"] == runtime.workspace.fingerprint()
31    runtime.sandbox.close()

34@pytest.mark.core

36async def test_blocked_report_retains_failures_and_cannot_clear_old_failures(tmp_path, unittest):
37    (tmp_path / "subject.py").write_text("VALUE = 0\n")
38    runtime = Runtime(tmp_path, Settings(approval="auto", memory_enabled=False))
39    runtime.start(TaskSpec(prompt="Verify", require_verification=True))
40    command = "pytest -q"
41    first = "def test_old(): assert False\n"
42    if unittest:
43        command = "python -m unittest -q"
44        first = "import unittest\nclass Checks(unittest.TestCase):\n    def test_old(self): raise RuntimeError('broken')\n"
45    await runtime.step(ToolCall(name="run_tests", arguments={"command": command, "code": first}))
46    code = (
47        "from pathlib import Path\n"
48        "def test_old(): assert True\n"
49        "def test_new(): assert False\n"
50        "def test_mutation(): Path('subject.py').write_text('tampered')\n"
51    )
52    if unittest:
53        code = (
54            "import unittest\nfrom pathlib import Path\nclass Checks(unittest.TestCase):\n"
55            "    def test_old(self): pass\n"
56            "    def test_new(self): raise RuntimeError('broken')\n"
57            "    def test_mutation(self): Path('subject.py').write_text('tampered')\n"
58        )
59    blocked = await runtime.step(ToolCall(name="run_tests", arguments={"command": command, "code": code}))
60    assert (blocked.status, blocked.reason) == ("blocked", "read_only")
61    assert all(status == "failed" for status in blocked.data["test_results"].values())
62    assert len(runtime.state.verification_failures) == 2
63    await runtime.step(ToolCall(name="run_tests", arguments={
64        "command": "pytest -q", "code": "def test_unrelated(): assert True\n",
65    }))
66    assert runtime.verifier.final(runtime.state).reason == "verification_unresolved_failures"
67    assert (tmp_path / "subject.py").read_text() == "VALUE = 0\n"
68    runtime.sandbox.close()
