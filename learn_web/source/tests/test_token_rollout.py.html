19def test_hermes_and_xml_tool_calls_parse_with_schema_types():
20    hermes = parse_completion('Let me look.\n<tool_call>\n{"name": "read_file", "arguments": {"path": "a.py"}}\n</tool_call>')
21    assert hermes.text == "Let me look." and hermes.calls == [("read_file", {"path": "a.py"})]
22    xml = parse_completion("<think>plan</think>\n<tool_call>\n<function=read_file>\n<parameter=path>\na.py\n"
23                           "</parameter>\n<parameter=start>\n3\n</parameter>\n</function>\n</tool_call>", TOOLS)
24    assert xml.calls == [("read_file", {"path": "a.py", "start": 3})] and xml.reasoning == "plan"
25    assert parse_completion("plan</think>Done.").text == "Done."  # template opened <think>

39class CharTokenizer:
40    """Stand-in for a chat template: one token per character of a JSON rendering."""
41
42    def apply_chat_template(self, conversation, *, tools=None, add_generation_prompt=False, tokenize=True, **_):
43        text = json.dumps({"messages": conversation, "tools": [t["function"]["name"] for t in tools]}) + "<|assistant|>"
44        return [ord(c) for c in text]
45
46    def decode(self, token_ids, **_):
47        return "".join(map(chr, token_ids))

50def scripted(*completions):
51    queue = list(completions)
52
53    async def generate(prompt_ids):
54        if not queue:
55            raise RuntimeError("generation server unavailable")
56        text = queue.pop(0)
57        return Generation([ord(c) for c in text], [-0.1] * len(text), "stop")
58    return generate

70@pytest.mark.asyncio
71async def test_episode_records_exact_tokens_and_survives_a_malformed_turn(tmp_path):
72    generate = scripted(
73        "<tool_call>\n<function=read_file>\n<parameter=path>\napp.py\n</parameter>\n</function>\n</tool_call>",
74        '<tool_call>{"name": "patch_file", "arguments": {"path": "app.py", "old_text": ',  # cut off
75        'Fixing.<tool_call>{"name": "patch_file", "arguments": {"path": "app.py", '
76        '"old_text": "VALUE = 0", "new_text": "VALUE = 1"}}</tool_call>',
77        "VALUE is now 1.",
78    )
79    trajectory = await rollout(env_for(tmp_path), "Set VALUE to 1 in app.py", CharTokenizer(), generate)
80    assert [bool(t.parse_errors) for t in trajectory.turns] == [False, True, False, False]
81    assert trajectory.label == "success" and trajectory.reward == 1.0 and trajectory.trainable
82    for turn in trajectory.turns:
83        assert len(turn.logprobs) == len(turn.completion_ids)
84        assert sum(turn.loss_mask) == len(turn.completion_ids)
85    # The malformed reply and its error are what the policy sees next.
86    assert "could not be executed" in CharTokenizer().decode(trajectory.turns[2].prompt_ids)
