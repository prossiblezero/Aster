25def test_rules_deny_precedence():

29            ExecRule(prefix=["git", "push"], decision="deny", reason="external write"),

37async def test_real_isolation_and_sync(tmp_path):
38    workspace = tmp_path / "repo"
39    workspace.mkdir()
40    secret = tmp_path / "host_secret.txt"
41    secret.write_text("fake-secret")
42    (workspace / "a.py").write_text("original")
43    runner = SandboxRunner(Settings(approval="auto"))
44    result = await runner.run(Workspace(workspace), "printf changed > a.py")
45    assert result.status == "success", result.model_dump()
46    assert result.data["isolation"] == "landlock+seccomp"
47    assert (workspace / "a.py").read_text() == "changed"
48    result = await runner.run(Workspace(workspace), f"cat {shlex.quote(str(secret))}")
49    assert result.status == "error"
50    assert "fake-secret" not in result.content
51    result = await runner.run(Workspace(workspace), "python -c 'import socket; socket.socket()'")
52    assert result.status == "error"
53    assert "Operation not permitted" in result.content
