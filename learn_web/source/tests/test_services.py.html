23def free_port():
24    with socket.socket() as s:

29def runtime_for(root):
30    runtime = Runtime(root, Settings(approval="auto", memory_enabled=False, sandbox="required", network=True,
31                                     completion="off"), approval_callback=lambda *_: True)

36async def call(runtime, tool, **arguments):

40def ping(port):
41    with socket.create_connection(("127.0.0.1", port), timeout=3) as conn:
42        return conn.recv(4)

45@pytest.mark.asyncio
46@pytest.mark.parametrize("keep", [False, True])
47async def test_service_survives_tool_calls_and_ends_with_the_task_unless_kept(tmp_path, keep):
48    port = free_port()
49    (tmp_path / "server.py").write_text(SERVER.replace("PORT", str(port)))
50    runtime = runtime_for(tmp_path)
51    started = await call(runtime, "service_start", name="web", command="python3 server.py", keep_running=keep)
52    assert started.status == "success" and "running" in started.content and "ready" in started.content
53    await call(runtime, "run_shell", command="echo another command")  # would kill a plain background job
54    assert ping(port) == b"pong"
55    status = await call(runtime, "service_status", name="web")
56    assert "web: running" in status.content
57    runtime.finalize("Server is up.")
58    await asyncio.sleep(0.5)
59    if keep:
60        assert ping(port) == b"pong" and "Services left running: web" in runtime.state.final
61        runtime.sandbox.stop_service("web")
62    else:
63        with pytest.raises(OSError):
64            ping(port)

67@pytest.mark.asyncio
68async def test_a_failing_service_reports_its_output(tmp_path):
69    runtime = runtime_for(tmp_path)
70    result = await call(runtime, "service_start", name="bad", command="python3 -c 'raise SystemExit(\"boom\")'")
71    assert result.reason == "service_exited" and "boom" in result.content
