174def test_workspace_snapshot_and_diff_track_permissions_not_mtime(tmp_path):
175    path = tmp_path / "script.sh"
176    path.write_text("true\n")
177    path.chmod(0o644)
178    workspace = Workspace(tmp_path)
179    baseline, text, modes = workspace.snapshot(), workspace.text_snapshot(), {"script.sh": 0o644}
180    stamp = path.stat().st_mtime + 60
181    os.utime(path, (stamp, stamp))
182    assert workspace.snapshot() == baseline
183    path.chmod(0o755)
184    assert Workspace.changes(baseline, workspace.snapshot()) == ["script.sh"]
185    diff = workspace.diff(text, modes)
186    assert "old mode 100644" in diff and "new mode 100755" in diff
