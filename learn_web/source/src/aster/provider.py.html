26def history_groups(history):
27    """Validate complete tool exchanges and keep their reasoning in the same cut group."""
28    groups, group, pending, seen, returned = [], [], set(), set(), set()
29    accepting_calls = False
30    for item in history:
31        kind, role = item.get("type"), item.get("role")
32        calls, outputs = [], []
33        if kind == "function_call":
34            calls = [item.get("call_id")]
35        elif kind == "function_call_output":
36            outputs = [item.get("call_id")]
37        if item.get("tool_calls"):
38            if role != "assistant":
39                raise ContextHistoryError("invalid_tool_role: calls require an assistant message")
40            calls.extend(call.get("id") for call in item["tool_calls"])
41        if role == "tool":
42            outputs.append(item.get("tool_call_id"))
43        content = item.get("content")
44        if isinstance(content, list):
45            saw_other_block = False
46            for block in content:
47                if not isinstance(block, dict):
48                    continue
49                if block.get("type") == "tool_use":
50                    if role != "assistant":
51                        raise ContextHistoryError(
52                            "invalid_tool_role: tool_use requires an assistant message"
53                        )
54                    calls.append(block.get("id"))
55                elif block.get("type") == "tool_result":
56                    if role != "user" or saw_other_block:
57                        raise ContextHistoryError(
58                            "invalid_tool_order: tool_result must lead a user message"
59                        )
60                    outputs.append(block.get("tool_use_id"))
61                else:
62                    saw_other_block = True
63        if any(not isinstance(key, str) or not key.strip() for key in calls + outputs):
64            raise ContextHistoryError("invalid_tool_id: tool calls and results need nonempty ids")
65        if len(calls) != len(set(calls)) or seen.intersection(calls):
66            raise ContextHistoryError("duplicate_tool_call: history contains a reused call id")
67        if calls and outputs:
68            raise ContextHistoryError(
69                "invalid_tool_order: calls and results cannot share a message"
70            )
71        if outputs:
72            for key in outputs:
73                if key in returned:
74                    raise ContextHistoryError("duplicate_tool_output: each call needs one result")
75                if key not in pending:
76                    raise ContextHistoryError(
77                        "orphan_tool_output: history must retain complete exchanges"
78                    )
79                pending.remove(key)
80                returned.add(key)
81            if role == "user" and pending:
82                raise ContextHistoryError("missing_tool_output: native result batch is incomplete")
83            group.append(item)
84            accepting_calls = False
85            if not pending:
86                groups.append(group)
87                group = []
88            continue
89        if pending and not (kind == "function_call" and accepting_calls):
90            raise ContextHistoryError("missing_tool_output: next turn starts before results")
91        new_turn = kind == "reasoning" or role in {"user", "assistant", "system"}
92        if group and not pending and new_turn:
93            # Contiguous reasoning belongs to the following assistant response/call batch.
94            if role in {"user", "system"} or any(x.get("type") != "reasoning" for x in group):
95                groups.append(group)
96                group = []
97        pending.update(calls)
98        seen.update(calls)
99        accepting_calls = bool(calls) and kind == "function_call"
100        group.append(item)
101        if role in {"user", "system"}:
102            groups.append(group)
103            group = []
104    if pending:
105        raise ContextHistoryError("missing_tool_output: history ends before results")
106    if group:
107        groups.append(group)
108    return groups

157def _numeric_usage(raw):
158    """Keep known counters, never arbitrary strings or provider-supplied mapping keys."""
159
160    def counts(value):
161        if not isinstance(value, dict):
162            return {}
163        return {
164            key: value[key]
165            for key in sorted(_USAGE_FIELDS)
166            if type(value.get(key)) is int and 0 <= value[key] <= 2**63 - 1
167        }
168
169    result = counts(raw)
170    if not isinstance(raw, dict):
171        return result
172    for key in (
173        "input_tokens_details",
174        "output_tokens_details",
175        "prompt_tokens_details",
176        "completion_tokens_details",
177        "cache_creation",
178    ):
179        if isinstance(raw.get(key), dict):
180            result[key] = counts(raw[key])
181    attribution = raw.get("attribution")
182    if isinstance(attribution, dict):
183        result["attribution"] = attr = {}
184        items = attribution.get("items")
185        if isinstance(items, dict):
186            attr["items_count"] = len(items)
187            # IDs and content are deliberately omitted; top-level item counters suffice.
188            attr["items"] = [counts(value) for value in islice(items.values(), 128)]
189        fields = attribution.get("request_fields")
190        if isinstance(fields, dict):
191            attr["request_fields_count"] = len(fields)
192            attr["request_fields"] = {
193                key: counts(fields[key])
194                for key in ("tools", "instructions", "messages", "input", "system")
195                if key in fields
196            }
197    return result

200def _usage_warnings(raw, usage):
201    flags = []
202    for left, right in (("input_tokens", "prompt_tokens"), ("output_tokens", "completion_tokens")):
203        if left in raw and right in raw and raw[left] != raw[right]:
204            flags.append(left + "_aliases_disagree")
205    if all(key in usage for key in ("input_tokens", "output_tokens", "total_tokens")):
206        if usage["total_tokens"] != usage["input_tokens"] + usage["output_tokens"]:
207            flags.append("total_tokens_disagree")
208    for part, total in (
209        ("cache_read_input_tokens", "input_tokens"),
210        ("reasoning_tokens", "output_tokens"),
211    ):
212        if part in usage and total in usage and usage[part] > usage[total]:
213            flags.append(part + "_exceeds_" + total)
214    attr = raw.get("attribution", {})
215    items, fields = attr.get("items", []), attr.get("request_fields", {})
216    if (
217        items
218        and len(items) == attr.get("items_count")
219        and len(fields) == attr.get("request_fields_count")
220        and all("input_tokens" in part for part in [*items, *fields.values()])
221        and "input_tokens" in usage
222    ):
223        total = sum(part["input_tokens"] for part in [*items, *fields.values()])
224        if total != usage["input_tokens"]:
225            flags.append("input_attribution_disagrees")
226    return flags

229def _validated_reply(text, calls, raw, data, usage, *, opaque=False):
230    if not isinstance(text, str) or not isinstance(data.get("model", ""), str):
231        raise ValueError("invalid response text/model")
232    ids = [call.call_id for call in calls]
233    if len(ids) != len(set(ids)) or any(not key.strip() for key in ids):
234        raise ValueError("duplicate or empty tool call ID")
235    if not text.strip() and not calls:
236        raise ProviderError(
237            "Response contains no supported text or tool call"
238            if opaque
239            else "Provider returned an empty response",
240            category="unsupported_output" if opaque else "empty_response",
241            usage=usage,
242        )
243    return ModelReply(
244        text=text, calls=calls, usage=usage or {}, raw=raw, provider_model=data.get("model", "")
245    )

258def _chat_messages(messages):
259    result, images = [], []
260    for item in messages:
261        kind = item.get("type")
262        if kind == "function_call_output":
263            output = item.get("output", "")
264            result.append(
265                {"role": "tool", "tool_call_id": item["call_id"], "content": _text(output)}
266            )
267            if isinstance(output, list):
268                images.extend(
269                    {"type": "image_url", "image_url": {"url": block["image_url"]}}
270                    for block in output
271                    if block.get("type") == "input_image"
272                )
273        elif kind == "function_call":
274            call = {
275                "id": item["call_id"],
276                "type": "function",
277                "function": {
278                    "name": item["name"],
279                    "arguments": item["arguments"],
280                },
281            }
282            if result and result[-1].get("role") == "assistant":
283                result[-1].setdefault("tool_calls", []).append(call)
284            else:
285                result.append({"role": "assistant", "content": None, "tool_calls": [call]})
286        elif kind != "reasoning":
287            message = {
288                key: value
289                for key, value in item.items()
290                if key
291                in {
292                    "role",
293                    "content",
294                    "tool_calls",
295                    "tool_call_id",
296                    "reasoning_content",
297                    "reasoning",
298                }
299            }
300            if isinstance(message.get("content"), list):
301                message["content"] = _text(message["content"])
302            result.append(message)
303    if images:
304        result.append({"role": "user", "content": images})
305    return result

308def _anthropic_messages(messages):
309    system, result = [], []
310    for item in messages:
311        if item.get("role") == "system":
312            system.append(_text(item.get("content", "")))
313            continue
314        kind = item.get("type")
315        if kind == "function_call_output":
316            output = item.get("output", "")
317            content = output
318            if isinstance(output, list):
319                content = []
320                for block in output:
321                    if block.get("type") == "input_image":
322                        media, data = block["image_url"].split(";base64,", 1)
323                        content.append(
324                            {
325                                "type": "image",
326                                "source": {
327                                    "type": "base64",
328                                    "media_type": media.removeprefix("data:"),
329                                    "data": data,
330                                },
331                            }
332                        )
333                    else:
334                        content.append({"type": "text", "text": block.get("text", "")})
335            role, blocks = (
336                "user",
337                [{"type": "tool_result", "tool_use_id": item["call_id"], "content": content}],
338            )
339        elif kind == "function_call":
340            role, blocks = (
341                "assistant",
342                [
343                    {
344                        "type": "tool_use",
345                        "id": item["call_id"],
346                        "name": item["name"],
347                        "input": json.loads(item["arguments"]),
348                    }
349                ],
350            )
351        elif kind == "reasoning":
352            continue  # Responses reasoning is opaque and cannot be translated across providers.
353        else:
354            role = item.get("role", "assistant")
355            content = item.get("content", "")
356            blocks = [{"type": "text", "text": content}] if isinstance(content, str) else content
357        if result and result[-1]["role"] == role:
358            result[-1]["content"].extend(blocks)
359        else:
360            result.append({"role": role, "content": list(blocks)})
361    return "\n\n".join(system), result

364def request_payload(messages, tools, *, protocol="responses", model="", output_tokens=2048):
365    """Shared by ContextBuilder and the transport, so budgeting sees the actual request shape."""
366    history_groups(messages)
367    if protocol in {"chat", "anthropic"} and any(
368        item.get("phase") is not None for item in messages
369    ):
370        # These protocols have no equivalent assistant-phase field. Fail preflight
371        # instead of turning intermediate commentary into an unlabelled answer.
372        raise ContextHistoryError("unsupported_history_phase_for_protocol")
373    if protocol == "chat":
374        history = _chat_messages(messages)
375        payload = {
376            "model": model,
377            "messages": history,
378            "max_tokens": output_tokens,
379            "tools": [
380                {
381                    "type": "function",
382                    "function": {key: value for key, value in tool.items() if key != "type"},
383                }
384                for tool in tools
385            ],
386        }
387    elif protocol == "anthropic":
388        system, history = _anthropic_messages(messages)
389        payload = {
390            "model": model,
391            "system": system,
392            "messages": history,
393            "max_tokens": output_tokens,
394            "tools": [
395                {
396                    "name": t["name"],
397                    "description": t["description"],
398                    "input_schema": t["parameters"],
399                }
400                for t in tools
401            ],
402        }
403    else:
404        payload = {
405            "model": model,
406            "input": messages,
407            "tools": tools,
408            "max_output_tokens": output_tokens,
409            "store": False,
410        }
411    history_groups(payload.get("input", payload.get("messages", [])))
412    return payload

415class ModelClient:
416    def __init__(self, settings: Settings):
417        self.settings = settings
418        self.last_usage: dict[str, int] = {}
419        self.last_diagnostics: dict[str, Any] = {}
420        self.redaction_secrets = [settings.api_key.get_secret_value()]
421
422    async def complete(
423        self, messages: list[dict[str, Any]], tools: list[dict], *, on_event=None
424    ) -> ModelReply:
425        protocol = self.settings.protocol
426        if self.settings.stream and protocol != "responses":
427            raise ProviderError("Streaming currently requires Responses", category="configuration")
428        payload = request_payload(
429            messages,
430            tools,
431            protocol=protocol,
432            model=self.settings.model,
433            output_tokens=self.settings.effective_output_tokens,
434        )
435        suffix = {"responses": "/responses", "chat": "/chat/completions", "anthropic": "/messages"}[
436            protocol
437        ]
438        headers = (
439            {
440                "x-api-key": self.settings.api_key.get_secret_value(),
441                "anthropic-version": "2023-06-01",
442            }
443            if protocol == "anthropic"
444            else {}
445        )
446        if self.settings.stream:
447            from .streaming import complete_responses
448
449            return await complete_responses(self, payload, headers, on_event)
450        data = await self._post(suffix, payload, headers, on_event=on_event)
451        reply = self._parse(data, self.last_usage)
452        reply.diagnostics = self.last_diagnostics
453        return reply

455    def _parse(self, data, usage):
456        try:
457            if not isinstance(data, dict):
458                raise ValueError("response must be an object")
459            return {
460                "responses": self._responses_reply,
461                "chat": self._chat_reply,
462                "anthropic": self._anthropic_reply,
463            }[self.settings.protocol](data, usage)
464        except (ValueError, KeyError, TypeError, IndexError, AttributeError) as exc:
465            raise ProviderError(
466                "Malformed model response or tool arguments",
467                category="invalid_response",
468                usage=usage,
469            ) from exc

471    def _record_usage(self, data):
472        usage = self._usage(data.get("usage"), self.settings.protocol)
473        for key, value in usage.items():
474            self.last_usage[key] = self.last_usage.get(key, 0) + value
475        return usage

477    def _diagnostic(self, payload, data, response, attempt, started, category, usage, callback):
478        """Only bounded metadata and allowlisted numeric usage leave this boundary."""
479        redactor = Redactor(self.redaction_secrets)
480
481        def label(value, limit=160):
482            if not isinstance(value, str):
483                return ""
484            value = redactor.text(value)
485            value = re.sub(r"https?://\S+", "[provider URL]", value)
486            value = re.sub(
487                r"(?i)\b(password|api[_ -]?key|authorization|secret)\s*[:=]\s*\S+",
488                r"\1=[REDACTED]",
489                value,
490            )
491            return value[:limit]
492
493        data = data if isinstance(data, dict) else {}
494        raw = _numeric_usage(data.get("usage"))
495        returned_model = data.get("model")
496        flags = _usage_warnings(raw, usage)
497        if isinstance(returned_model, str) and returned_model != self.settings.model:
498            flags.append("response_model_mismatch")
499        structure = []
500        terminal_status = data.get("status", data.get("stop_reason", ""))
501        output = data.get("output", data.get("content", []))
502        if self.settings.protocol == "chat":
503            choices = data.get("choices")
504            if isinstance(choices, list) and choices and isinstance(choices[0], dict):
505                terminal_status = choices[0].get("finish_reason", "")
506            output = (
507                [c.get("message") for c in choices[:32] if isinstance(c, dict)]
508                if isinstance(choices, list)
509                else []
510            )
511        if isinstance(output, list):
512            for item in output[:32]:
513                if not isinstance(item, dict):
514                    structure.append({"type": "malformed"})
515                    continue
516                row = {"type": label(item.get("type", item.get("role", "unknown")), 80)}
517                for key in (
518                    "text",
519                    "arguments",
520                    "thinking",
521                    "encrypted_content",
522                    "content",
523                    "reasoning_content",
524                    "reasoning",
525                ):
526                    if isinstance(item.get(key), str):
527                        row[key + "_length"] = len(item[key])
528                tool_calls = item.get("tool_calls")
529                if isinstance(tool_calls, list):
530                    row["tool_call_count"] = len(tool_calls)
531                    row["tool_calls"] = [
532                        {
533                            "type": label(call.get("type"), 80),
534                            "arguments_length": len(call["function"]["arguments"])
535                            if isinstance(call.get("function"), dict)
536                            and isinstance(call["function"].get("arguments"), str)
537                            else 0,
538                        }
539                        for call in tool_calls[:32]
540                        if isinstance(call, dict)
541                    ]
542                if isinstance(item.get("input"), dict):
543                    row["argument_keys_count"] = len(item["input"])
544                content = item.get("content")
545                if isinstance(content, list):
546                    row["content_count"] = len(content)
547                    row["content"] = [
548                        {
549                            "type": label(block.get("type"), 80),
550                            "text_length": len(block["text"])
551                            if isinstance(block.get("text"), str)
552                            else 0,
553                        }
554                        for block in content[:32]
555                        if isinstance(block, dict)
556                    ]
557                structure.append(row)
558        diagnostic = {
559            "requested_model": label(self.settings.model),
560            "response_model": label(returned_model),
561            "protocol": self.settings.protocol,
562            "response_id": label(data.get("id")),
563            "request_ids": {
564                key: label(response.headers[key])
565                for key in ("x-request-id", "request-id", "anthropic-request-id")
566                if response is not None and key in response.headers
567            },
568            # httpx uses this same JSON encoder for the actual json= request body.
569            "payload_sha256": hashlib.sha256(
570                httpx.Request("POST", "https://unused", json=payload).content
571            ).hexdigest(),
572            "attempt": attempt,
573            "elapsed_seconds": round(time.monotonic() - started, 6),
574            "classification": category,
575            "terminal_status": label(terminal_status, 80),
576            "output_count": len(output) if isinstance(output, list) else None,
577            "output_structure": structure,
578            "output_text_length": len(data["output_text"])
579            if isinstance(data.get("output_text"), str)
580            else 0,
581            "raw_usage": raw,
582            "usage": dict(usage),
583            "usage_observed": isinstance(data.get("usage"), dict),
584            "flags": flags,
585        }
586        self.last_diagnostics = diagnostic
587        if callback is not None:
588            try:
589                callback(deepcopy({"type": "diagnostic", **diagnostic}))
590            except Exception as exc:
591                logging.getLogger(__name__).warning(
592                    "Provider observer failed: %s", type(exc).__name__
593                )
594        return diagnostic

596    async def _post(self, suffix: str, payload: dict, extra: dict, *, on_event=None) -> dict:
597        secret = self.settings.api_key.get_secret_value()
598        if not secret and self.settings.auth != "none":
599            raise ProviderError("ASTER_API_KEY is empty", category="configuration")
600        self.last_usage, self.last_diagnostics = {}, {}
601        headers = {"Content-Type": "application/json", **extra}
602        if self.settings.auth != "none":
603            headers["Authorization"] = f"Bearer {secret}"
604        else:
605            headers.pop("x-api-key", None)
606        url = self.settings.base_url.rstrip("/") + suffix
607        redactor = Redactor([*self.redaction_secrets, secret])
608        empty_retries = 0
609        async with httpx.AsyncClient(timeout=self.settings.request_timeout) as client:
610            for attempt in range(1, 4):
611                started = time.monotonic()
612                data, response, usage = None, None, {}
613                delay = 0.25 * 2 ** (attempt - 1)
614                try:
615                    response = await client.post(url, headers=headers, json=payload)
616                    try:
617                        data = response.json()
618                    except ValueError:
619                        data = None
620                    if isinstance(data, dict) and (response.is_success or "usage" in data):
621                        usage = self._record_usage(data)
622                    if (
623                        response.is_success
624                        and isinstance(data, dict)
625                        and data
626                        and not data.get("error")
627                    ):
628                        self._parse(data, usage)  # Validate the entire batch before publication.
629                        self._diagnostic(
630                            payload, data, response, attempt, started, "success", usage, on_event
631                        )
632                        return data
633                    raw = data.get("error", {}) if isinstance(data, dict) else {}
634                    raw = raw if isinstance(raw, dict) else {"message": str(raw)}
635                    detail = redactor.text(
636                        str(raw.get("message", "Provider returned no valid JSON response"))
637                    )
638                    detail = re.sub(r"https?://\S+", "[provider URL]", detail)
639                    detail = re.sub(
640                        r"(?i)\b(password|api[_ -]?key|authorization|secret)\s*[:=]\s*\S+",
641                        r"\1=[REDACTED]",
642                        detail,
643                    )
644                    code = redactor.text(str(raw.get("code", raw.get("type", ""))))[:120]
645                    code = code if re.fullmatch(r"[a-zA-Z0-9_.:-]*", code) else ""
646                    retryable = response.status_code in {
647                        408,
648                        409,
649                        425,
650                        429,
651                        500,
652                        502,
653                        503,
654                        504,
655                        524,
656                    }
657                    raise ProviderError(
658                        detail[:800],
659                        category="transient_http"
660                        if retryable
661                        else ("invalid_response" if response.is_success else "request_rejected"),
662                        status=response.status_code,
663                        code=code,
664                    )
665                except asyncio.CancelledError:
666                    self._diagnostic(
667                        payload, data, response, attempt, started, "cancelled", usage, on_event
668                    )
669                    raise
670                except (
671                    httpx.TimeoutException,
672                    httpx.NetworkError,
673                    httpx.RemoteProtocolError,
674                ) as exc:
675                    error = ProviderError(type(exc).__name__, category="transport")
676                except ProviderError as exc:
677                    error = exc
678                diagnostic = self._diagnostic(
679                    payload,
680                    data,
681                    response,
682                    attempt,
683                    started,
684                    error.details["category"],
685                    usage,
686                    on_event,
687                )
688                error = ProviderError(
689                    **{
690                        **error.details,
691                        "attempts": attempt,
692                        "usage": self.last_usage,
693                        "diagnostics": diagnostic,
694                    }
695                )
696                category = error.details["category"]
697                retry = category in {"transport", "transient_http"}
698                if category == "empty_response" and empty_retries == 0:
699                    empty_retries += 1
700                    retry = True
701                if not retry or attempt == 3:
702                    raise error
703                if response is not None:
704                    try:
705                        delay = min(2, max(delay, float(response.headers.get("retry-after", "0"))))
706                    except ValueError:
707                        pass
708                await asyncio.sleep(delay)
709        raise error  # pragma: no cover

711    @staticmethod
712    def _usage(raw: dict, protocol: str = "responses") -> dict[str, int]:
713        """Known token counts plus per-metric coverage; absent is unknown, not zero."""
714        raw = raw if isinstance(raw, dict) else {}
715
716        def find(*path):
717            value = raw
718            for key in path:
719                value = value.get(key) if isinstance(value, dict) else None
720            return value if type(value) is int and 0 <= value <= 2**63 - 1 else None
721
722        # Some compatible gateways use Responses names in Chat usage envelopes.
723        inputs, outputs = find("input_tokens"), find("output_tokens")
724        if inputs is None:
725            inputs = find("prompt_tokens")
726        if outputs is None:
727            outputs = find("completion_tokens")
728        counts = {"input_tokens": inputs, "output_tokens": outputs}
729        if protocol == "anthropic":
730            counts.update(
731                uncached_input_tokens=inputs,
732                cache_read_input_tokens=find("cache_read_input_tokens"),
733                cache_creation_input_tokens=find("cache_creation_input_tokens"),
734            )
735            components = [
736                counts[key]
737                for key in (
738                    "uncached_input_tokens",
739                    "cache_read_input_tokens",
740                    "cache_creation_input_tokens",
741                )
742            ]
743            # Anthropic cache tokens are additional input; OpenAI cache tokens are a subset.
744            counts["input_tokens"] = (
745                sum(components) if all(value is not None for value in components) else None
746            )
747        else:
748            input_details = (
749                "prompt_tokens_details" if protocol == "chat" else "input_tokens_details"
750            )
751            output_details = (
752                "completion_tokens_details" if protocol == "chat" else "output_tokens_details"
753            )
754            counts["cache_read_input_tokens"] = find(input_details, "cached_tokens")
755            counts["reasoning_tokens"] = find(output_details, "reasoning_tokens")
756        total = find("total_tokens")
757        if total is None and counts["input_tokens"] is not None and outputs is not None:
758            total = counts["input_tokens"] + outputs
759        counts["total_tokens"] = total
760        usage = {"usage_requests": 1}
761        for name, value in counts.items():
762            if value is not None:
763                usage[name] = value
764                usage[f"{name}_requests"] = 1
765        return usage

767    @staticmethod
768    def _check_completion(data, protocol, usage):
769        if protocol == "responses":
770            reason = data.get("status", "")
771            if reason not in {
772                None,
773                "",
774                "completed",
775                "incomplete",
776                "failed",
777                "cancelled",
778                "queued",
779                "in_progress",
780            }:
781                raise ValueError("unknown response status")
782            incomplete = reason in {"incomplete", "failed", "cancelled", "queued", "in_progress"}
783            if reason == "incomplete":
784                details = data.get("incomplete_details") or {}
785                cause = details.get("reason") if isinstance(details, dict) else None
786                if cause in {"max_output_tokens", "content_filter"}:
787                    reason = cause
788        elif protocol == "chat":
789            reason = data["choices"][0].get("finish_reason", "")
790            if reason not in {
791                None,
792                "",
793                "stop",
794                "tool_calls",
795                "length",
796                "content_filter",
797                "function_call",
798            }:
799                raise ValueError("unknown finish reason")
800            incomplete = reason in {"length", "content_filter"}
801        else:
802            reason = data.get("stop_reason", "")
803            if reason not in {
804                None,
805                "",
806                "end_turn",
807                "tool_use",
808                "stop_sequence",
809                "max_tokens",
810                "pause_turn",
811                "refusal",
812            }:
813                raise ValueError("unknown stop reason")
814            incomplete = reason in {"max_tokens", "pause_turn"}
815        if incomplete:
816            raise ProviderError(
817                "Provider stopped before a complete model response",
818                category="incomplete_response",
819                code=f"{protocol}:{reason}",
820                usage=usage,
821            )

823    @staticmethod
824    def _responses_reply(data, usage=None):
825        ModelClient._check_completion(data, "responses", usage)
826        calls, texts, opaque = [], [], False
827        output = data.get("output", [])
828        if not isinstance(output, list) or data.get("error"):
829            raise ValueError("invalid output")
830        for item in output:
831            if not isinstance(item, dict) or item.get("status") not in {None, "completed"}:
832                raise ValueError("invalid output item")
833            kind = item.get("type")
834            if kind == "function_call":
835                calls.append(
836                    ToolCall(
837                        name=item["name"],
838                        arguments=json.loads(item["arguments"]),
839                        call_id=item.get("call_id", item.get("id", "")),
840                    )
841                )
842            elif kind == "message":
843                if item.get("role", "assistant") != "assistant" or not isinstance(
844                    item.get("content"), list
845                ):
846                    raise ValueError("invalid output message")
847                for block in item["content"]:
848                    if not isinstance(block, dict):
849                        raise ValueError("invalid content block")
850                    if block.get("type") == "refusal":
851                        _refusal(usage)
852                    if block.get("type") not in {"output_text", "text"}:
853                        raise ProviderError(
854                            "Unsupported response content",
855                            category="unsupported_output",
856                            usage=usage,
857                        )
858                    if not isinstance(block.get("text"), str):
859                        raise ValueError("invalid text")
860                    texts.append(block["text"])
861            elif kind == "reasoning":
862                opaque = True
863            else:
864                raise ProviderError(
865                    "Unsupported response output", category="unsupported_output", usage=usage
866                )
867        text = "".join(texts)
868        # Legacy compatible endpoints may supply only the top-level output_text shortcut.
869        if not output:
870            text = data.get("output_text", "")
871        return _validated_reply(text, calls, output, data, usage, opaque=opaque)

873    @staticmethod
874    def _chat_reply(data, usage=None):
875        ModelClient._check_completion(data, "chat", usage)
876        choices = data["choices"]
877        if not isinstance(choices, list) or len(choices) != 1:
878            raise ValueError("expected one response choice")
879        message = choices[0]["message"]
880        if message.get("role", "assistant") != "assistant":
881            raise ValueError("invalid assistant role")
882        if message.get("refusal"):
883            _refusal(usage)
884        if (
885            message.get("audio")
886            or message.get("function_call")
887            or isinstance(message.get("content"), list)
888        ):
889            raise ProviderError(
890                "Unsupported chat output", category="unsupported_output", usage=usage
891            )
892        calls = []
893        for item in message.get("tool_calls", []):
894            if item.get("type", "function") != "function":
895                raise ProviderError(
896                    "Unsupported chat tool type", category="unsupported_output", usage=usage
897                )
898            calls.append(
899                ToolCall(
900                    name=item["function"]["name"],
901                    arguments=json.loads(item["function"]["arguments"]),
902                    call_id=item["id"],
903                )
904            )
905        return _validated_reply(
906            message.get("content") if message.get("content") is not None else "",
907            calls,
908            [message],
909            data,
910            usage,
911            opaque=bool(message.get("reasoning_content") or message.get("reasoning")),
912        )

914    @staticmethod
915    def _anthropic_reply(data, usage=None):
916        ModelClient._check_completion(data, "anthropic", usage)
917        if data.get("stop_reason") == "refusal":
918            _refusal(usage)
919        text, calls, opaque = "", [], False
920        content = data.get("content", [])
921        if not isinstance(content, list):
922            raise ValueError("invalid content")
923        for block in content:
924            kind = block.get("type")
925            if kind == "text":
926                text += block["text"]
927            elif kind == "tool_use":
928                calls.append(
929                    ToolCall(name=block["name"], arguments=block["input"], call_id=block["id"])
930                )
931            elif kind in {"thinking", "redacted_thinking"}:
932                opaque = True
933            else:
934                raise ProviderError(
935                    "Unsupported Anthropic output", category="unsupported_output", usage=usage
936                )
937        return _validated_reply(
938            text, calls, [{"role": "assistant", "content": content}], data, usage, opaque=opaque
939        )
