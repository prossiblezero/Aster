19def create_server(root: Path, *, allow_write=False):
20    from mcp.server import MCPServer
21    from mcp.server.mcpserver.exceptions import ToolError
22    from mcp.types import ToolAnnotations
23
24    if not root.is_dir():
25        raise ValueError("filesystem MCP root must be an existing directory")
26    workspace = Workspace(root)
27    server = MCPServer("aster-filesystem", log_level="ERROR")
28    read = ToolAnnotations(readOnlyHint=True, openWorldHint=False)
29    write = ToolAnnotations(readOnlyHint=False, destructiveHint=True, openWorldHint=False)
30
31    def tool(annotations):
32        def register(fn):
33            @wraps(fn)
34            def checked(*args, **kwargs):
35                try:
36                    return fn(*args, **kwargs)
37                except (OSError, ValueError) as exc:
38                    # Expected path/freshness errors must reach the model as corrective feedback.
39                    raise ToolError(str(exc)) from None
40
41            return server.tool(annotations=annotations)(checked)
42        return register
43
44    @contextmanager
45    def parent(raw: str, *, writing=False):
46        target = workspace.path(raw, write=writing)
47        parts = target.relative_to(workspace.root).parts
48        fd = os.open(workspace.root, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
49        try:
50            for part in parts[:-1]:
51                child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
52                os.close(fd)
53                fd = child
54            yield fd, parts[-1] if parts else "."
55        finally:
56            os.close(fd)
57
58    def file_bytes(fd: int, name: str) -> bytes:
59        handle = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=fd)
60        with os.fdopen(handle, "rb") as stream:
61            info = os.fstat(stream.fileno())
62            if not stat.S_ISREG(info.st_mode) or info.st_nlink != 1:
63                raise ValueError("only regular files without hard links are allowed")
64            data = stream.read(MAX_FILE_BYTES + 1)
65            if len(data) > MAX_FILE_BYTES:
66                raise ValueError("file exceeds the 1 MiB limit")
67            return data
68
69    @tool(read)
70    def list_directory(path: str = ".", offset: int = 0, limit: int = 100) -> dict:
71        """List one directory in pages. Paths are relative to the configured root."""
72        if offset < 0 or not 1 <= limit <= 500:
73            raise ValueError("offset must be nonnegative; limit must be 1..500")
74        with parent(path) as (fd, name):
75            directory = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
76            try:
77                entries = []
78                for child in sorted(os.listdir(directory)):
79                    relative = str(Path(path) / child)
80                    try:
81                        workspace.path(relative)
82                    except ValueError:
83                        continue
84                    info = os.stat(child, dir_fd=directory, follow_symlinks=False)
85                    if stat.S_ISLNK(info.st_mode) or (stat.S_ISREG(info.st_mode) and info.st_nlink != 1):
86                        continue
87                    if not (stat.S_ISREG(info.st_mode) or stat.S_ISDIR(info.st_mode)):
88                        continue
89                    entries.append({"path": relative, "type": "directory" if stat.S_ISDIR(info.st_mode) else "file"})
90            finally:
91                os.close(directory)
92        end = offset + limit
93        return {"entries": entries[offset:end], "total": len(entries),
94                "next_offset": end if end < len(entries) else None}
95
96    @tool(read)
97    def read_file(path: str, start: int = 1, limit: int = 200) -> dict:
98        """Read UTF-8 lines and a whole-file sha256 for subsequent edits; at most 1 MiB."""
99        if start < 1 or not 1 <= limit <= 2000:
100            raise ValueError("start must be positive; limit must be 1..2000")
101        with parent(path) as (fd, name):
102            data = file_bytes(fd, name)
103        lines = data.decode("utf-8").splitlines(keepends=True)
104        end = min(start - 1 + limit, len(lines))
105        return {"path": path, "text": "".join(lines[start - 1:end]),
106                "start": start, "next_start": end + 1 if end < len(lines) else None,
107                "total_lines": len(lines), "sha256": hashlib.sha256(data).hexdigest()}
108
109    if allow_write:
110        @tool(write)
111        def write_file(path: str, content: str, expected_sha256: str | None = None) -> dict:
112            """Create a UTF-8 file, or replace it using the sha256 returned by read_file.
113
114            Parent directories must exist. Omitting expected_sha256 means create-only.
115            Concurrent external edits must be coordinated by the caller.
116            """
117            data = content.encode("utf-8")
118            if len(data) > MAX_FILE_BYTES:
119                raise ValueError("file exceeds the 1 MiB limit")
120            with parent(path, writing=True) as (fd, name):
121                mode = 0o600
122                if expected_sha256 is not None:
123                    current = file_bytes(fd, name)
124                    if hashlib.sha256(current).hexdigest() != expected_sha256:
125                        raise ValueError("file changed; read_file again before editing")
126                    mode = stat.S_IMODE(os.stat(name, dir_fd=fd, follow_symlinks=False).st_mode) & 0o777
127                temporary = ".mcp-write-" + uuid4().hex
128                handle = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW,
129                                 mode, dir_fd=fd)
130                try:
131                    with os.fdopen(handle, "wb") as stream:
132                        stream.write(data)
133                        stream.flush()
134                        os.fsync(stream.fileno())
135                    if expected_sha256 is None:
136                        # link is atomic create-only, even if another writer creates the target.
137                        os.link(temporary, name, src_dir_fd=fd, dst_dir_fd=fd, follow_symlinks=False)
138                    else:
139                        os.replace(temporary, name, src_dir_fd=fd, dst_dir_fd=fd)
140                finally:
141                    try:
142                        os.unlink(temporary, dir_fd=fd)
143                    except FileNotFoundError:
144                        pass
145            return {"path": path, "sha256": hashlib.sha256(data).hexdigest()}
146
147        @tool(write)
148        def create_directory(path: str) -> dict:
149            """Create one directory under an existing parent. Refuses existing paths."""
150            with parent(path, writing=True) as (fd, name):
151                os.mkdir(name, mode=0o700, dir_fd=fd)
152            return {"path": path}
153
154        @tool(write)
155        def delete_file(path: str, expected_sha256: str) -> dict:
156            """Delete one previously read file after checking its sha256. Never deletes directories."""
157            with parent(path, writing=True) as (fd, name):
158                data = file_bytes(fd, name)
159                if hashlib.sha256(data).hexdigest() != expected_sha256:
160                    raise ValueError("file changed; read_file again before deleting")
161                os.unlink(name, dir_fd=fd)
162            return {"path": path, "deleted": True}
163
164    return server

167def main(argv=None):
168    parser = argparse.ArgumentParser(description=__doc__)
169    parser.add_argument("--root", type=Path, required=True, help="Existing allowed directory")
170    parser.add_argument("--allow-write", action="store_true", help="Expose create, write and delete tools")
171    args = parser.parse_args(argv)
172    create_server(args.root, allow_write=args.allow_write).run(transport="stdio")
