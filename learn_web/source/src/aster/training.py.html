44def validate_splits(tasks: list[TaskRecord]) -> None:
45    """The same repository or issue family cannot cross the frozen splits."""
46    identifiers, repositories, families = set(), {}, {}
47    for task in tasks:
48        if task.task_id in identifiers:
49            raise ValueError("duplicate task id")
50        identifiers.add(task.task_id)
51        for key, assignments in ((task.repository, repositories), (task.issue_family, families)):
52            if key in assignments and assignments[key] != task.split:
53                raise ValueError("repository or issue-family split leakage")
54            assignments[key] = task.split

73def _event_hash(event: dict) -> str:
74    body = {key: value for key, value in event.items() if key != "hash"}
75    expected = digest(json.dumps(body, sort_keys=True, ensure_ascii=False))
76    if event.get("hash") != expected:
77        raise ValueError("event hash mismatch")
78    return expected

81def prepare_sft(
82    artifact: dict,
83    context_event: dict,
84    reply_event: dict,
85    *,
86    task: TaskRecord,
87    provider: str,
88    protocol: Literal["responses", "chat", "anthropic"],
89) -> SFTRecord:
90    """Join the actual saved request with its successful assistant event, not a checkpoint.
91
92    The original artifacts stay in the audit store. This portable projection
93    drops opaque input reasoning and exports only visible text/structured calls.
94    Tokenization and alignment to a target model remain the caller's obligation.
95    """
96    if task.split != "train":
97        raise ValueError("SFT export accepts only admitted train tasks")
98    _event_hash(context_event)
99    completion_hash = _event_hash(reply_event)
100    if context_event.get("event") != "context_built" or reply_event.get("event") != "model_reply":
101        raise ValueError("request must pair with a completed model_reply event")
102    run_id, turn = reply_event.get("run_id"), reply_event.get("turn")
103    if not isinstance(run_id, str) or type(turn) is not int or turn < 1:
104        raise ValueError("invalid completion identity")
105    payload = artifact.get("payload")
106    if not isinstance(payload, dict):
107        raise ValueError("missing actual request payload")
108    request_hash = digest(json.dumps(payload, sort_keys=True, ensure_ascii=False))
109    if (
110        context_event.get("run_id") != run_id
111        or context_event.get("request_sha256") != request_hash
112        or reply_event.get("request_sha256") != request_hash
113        or context_event.get("policy_version") != artifact.get("policy_version")
114        or context_event.get("request_artifact")
115        != f"runs/{run_id}/requests/{turn:04d}-{request_hash}.json"
116    ):
117        raise ValueError("request/completion provenance mismatch")
118    messages = deepcopy(payload.get("input" if protocol == "responses" else "messages"))
119    if not isinstance(messages, list):
120        raise ValueError("missing actual request messages")
121    if protocol == "anthropic" and payload.get("system"):
122        messages = [{"role": "system", "content": payload["system"]}, *messages]
123    converted = canonicalize(
124        messages, provider=provider, protocol=protocol, model=payload.get("model")
125    )
126    text, calls = reply_event.get("text"), reply_event.get("calls")
127    if not isinstance(text, str) or not isinstance(calls, list):
128        raise ValueError("invalid assistant target")
129    target = [{"role": "assistant", "content": text}] if text else []
130    seen = set()
131    for raw in calls:
132        call = ToolCall.model_validate(raw)
133        if not call.call_id.strip() or call.call_id in seen:
134            raise ValueError("duplicate or empty assistant tool id")
135        seen.add(call.call_id)
136        target.append(
137            {
138                "type": "function_call",
139                "call_id": call.call_id,
140                "name": call.name,
141                "arguments": json.dumps(call.arguments, ensure_ascii=False, allow_nan=False),
142            }
143        )
144    return SFTRecord(
145        task=task,
146        run_id=run_id,
147        turn=turn,
148        request_sha256=request_hash,
149        completion_event_sha256=completion_hash,
150        provider=provider,
151        protocol=protocol,
152        teacher_model=payload["model"],
153        context_policy_version=artifact["policy_version"],
154        prompt_messages=converted.messages,
155        source_tools=deepcopy(payload.get("tools", [])),
156        assistant_target=target,
157        omitted_prompt_reasoning=converted.omitted_reasoning,
158    )

161class TokenizedDecision(Record):
162    request_sha256: Hash
163    rendered_sample_sha256: Hash
164    input_ids: list[Token] = Field(min_length=2)
165    prompt_length: int = Field(ge=1)
166    assistant_mask: list[StrictBool]
167
168    @model_validator(mode="after")
169    def aligned(self):
170        if self.prompt_length >= len(self.input_ids) or len(self.assistant_mask) != len(
171            self.input_ids
172        ):
173            raise ValueError("token and mask lengths must match a nonempty completion")
174        if any(self.assistant_mask[: self.prompt_length]) or not any(self.assistant_mask):
175            raise ValueError("only the target assistant completion may be supervised")
176        return self
177
178    @property
179    def labels(self) -> list[int]:
180        return [
181            token if learn else -100
182            for token, learn in zip(self.input_ids, self.assistant_mask, strict=True)
183        ]

186class PolicyDecision(TokenizedDecision):
187    behavior_logprobs: list[FiniteFloat]
188    reference_logprobs: list[FiniteFloat]
189
190    @model_validator(mode="after")
191    def probabilities(self):
192        length = len(self.input_ids) - self.prompt_length
193        if any(
194            len(values) != length for values in (self.behavior_logprobs, self.reference_logprobs)
195        ):
196            raise ValueError(
197                "logprobs must align with every generated token, including masked tokens"
198            )
199        if any(value > 0 for value in self.behavior_logprobs + self.reference_logprobs):
200            raise ValueError("log probabilities cannot be positive")
201        return self

222class GRPORollout(Record):
223    task: TaskRecord
224    run_id: Name
225    group_id: Name
226    sample_index: int = Field(ge=0)
227    seed: Annotated[StrictInt, Field(ge=0)]
228    versions: RolloutVersions
229    decisions: list[PolicyDecision] = Field(min_length=1)
230    termination: Literal[
231        "success",
232        "candidate_failure",
233        "budget_exhausted",
234        "parse_error",
235        "cancelled",
236        "infrastructure_error",
237        "verifier_error",
238    ]
239    reward: FiniteFloat | None
240    independent_verifier: StrictBool
241    max_steps: int = Field(ge=1)
242    max_turns: int = Field(ge=1)
243    max_generated_tokens: int = Field(ge=1)
244    max_seconds: FiniteFloat = Field(gt=0)
245    tool_steps: int = Field(ge=0)
246    elapsed_seconds: FiniteFloat = Field(ge=0)
247
248    @model_validator(mode="after")
249    def eligible(self):
250        if self.task.split != "train":
251            raise ValueError("training rollout requires an admitted train task")
252        if self.reward is not None and not self.independent_verifier:
253            raise ValueError("a training reward requires an independent verifier")
254        if (
255            self.termination in {"cancelled", "infrastructure_error", "verifier_error"}
256            and self.reward is not None
257        ):
258            raise ValueError("operational failures must remain ungraded")
259        generated = sum(len(item.input_ids) - item.prompt_length for item in self.decisions)
260        if self.tool_steps > len(self.decisions):
261            raise ValueError("single-action decisions allow at most one tool step each")
262        if (
263            self.tool_steps > self.max_steps
264            or len(self.decisions) > self.max_turns
265            or generated > self.max_generated_tokens
266        ):
267            raise ValueError("rollout exceeds the frozen action/token budget")
268        if self.elapsed_seconds > self.max_seconds and self.termination != "budget_exhausted":
269            raise ValueError("wall-clock exhaustion needs an explicit budget termination")
270        return self

273def validate_group(rollouts: list[GRPORollout]) -> None:
274    if len(rollouts) < 2 or any(item.reward is None for item in rollouts):
275        raise ValueError("a GRPO update group requires at least two graded samples")
276    first = rollouts[0]
277    fields = (
278        "task",
279        "group_id",
280        "versions",
281        "max_steps",
282        "max_turns",
283        "max_generated_tokens",
284        "max_seconds",
285    )
286    if any(
287        any(getattr(item, field) != getattr(first, field) for field in fields) for item in rollouts
288    ):
289        raise ValueError("GRPO group must freeze task, versions, sampling and budgets")
290    if len({item.sample_index for item in rollouts}) != len(rollouts) or len(
291        {item.run_id for item in rollouts}
292    ) != len(rollouts):
293        raise ValueError("duplicate rollout identity in group")
