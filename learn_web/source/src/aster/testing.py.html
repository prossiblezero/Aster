5import xml.etree.ElementTree as ET

8def parse_junit(text: str) -> list[dict[str, str]]:
9    root = ET.fromstring(text)
10    if root.tag not in {"testsuites", "testsuite"}:
11        raise ValueError("invalid_junit_report")
12    cases = []
13    for case in root.iter("testcase"):
14        if not case.get("name"):
15            raise ValueError("unnamed_junit_test")
16        children = {child.tag for child in case}
17        status = next(
18            (
19                result
20                for tag, result in (
21                    ("error", "error"),
22                    ("failure", "failed"),
23                    ("skipped", "skipped"),
24                )
25                if tag in children
26            ),
27            "passed",
28        )
29        cases.append(
30            {"name": case.get("name"), "classname": case.get("classname", ""), "status": status}
31        )
32    return cases
