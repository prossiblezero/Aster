21def safety_identity(identity: dict) -> dict:
22    identity = {"read_parallelism": 1, "reuse_environment": False, **identity}
23    model_fields = {
24        "provider",
25        "profile",
26        "model",
27        "protocol",
28        "auth",
29        "stream",
30        "context_tokens",
31        "output_tokens",
32    }
33    return {key: value for key, value in identity.items() if key not in model_fields}

36def redact_skill_snapshots(state: RunState, redactor: Redactor) -> None:
37    # New model credentials may match text selected before the switch.
38    if any(redactor.text(name) != name for name in {*state.active_skills, *state.skill_snapshots}):
39        raise ValueError("skill_name_contains_credential")
40    for name, snapshot in state.skill_snapshots.items():
41        body = redactor.text(snapshot.body)
42        if body != snapshot.body:
43            state.skill_snapshots[name] = SkillSnapshot(
44                body=body, sha256=hashlib.sha256(body.encode()).hexdigest()
45            )

48def archive_native_history(
49    evidence: EvidenceStore,
50    run_id: str,
51    history: list[dict],
52    converted: CanonicalHistory,
53    origins: list[dict],
54) -> dict:
55    artifact = ("runs", run_id, f"native-history-{len(origins) + 1}.json")
56    evidence.save(
57        artifact,
58        {"source": converted.source, "structured_history": history},
59        preserve_native_history=True,
60    )
61    return {
62        "artifact": "/".join(artifact),
63        "source": converted.source,
64        "omitted_reasoning": converted.omitted_reasoning,
65    }

103def prepare_continuation(
104    previous: RunState,
105    history: list[dict],
106    verifier: Verifier,
107    *,
108    feedback: str | None,
109    drift: list[str],
110    max_steps: int | None,
111    max_turns: int | None,
112) -> tuple[RunState, list[dict]]:
113    """Copy a conversation into a fresh bounded run without mutating its predecessor."""
114    state = previous.model_copy(deep=True)
115    state.run_id = uuid4().hex
116    state.phase, state.stop_reason, state.final, state.feedback = "running", "", "", ""
117    state.step = state.turn = state.final_revisions = 0
118    state.usage = {}
119    history = deepcopy(history)
120    if feedback is not None:
121        inferred = verifier.infer(feedback)
122        unfinished = previous.phase != "completed"
123        if unfinished:
124            inferred = {
125                key: value or getattr(previous.task, key) for key, value in inferred.items()
126            }
127        if inferred["require_edit"] and previous.task.require_verification:
128            inferred["require_verification"] = True
129        state.task = TaskSpec(
130            prompt=previous.task.prompt,
131            max_steps=previous.task.max_steps,
132            max_turns=previous.task.max_turns,
133            required_artifacts=previous.task.required_artifacts
134            if inferred["require_edit"] or unfinished
135            else [],
136            **inferred,
137        )
138        if not unfinished:
139            state.observed, state.changed_paths = False, []
140        history.append({"role": "user", "content": feedback})
141        state.messages.append({"role": "user", "content": feedback, "important": True})
142    limits = {
143        key: value
144        for key, value in {"max_steps": max_steps, "max_turns": max_turns}.items()
145        if value is not None
146    }
147    if limits:
148        state.task = TaskSpec.model_validate({**state.task.model_dump(), **limits})
149    if feedback is not None or drift:
150        state.verification = Verification()
151        state.read_evidence, state.working_memory, state.recalled_ids = {}, {}, []
152    if drift:
153        note = (
154            "User adopted external workspace edits; read current files before editing: "
155            + ", ".join(drift)
156        )
157        history.append({"role": "user", "content": note})
158    return state, history

172def load_checkpoint(
173    evidence: EvidenceStore,
174    workspace: Workspace,
175    settings: Settings,
176    run_id: str,
177    *,
178    allow_model_change: bool,
179    adopt_workspace: bool,
180) -> SessionCheckpoint:
181    """Validate persisted identity, history and archive ownership before runtime mutation."""
182    data = json.loads(evidence.path("runs", run_id, "checkpoint.json").read_text())
183    if data.get("checkpoint_version", 1) not in {1, 2}:
184        raise ValueError("unsupported_checkpoint_version: use the Aster version that saved this run")
185    assert_settled_tools(evidence, run_id, data)
186    # Opt-in workers share the workspace. Adopting edits cannot erase a child's
187    # uncertain effects merely because worker_start already returned to the parent.
188    for event in evidence.events(run_id):
189        if event["event"] == "worker_started":
190            child = event["child_run_id"]
191            child_data = json.loads(evidence.path("runs", child, "checkpoint.json").read_text())
192            if child_data["state"].get("parent_run_id") != run_id:
193                raise ValueError("worker_checkpoint_identity_mismatch")
194            assert_settled_tools(evidence, child, child_data)
195    current_identity = settings.identity()
196    # Old checkpoints predate named providers and streaming; their defaults are explicit.
197    saved_identity = {
198        "provider": "openai",
199        "profile": None,
200        "auth": "required",
201        "stream": False,
202        "read_parallelism": 1,
203        "reuse_environment": False,
204        **data["runtime_identity"],
205    }
206    model_changed = saved_identity != current_identity
207    if model_changed and (
208        not allow_model_change
209        or safety_identity(saved_identity) != safety_identity(current_identity)
210    ):
211        differences = ", ".join(
212            key for key in sorted(set(saved_identity) | set(current_identity))
213            if saved_identity.get(key) != current_identity.get(key)
214        )
215        raise ValueError(
216            "runtime_identity_mismatch: " + differences
217            + ". Restore these explicit settings from the saved checkpoint; "
218            "--allow-model-change permits only model settings, not safety or capability changes."
219        )
220    if data["workspace_hash"] != workspace.fingerprint() and not adopt_workspace:
221        raise ValueError("workspace_mismatch")
222    state = RunState.model_validate(data["state"])
223    redact_skill_snapshots(state, evidence.redactor)
224    state.messages = evidence.redactor.value(state.messages, mask_fields=False)
225    if not state.terminal and "initial_text" not in data:
226        raise ValueError("legacy_checkpoint_missing_diff_baseline")
227    history_groups(data.get("structured_history", []))
228    evidence.validate_native_history(data.get("structured_history", []))
229    history = evidence.redact_history(data.get("structured_history", []))
230    sources = data.get("archive_sources", {})
231    if not isinstance(sources, dict):
232        raise ValueError("invalid_archive_sources")
233    for observation_id, source_run in sources.items():
234        source = json.loads(evidence.path("runs", source_run, "checkpoint.json").read_text())
235        if source["state"]["session_id"] != state.session_id or safety_identity(
236            source["runtime_identity"]
237        ) != safety_identity(current_identity):
238            raise ValueError("archive_source_identity_mismatch")
239        evidence.read_observation(source_run, observation_id, limit=1)
240    return SessionCheckpoint(
241        data=data,
242        state=state,
243        history=history,
244        sources=sources,
245        origins=list(data.get("history_origins", [])),
246        saved_identity=saved_identity,
247        model_changed=model_changed,
248    )

251def assert_settled_tools(evidence: EvidenceStore, run_id: str, data: dict) -> None:
252    marker = evidence.path("runs", run_id, "inflight.json")
253    if not marker.exists():
254        return
255    pending = json.loads(marker.read_text())
256    answered = {
257        item.get("call_id")
258        for item in data.get("structured_history", [])
259        if item.get("type") == "function_call_output"
260    }
261    calls = pending.get("calls", [])
262    if not calls or any(call.get("call_id") not in answered for call in calls):
263        raise ValueError(
264            "interrupted_tool_batch: tool effects may have happened without a saved result. "
265            "Inspect the workspace and .aster/runs/" + run_id
266            + "/ events and inflight.json; start a new task with the confirmed state. "
267            "Resume will not replay uncertain actions; --adopt-workspace cannot bypass this."
268        )

271def read_session_observation(
272    evidence: EvidenceStore, run_id: str, sources: dict[str, str], args: dict
273) -> Outcome:
274    """Read only this run and explicitly authorized ancestor observation IDs."""
275    try:
276        if args["observation_id"]:
277            page = evidence.read_observation(
278                sources.get(args["observation_id"], run_id),
279                args["observation_id"],
280                offset=args["offset"],
281                limit=args["limit"],
282                query=args["query"],
283            )
284        else:
285            if not sources:
286                page = evidence.list_observations(
287                    run_id,
288                    offset=args["offset"],
289                    limit=min(10, args["limit"]),
290                    query=args["query"],
291                )
292            else:
293                records = {}
294                for source_run in [run_id, *sorted(set(sources.values()))]:
295                    offset = 0
296                    while True:
297                        listing = evidence.list_observations(
298                            source_run, offset=offset, limit=20, query=args["query"]
299                        )
300                        for ref in listing["observations"]:
301                            if (
302                                source_run == run_id
303                                or sources.get(ref["id"]) == source_run
304                            ):
305                                records[ref["id"]] = ref
306                        offset = listing["next_offset"]
307                        if offset is None:
308                            break
309                ordered = [records[key] for key in sorted(records)]
310                start, end = args["offset"], args["offset"] + min(10, args["limit"])
311                page = {
312                    "observations": ordered[start:end],
313                    "offset": start,
314                    "next_offset": end if end < len(ordered) else None,
315                    "total": len(ordered),
316                }
317    except (OSError, ValueError) as exc:
318        return Outcome(status="blocked", reason="observation_unavailable", content=str(exc))
319    # Wire previews need only the locator and paging fields; full provenance
320    # stays in the archive and trace rather than every prompt.
321    if args["observation_id"]:
322        page = {
323            key: value
324            for key, value in page.items()
325            if key not in {"sha256", "tool", "call_id"}
326        }
327    else:
328        page["observations"] = [
329            {key: value for key, value in ref.items() if key != "sha256"}
330            for ref in page["observations"]
331        ]
332    return Outcome(status="success", content=json.dumps(page, ensure_ascii=False))
