1"""What the model sees of a tool result, rendered from the harness's full record.
2
3The structured history keeps one JSON record per tool output: status, reason, evidence and
4archive references, paging cursors and byte counts. Audit, recovery, context budgeting and
5training read that record. The model needs much less: what came back, whether it failed,
6and the few cursors or references it must quote to act next. Sending the record itself cost
7about 111 tokens per tool result on fresh30b (pi: 2), resent on every later request, and
8JSON-escaped every line of code and test output.
9
10``render_output`` turns a record into plain text: an optional bracketed status line, the
11content verbatim, and an optional bracketed footer. It is pure and deterministic, so the
12rendered prefix stays stable for provider prompt caching, and a record that is already plain
13text passes through unchanged.
14"""
15
16from __future__ import annotations
17
18import json
19import re
20
21# Results the model may cite as evidence (memory_store), so they keep a short reference.
22CITABLE = {"read_file", "run_shell", "run_tests"}
23# Their own result already shows what changed.
24SELF_DESCRIBING_CHANGES = {"write_file", "patch_file"}
25PASSTHROUGH_KEYS = ("idempotency_replayed", "original_status", "uncertain", "mcp_endpoint", "failover")
26# Archive ids are shown shortened; observation_read resolves unique prefixes (storage/evidence.py).
27SHOWN_ID = 16
28ARCHIVE_ID = re.compile(r"(observation_id=|observation_read id=)([0-9a-f]{64})\b")
29LISTED_ID = re.compile(r'("id": ")([0-9a-f]{64})(?=")')
30EXCERPT_MARKER = "[excerpt; observation_read for archived content]"
31EMPTY = {"list_files": "[no files]", "search": "[no matches]", "git_diff": "[no changes]",
32         "read_file": "[empty file]", "run_shell": "[no output]", "run_tests": "[no output]"}
33

35def output_record(value) -> dict:
36    if isinstance(value, str):
37        try:
38            record = json.loads(value)
39        except ValueError:
40            record = None
41        if isinstance(record, dict) and "content" in record:
42            return record
43    return {"content": value}

46def render_output(value, tool: str = "", *, references: bool = False):
47    """Plain text for the model; the full record stays in the harness."""
48    record = output_record(value)
49    content = record.get("content", "")
50    if len(record) == 1 or not isinstance(content, str):
51        return value  # already plain text, or structured media blocks
52    head = []
53    status, reason = record.get("status"), record.get("reason")
54    if status and status != "success":
55        head.append(status + (f": {reason}" if reason else ""))
56    elif reason:
57        head.append(reason)
58    changed = record.get("changed_paths") or []
59    if changed and tool not in SELF_DESCRIBING_CHANGES:
60        head.append("changed: " + ", ".join(changed[:8]) + (f" (+{len(changed) - 8} more)" if len(changed) > 8 else ""))
61    head.extend(f"{key}: {record[key]}" for key in PASSTHROUGH_KEYS if key in record)
62    if record.get("updated"):
63        head.append(f"updated: version {str(record.get('previous_version'))[:12]} -> {str(record.get('version'))[:12]}")
64    if record.get("source_ref"):
65        head.append(f"source_ref {record['source_ref']}")
66    elif references and tool in CITABLE and record.get("evidence_id"):
67        head.append(f"evidence {record['evidence_id']}")
68    tail = []
69    if tool == "read_file" and record.get("truncated") and record.get("next_start"):
70        tail.append(f"more from line {record['next_start']}")
71    if record.get("truncated") and record.get("next_offset") is not None:
72        shown = f"{record['count']} of {record['total']} shown; " if "count" in record and "total" in record else ""
73        tail.append(f"{shown}next_offset={record['next_offset']}")
74    if record.get("output_truncated"):
75        tail.append(f"output truncated: kept {record.get('captured_bytes')} of {record.get('output_bytes')} bytes")
76    observation = record.get("observation")
77    if observation and EXCERPT_MARKER in content:
78        # Only a body clipped to fit the context needs the way back; a complete one does not.
79        tail.append(f"full output: observation_read observation_id={observation['id'][:SHOWN_ID]}")
80    content = ARCHIVE_ID.sub(lambda m: m.group(1) + m.group(2)[:SHOWN_ID], content)
81    if tool == "observation_read":
82        content = LISTED_ID.sub(lambda m: m.group(1) + m.group(2)[:SHOWN_ID], content)
83    if not content and status in (None, "success"):
84        # An empty success must still say what it means: "[done]" after a search reads like
85        # success, not like "nothing matched".
86        content = EMPTY.get(tool) or ("[no results]" if record.get("count") == 0 else "[done]")
87    parts = (["[" + "; ".join(head) + "]"] if head else []) + ([content] if content else []) + (
88        ["[" + "; ".join(tail) + "]"] if tail else [])
89    return "\n".join(parts) if parts else "[done]"

92def render_workflow(state: dict) -> str:
93    """The workflow-state context note as a few readable lines instead of a JSON object."""
94    lines = []
95    if state.get("mode") and state["mode"] != "work":
96        lines.append(f"mode: {state['mode']}")
97    if state.get("plan"):
98        lines.append("plan: " + state["plan"])
99    for todo in state.get("unfinished_todos") or []:
100        lines.append(f"todo {todo.get('id', '')} [{todo.get('status', '')}]: {todo.get('content', '')}"
101                     + (f" (acceptance: {todo['acceptance']})" if todo.get("acceptance") else "")
102                     + (f" (resolution: {todo['resolution']})" if todo.get("resolution") else ""))
103    if state.get("active_skills"):
104        versions = state.get("active_skill_versions") or {}
105        lines.append("active skills: " + ", ".join(
106            f"{name}@{versions[name][:12]}" if name in versions else name for name in state["active_skills"]))
107    changed = state.get("changed_paths") or []
108    if changed:
109        count = state.get("changed_path_count", len(changed))
110        lines.append("changed files: " + ", ".join(changed) + (f" (+{count - len(changed)} more)" if count > len(changed) else ""))
111    verification = state.get("verification") or {}
112    if verification.get("state") and (changed or verification["state"] != "missing"):
113        lines.append("verification: " + verification["state"]
114                     + (f" ({verification['command']})" if verification.get("command") else ""))
115    budget = state.get("budget") or {}
116    if budget.get("level") == "critical":
117        lines.append(f"budget: critical, {budget.get('remaining_turns')} turns and "
118                     f"{budget.get('remaining_steps')} tool steps left"
119                     + (f", {budget['remaining_team_tokens']} tokens" if "remaining_team_tokens" in budget else ""))
120    elif budget.get("level"):
121        lines.append(f"budget: {budget['level']} (at most {round(budget.get('remaining_fraction_at_most', 0) * 100)}% left)")
122    for key, value in state.items():
123        if key in {"budget", "mode", "plan", "unfinished_todos", "active_skills", "active_skill_versions",
124                   "changed_paths", "changed_path_count", "verification"} or value in (None, "", [], {}):
125            continue
126        if isinstance(value, list):
127            lines.append(f"{key.replace('_', ' ')}: " + "; ".join(
128                v if isinstance(v, str) else json.dumps(v, ensure_ascii=False) for v in value))
129        elif isinstance(value, dict):
130            lines.append(f"{key.replace('_', ' ')}: " + "; ".join(
131                f"{k} ({v})" if v else k for k, v in value.items()))
132        else:
133            lines.append(f"{key.replace('_', ' ')}: {value}")
134    return "\n".join(lines) or "no files changed yet"
