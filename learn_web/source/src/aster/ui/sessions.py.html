31    def compose(self):
32        with Vertical(id="session-picker"):
33            yield Static(self.title, markup=False)
34            yield Input(value=self.initial_query, placeholder="关键词或编号筛选（最多显示 300 项）；↓ 选择，Enter 打开", id="session-query")
35            yield OptionList(id="session-choices", markup=False)

38    def on_mount(self):
39        self.filter(self.initial_query)

42    def filter(self, query):
43        choices = self.query_one(OptionList)
44        choices.clear_options()
45        rows = [row for row in self.rows if query.casefold() in row[1].casefold() or query in row[0]]
46        choices.add_options(Option(text, id=identifier) for identifier, text in rows[:300])
47        choices.highlighted = 0 if rows else None

49    def on_input_changed(self, event: Input.Changed):
50        self.filter(event.value)

52    def on_key(self, event):
53        if event.key == "down" and isinstance(self.focused, Input):
54            self.query_one(OptionList).focus()
55            event.stop()

57    def selected(self, summary=False):
58        choices = self.query_one(OptionList)

60            self.dismiss((choices.get_option_at_index(choices.highlighted).id, summary))

86    def _init_views(self):
87        self._views = []
88        self._register_view(self.runtime)
89
90    def _register_view(self, runtime):
91        view = {name: getattr(self, name) for name in VIEW_FIELDS}
92        view.update(runtime=runtime, log=[], events=[])
93        def listener(row):
94            self._route_session_event(runtime, row)
95        view["listener"] = listener
96        runtime.evidence.listeners.append(listener)
97        runtime.approval_callback = lambda call, decision: self._session_approve(runtime, call, decision)
98        runtime.question_callback = lambda question: self._session_question(runtime, question)
99        self._views.append(view)
100        return view
101
102    def _view(self, runtime=None):
103        return next(view for view in self._views if view["runtime"] is (runtime or self.runtime))
104
105    def _save_view(self):
106        self._view().update({name: getattr(self, name) for name in VIEW_FIELDS})
107        if self.screen_stack and self.screen_stack[0].query("#prompt"):
108            self._view()["draft"] = self._main_query("#prompt", Input).value
109
110    def _route_session_event(self, runtime, row):
111        if runtime is self.runtime:
112            self._event(row)
113        else:
114            view = self._view(runtime)
115            if row["event"] == "model_stream":
116                kind = row["type"]
117                if kind == "begin":
118                    view["_stream_text"] = view["_reasoning_text"] = ""
119                    view["_stream_active"] = True
120                elif kind in {"text_delta", "reasoning_delta"}:
121                    field = "_stream_text" if kind == "text_delta" else "_reasoning_text"
122                    view[field] = (view[field] + row.get("text", ""))[-65536:]
123                elif kind == "completed":
124                    view["_stream_active"] = False
125                    if view["_reasoning_text"] and self._show_thinking:
126                        view["events"].append({"event": "ui_note", "text": "思考摘要\n" + view["_reasoning_text"]})
127                return
128            view["events"].append(row)
129            view["events"] = view["events"][-2000:]

131    async def _session_approve(self, runtime, call, decision):
132        arguments = runtime.evidence.redactor.value(call.arguments)
133        text = f"会话 {runtime.state.session_id[:8]} · {runtime.workspace.root}\n是否仅允许本次 {call.name} 操作？\n{decision.reason}\n{json.dumps(arguments, ensure_ascii=False, indent=2)}"
134        return await self._prompt(HumanPrompt(text, approval=True)) == "yes"
135
136    async def _session_question(self, runtime, question):
137        return await self._prompt(HumanPrompt(f"会话 {runtime.state.session_id[:8]} · {runtime.workspace.root}\n{question}"))

139    def _switch_view(self, view):
140        self._save_view()
141        for name in VIEW_FIELDS:
142            setattr(self, name, view[name])
143        log = self._main_query("#log", RichLog)
144        log.clear()
145        from rich.markdown import Markdown
146        from rich.text import Text
147        for text, style in view["log"]:
148            log.write(Markdown(text) if style == "markdown" else Text(text, style=style))
149        events, view["events"] = view["events"], []
150        for event in events:
151            self._event(event)
152        self._main_query("#prompt", Input).value = view.get("draft", "")
153        self._main_query("#model-output", Static).update(self._stream_text)
154        self._main_query("#model-output").display = self._stream_active and self.busy
155        self._main_query("#reasoning", Static).update("思考摘要\n" + self._reasoning_text)
156        self._main_query("#reasoning").display = self._show_thinking and self._stream_active and bool(self._reasoning_text)
157        self._refresh_model()
158        self._tick()

160    def _open_session(self, root=None):
161        self._save_view()
162        target = Path(root).expanduser().resolve() if root else self.root_path
163        runtime = Runtime(target, self.settings.model_copy(deep=True))
164        if not isinstance(self.runtime.model, ModelClient):
165            runtime.model = self.runtime.model
166        self.runtime, self.root_path, self.settings = runtime, target, runtime.settings
167        self.task_worker = None
168        self._activity, self._mode, self._context_report = "就绪", "work", None
169        self._stream_text = self._reasoning_text = self._stream_status = self._last_model_text = ""
170        self._stream_active = False
171        self._register_view(runtime)
172        self._main_query("#prompt", Input).value = ""
173        self.action_clear_screen()
174        self._refresh_model()
175        self._write(f"已打开独立会话 · {target}\n输入任务开始；/sessions 切换会话。")

177    def _select_entry(self, result):
178        if result is None:
179            return
180        entry, summary = result
181        try:
182            self.runtime.checkout(entry, summary=summary)
183            self._after_checkout()
184        except (OSError, ValueError, RuntimeError) as exc:
185            self._write(f"未切换：{exc}\n如需保留当前文件并采用历史上下文，使用 /branch {entry} --adopt-workspace。", style="#efb38f")

187    def _after_checkout(self):
188        self.settings = self.runtime.settings
189        self._mode = self.runtime.state.mode
190        self._context_report = None
191        self._activity = "已选择历史位置"
192        self.action_clear_screen()
193        self._write("最初请求：" + self.runtime.state.task.prompt)
194        for message in self.runtime._structured_history[-20:]:
195            if message.get("role") in {"user", "assistant"}:
196                content = message.get("content", "")
197                if isinstance(content, list):
198                    content = "\n".join(block.get("text", "") for block in content if isinstance(block, dict))
199                self._write(f"{'你' if message['role'] == 'user' else 'Aster'}：{content}")
200        self._write(f"会话 {self.runtime.state.session_id[:12]} · 位置 {self.runtime.session_tree.cursor[:12]}\n历史和模型已恢复，文件保持当前状态。输入新请求会从此处继续。")
201        self._refresh_model()
202        self._tick()

204    def _select_session(self, result):
205        if result is None:
206            return
207        reference = result[0]
208        try:
209            if reference.startswith("live:"):
210                self._switch_view(self._views[int(reference[5:])])
211            else:
212                self._checkout_session(reference)
213        except (ValueError, RuntimeError, OSError) as exc:
214            self._write(f"未切换：{exc}", style="#efb38f")
215
216    def _checkout_session(self, reference, *, adopt_workspace=False):
217        for view in self._views:
218            state = view["runtime"].state
219            if state and state.session_id == reference:
220                self._switch_view(view)
221                return
222        head = self.runtime.session_tree.session_head(reference)
223        data = self.runtime.session_tree.restore(head)
224        for view in self._views:
225            if view["runtime"].state and view["runtime"].state.session_id == data["state"]["session_id"]:
226                self._switch_view(view)
227                return
228        # Validate before installing another view, keeping current work intact on failure.
229        restored = Runtime(self.root_path, self.settings.model_copy(deep=True))
230        restored.checkout(head, adopt_workspace=adopt_workspace)
231        self._save_view()
232        self.runtime, self.settings = restored, restored.settings
233        self.task_worker = None
234        self._stream_active = False
235        self._register_view(restored)
236        self._after_checkout()

238    def _session_command(self, command, argument):
239        if command == "/new":
240            args = shlex.split(argument)
241            if len(args) > 1:
242                raise ValueError('用法：/new ["工作目录"]')
243            self._open_session(args[0] if args else None)
244        elif command == "/sessions":
245            self._save_view()
246            rows, opened = [], set()
247            titles = {row["id"]: row["title"] for row in self.runtime.session_tree.sessions()}
248            for index, view in enumerate(self._views):
249                runtime, state = view["runtime"], view["runtime"].state
250                title = (titles.get(state.session_id, view.get("title", state.task.prompt))[:80] if state else "新会话")
251                busy = view["task_worker"] is not None and not view["task_worker"].is_finished
252                rows.append((f"live:{index}", f"{index + 1} · {'运行中' if busy else '空闲'} · {title} · {runtime.workspace.root}"))
253                if state:
254                    opened.add(state.session_id)
255            rows.extend((row["id"], f"{row['id'][:12]} · {row['title']}")
256                        for row in self.runtime.session_tree.sessions() if row["id"] not in opened)
257            self.push_screen(SessionPicker(rows, title="会话 · 当前项目的历史与已打开的任务", query=argument), self._select_session)
258        elif command == "/switch":
259            args = shlex.split(argument)
260            adopt = "--adopt-workspace" in args
261            args = [arg for arg in args if arg != "--adopt-workspace"]
262            if len(args) != 1:
263                raise ValueError("用法：/switch 会话编号或序号 [--adopt-workspace]")
264            if args[0].isdigit() and 1 <= int(args[0]) <= len(self._views):
265                self._switch_view(self._views[int(args[0]) - 1])
266            else:
267                self._checkout_session(args[0], adopt_workspace=adopt)
268        elif command == "/tree":
269            if not self.runtime.state:
270                raise ValueError("尚无会话，请先输入任务或恢复历史会话")
271            rows, depths, boundaries = [], {}, {}
272            labels = {"model_change": "模型配置", "compaction": "上下文压缩", "context_transform": "模型历史转换", "branch_summary": "分支摘录", "branch": "分支选择", "turn_start": "用户请求"}
273            changes = []
274            for entry in self.runtime.session_tree.entries(self.runtime.state.session_id):
275                depth = depths.get(entry["parent_id"], 0)
276                depths[entry["id"]] = depth + int(entry["kind"] == "turn_start")
277                boundary = boundaries.get(entry["parent_id"])
278                boundaries[entry["id"]] = entry["id"] if entry["restorable"] else boundary
279                if entry["kind"] in labels:
280                    changes.append(f"{labels[entry['kind']]} {entry['id'][:8]}")
281                if entry["restorable"]:
282                    payload = entry["payload"]
283                    marker = "●" if entry["id"] == self.runtime.session_tree.cursor else "◇"
284                    phase = {"running": "运行中", "completed": "完成", "stopped": "停止", "failed": "失败"}.get(payload.get("phase"), "就绪")
285                    parent = f" ← {boundary[:8]}" if boundary else " · 起点"
286                    rows.append((entry["id"], f"{'  ' * min(depth, 8)}{marker} {entry['id'][:12]}{parent} · {phase} · {payload.get('prompt', '')[:160]}"
287                                 + ("\n  " + " / ".join(changes) if changes else "")))
288                    changes.clear()
289            self.push_screen(SessionPicker(rows, title="会话树 · ◇ 可恢复位置 · 输入新请求创建后续分支", query=argument, summaries=True), self._select_entry)
290        elif command == "/branch":
291            args = shlex.split(argument)
292            flags = {arg for arg in args if arg.startswith("--")}
293            ids = [arg for arg in args if not arg.startswith("--")]
294            if len(ids) != 1 or flags - {"--summary", "--adopt-workspace"}:
295                raise ValueError("用法：/branch 位置编号 [--summary] [--adopt-workspace]")
296            self.runtime.checkout(ids[0], summary="--summary" in flags, adopt_workspace="--adopt-workspace" in flags)
297            self._after_checkout()
298        elif command == "/rename":
299            if not self.runtime.state:
300                raise ValueError("尚无会话")
301            self.runtime.session_tree.rename(self.runtime.state.session_id, argument)
302            self._view()["title"] = argument.strip()
303            self._write("会话名称已保存。")
304        elif command == "/entry":
305            if not self.runtime.state:
306                raise ValueError("尚无会话")
307            entries = self.runtime.session_tree.entries(self.runtime.state.session_id, argument)
308            if not argument or len(entries) != 1:
309                raise ValueError("请输入唯一的记录编号")
310            self._write(json.dumps(entries[0], ensure_ascii=False, indent=2))
311        else:
312            return False
313        return True
