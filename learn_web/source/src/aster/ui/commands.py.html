41VERIFICATION = {"missing": "未验证", "passed": "通过", "failed": "失败"}
42def saved_runs(evidence) -> list[tuple[str, dict]]:
43    runs = []
44    for path in evidence.path("runs").glob("*/checkpoint.json"):
45        try:
46            safe = evidence.path("runs", path.parent.name, "checkpoint.json")
47            data = json.loads(safe.read_text())
48            if isinstance(data, dict) and isinstance(data.get("state"), dict):
49                runs.append((safe.stat().st_mtime_ns, path.parent.name, data["state"]))
50        except (OSError, ValueError):
51            continue
52    return [
53        (run_id, state)
54        for _, run_id, state in sorted(runs, key=lambda row: row[0], reverse=True)[:20]
55    ]

57def inspect_command(command: str, runtime, *, settings, activity, root_path, context_report, budget_text, write) -> None:
58    state = runtime.state
59    if command == "/history":
60        runs = saved_runs(runtime.evidence)
61        write("最近的运行（最多 20 条）：" if runs else "暂无历史运行。")
62        for run_id, saved in runs:
63            write(
64                f"  /resume {run_id}\n    {PHASES.get(saved.get('phase'), '未知')} · {saved.get('task', {}).get('prompt', '')[:100]}"
65            )
66    elif command == "/status":
67        write(
68            f"模型：{settings.provider}/{settings.model}\n协议：{settings.protocol} · 流式：{'开启' if settings.stream else '关闭'}\n状态：{PHASES[state.phase] if state else '就绪'} · {activity}\n目录：{root_path}"
69        )
70        if state:
71            write(
72                f"轮次：{state.turn}/{state.task.max_turns} · 工具：{state.step}/{state.task.max_steps}\n修改文件：{len(state.changed_paths)} · 验证：{VERIFICATION[state.verification.state]}\n停止原因：{state.stop_reason or '无'}"
73            )
74    elif command == "/session":
75        write(
76            f"会话：{state.session_id}\n运行：{state.run_id}\n保存位置：{runtime.evidence.path('runs', state.run_id)}"
77            if state
78            else "尚未开始会话。"
79        )
80    elif command == "/context":
81        write(budget_text)
82        report = context_report
83        write(
84            f"最近请求估算：{report['estimated_tokens']:,} token · {'已压缩' if report['compacted'] else '未压缩'}\n分区：{json.dumps(report['sections'], ensure_ascii=False)}"
85            if report
86            else "本次界面尚无请求记录；发送任务后显示实际估算。"
87        )
88    elif command == "/usage":
89        usage = state.usage if state else {}
90        write("当前运行已报告的 token 用量：")
91        for key, label in (
92            ("input_tokens", "输入"),
93            ("output_tokens", "输出"),
94            ("total_tokens", "合计"),
95            ("cache_read_input_tokens", "缓存命中"),
96            ("reasoning_tokens", "推理"),
97        ):
98            write(
99                f"  {label}：{usage[key]:,}" if key in usage else f"  {label}：接口尚未报告"
100            )
101    elif command == "/tools":
102        tools = runtime.tool_definitions()
103        write(
104            f"可用工具（{len(tools)}）：\n"
105            + "\n".join(
106                f"  {item['name']} · {item.get('description', '')}" for item in tools
107            )
108        )
109    elif command == "/skills":
110        skills = runtime.skills.catalog()
111        write(
112            "项目技能：\n"
113            + "\n".join(
114                f"  {item['name']} · {item.get('error') or item['description']}"
115                for item in skills
116            )
117            if skills
118            else "暂无项目技能。可放置在 .aster/skills 或 .skills 目录。"
119        )
120    elif command == "/memory":
121        lessons = runtime.memory.state.get("lessons", [])
122        write(
123            f"长期记忆：{'开启' if settings.memory_enabled else '关闭'} · {len(lessons)} 条\n位置：{runtime.memory.path}"
124        )
125        for item in lessons[-20:]:
126            write(f"  {item['id']} · {item.get('text', '')[:300]}")
127    elif command == "/working-memory":
128        write(
129            runtime.memory.working_section(state.working_memory)
130            if state and state.working_memory
131            else "当前会话暂无工作记忆。"
132        )
133    elif command == "/agents":
134        workers = runtime.workers.workers.values()
135        write(
136            f"子任务：{len(runtime.workers.workers)} · 并行上限：{settings.max_workers}"
137        )
138        for worker in workers:
139            write(
140                f"  {worker.worker_id} · {'已结束' if worker.future.done() else '运行中'} · {worker.task[:120]}"
141            )
142    elif command == "/todos":
143        write(state.plan if state and state.plan else "暂无计划。")
144        for todo in state.todos if state else []:
145            write(f"  {todo.get('status', '')} · {todo.get('content', '')}")

147def execute_command(command: str, argument: str, runtime, *, write, show_help,
148                    clear_screen, model_command, new_session, start_task,
149                    reset_context, configure_output, configure_mode, inspect) -> None:
150    if command not in COMMANDS:
151        write(f"未知命令：{command}。输入 /help 查看帮助。", style="#efb38f")
152        return
153    if argument and not COMMANDS[command][0]:
154        raise ValueError(f"{command} 不接受参数；输入 /help 查看用法。")
155    if command == "/help":
156        show_help()
157    elif command == "/clear":
158        clear_screen()
159    elif command == "/model":
160        model_command(argument)
161    elif command == "/new":
162        new_session()
163    elif command in {"/continue", "/resume"}:
164        parts = argument.split()
165        flags = {part for part in parts if part.startswith("--")}
166        args = [part for part in parts if not part.startswith("--")]
167        if (
168            flags - {"--adopt-workspace", "--allow-model-change"}
169            or (command == "/resume" and len(args) != 1)
170            or (command == "/continue" and (args or "--allow-model-change" in flags))
171        ):
172            raise ValueError(f"用法：{command} {COMMANDS[command][0]}")
173        if command == "/resume":
174            run_id = args[0]
175            if run_id == "latest":
176                runs = saved_runs(runtime.evidence)
177                if not runs:
178                    raise ValueError("没有可恢复的历史运行。")
179                run_id = runs[0][0]
180            runtime.resume(
181                run_id,
182                allow_model_change="--allow-model-change" in flags,
183                adopt_workspace="--adopt-workspace" in flags,
184            )
185            reset_context()
186        start_task(None, adopt_workspace="--adopt-workspace" in flags)
187    elif command in {"/stream", "/thinking"}:
188        configure_output(command, argument)
189    elif command in {"/plan", "/plan-exit"}:
190        configure_mode(command, argument)
191    elif command in {"/review", "/test"}:
192        prompt = (
193            "审查代码，按严重程度报告可确认的问题，提供文件位置与依据。审查范围："
194            if command == "/review"
195            else "运行相关测试并分析失败原因；使用 run_tests 记录正式验证。测试目标："
196        )
197        start_task(prompt + (argument or "当前项目"))
198    else:
199        inspect(command)

202def parse_command(prompt):
203    command, _, argument = prompt.partition(" ")
204    argument = argument.strip()
205    command = {
206        "quit": "/quit",
207        "exit": "/quit",
208        "/exit": "/quit",
209        "/q": "/quit",
210        "/stop": "/cancel",
211    }.get(command.lower(), command.lower())
212    return command, argument
