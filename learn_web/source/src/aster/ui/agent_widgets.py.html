33    def compose(self):
34        with Vertical(id="agent-panel"):

46    def on_mount(self):

51    def selected_id(self):
52        choices = self.query_one(OptionList)
53        return choices.get_option_at_index(choices.highlighted).id if choices.highlighted is not None else None

55    @staticmethod
56    def status_label(row):
57        if row.get("stop_reason") in {"worker_cancelled", "cancelled"}:
58            return "已中断"
59        return {"running": "运行中", "ready": "运行中", "completed": "已完成",
60                "failed": "失败", "stopped": "已中断", "cancelled": "已中断",
61                "interrupted": "已中断", "unavailable": "不可用"}.get(row["status"], row["status"])

63    def refresh_agents(self):
64        rows = self.runtime.workers.overview()
65        selected = self.selected_id()
66        if rows != self.rows:
67            self.rows = rows
68            choices = self.query_one(OptionList)
69            choices.clear_options().add_options(Option(self.runtime.evidence.redactor.text(
70                f"{self.status_label(row)} · {row['worker_id']} · {' '.join(row['task'].split())[:160]}"
71            ), id=row["worker_id"]) for row in rows)
72            choices.highlighted = next((i for i, row in enumerate(rows) if row["worker_id"] == selected), 0) if rows else None
73        self.query_one("#agent-send", Button).disabled = not rows or self._acting
74        row = next((row for row in rows if row["worker_id"] == self.selected_id()), {})
75        self.query_one("#agent-stop", Button).disabled = row.get("status") not in {"running", "ready"} or self._acting
76        self.show_detail()

78    def show_detail(self):
79        row = next((row for row in self.rows if row["worker_id"] == self.selected_id()), None)
80        text = "暂无子任务。输入明确的调查或审查任务后点击启动。"
81        if row:
82            text = f"{self.status_label(row)} · {row['task']}\n\n{row.get('answer') or '尚无最终结果。'}"
83            if row.get("stop_reason"):
84                text += "\n停止原因：" + row["stop_reason"]
85            if row.get("stale_paths"):
86                text += "\n文件已变化，采用结论前需重新核对：" + ", ".join(row["stale_paths"])
87            if row.get("read_evidence"):
88                text += "\n已读取文件：\n" + "\n".join(row["read_evidence"])
89            key = (row.get("run_id"), row.get("steps"), row["status"])
90            if key != self._events_key:
91                self._events_key = key
92                try:
93                    events = self.runtime.evidence.events(row["run_id"]) if row.get("run_id") else []
94                    self._events = [f"{event['call']['name']} · {event['outcome']['status']}\n"
95                                    + str(event['outcome'].get('content', ''))[:1200]
96                                    for event in events if event["event"] == "tool_result"][-8:]
97                except (OSError, ValueError):
98                    self._events = []
99            if self._events:
100                text += "\n最近工具事件：\n" + "\n".join(self._events)
101        text = self.runtime.evidence.redactor.text(text)
102        if text != self._detail:
103            self._detail = text
104            self.query_one(RichLog).clear().write(text)

114    async def operate(self, action):
115        if self._acting:
116            return
117        self._acting = True
118        self.query_one("#agent-start", Button).disabled = True
119        self.refresh_agents()
120        try:
121            prompt = self.query_one(Input)
122            message = prompt.value.strip()
123            worker_id = self.selected_id()
124            if action != "stop" and not message:
125                self.notice("请先输入任务或补充说明。")
126                return
127            if action == "start":
128                if self.runtime.state is None:
129                    self.runtime.start(TaskSpec(prompt="用户指定的并行调查：" + message,
130                                               **self.app.task_limits, require_edit=False, require_verification=False))
131                result = await self.runtime.workers.start(message, mode="explore", explicit=True)
132            elif worker_id:
133                result = (await self.runtime.workers.stop(worker_id) if action == "stop" else
134                          await self.runtime.workers.message(worker_id, message, resume=True))
135            else:
136                self.notice("请先选择子任务。")
137                return
138            if result.get("status") in {"blocked", "missing"}:
139                reason = result.get("reason", result["status"])
140                self.notice("未执行：" + {
141                    "worker_limit": "已达到子任务并行上限，请等待或取消一个任务。",
142                    "team_budget_reserved_for_parent": "剩余预算不足，需为主任务整合保留额度。",
143                    "worker_continuation_unavailable": "剩余预算不足，无法继续子任务。",
144                    "worker_permissions_changed": "主会话权限已变化，无法继续原子任务。",
145                    "worker_restore_blocked": "无法恢复，请核对历史记录与当前配置。",
146                }.get(reason, reason))
147            else:
148                if action != "stop":
149                    prompt.value = ""
150                self.notice("已取消子任务。" if action == "stop" else "已发送；在列表中查看进展。")
151        except (OSError, ValueError, RuntimeError) as exc:
152            self.notice(f"未执行：{exc}")
153        finally:
154            self._acting = False
155            if self.is_mounted:
156                self.query_one("#agent-start", Button).disabled = False
157                self.refresh_agents()

159    def on_button_pressed(self, event: Button.Pressed):
160        event.stop()
161        action = event.button.id.removeprefix("agent-")
162        if action == "close":
163            self.action_close()
164        else:
165            self.run_worker(self.operate(action), exit_on_error=False)

167    def on_input_submitted(self, event: Input.Submitted):
