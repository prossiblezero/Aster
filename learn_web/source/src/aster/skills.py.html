1"""Directory-only lazy skills; skill text is untrusted context."""
2
3from __future__ import annotations
4
5import re
6from pathlib import Path
7
8SKILL_NAME = re.compile(r"[A-Za-z0-9_.-]{1,80}")
9MAX_SKILL_BYTES = 20 * 1024
10
11
12class Skills:
13    def __init__(self, workspace: Path):
14        self.workspace = workspace.resolve()
15        self.dirs = [self.workspace / ".aster" / "skills", self.workspace / ".skills"]
16
17    def _safe_path(self, path: Path) -> bool:
18        """Skill metadata may be read, but no component may redirect outside its root."""
19        if not path.is_relative_to(self.workspace):
20            return False
21        cursor = self.workspace
22        for part in path.relative_to(self.workspace).parts:
23            cursor /= part
24            if cursor.is_symlink():
25                return False
26        return path.resolve().is_relative_to(self.workspace)
27
28    def list(self) -> list[str]:
29        return sorted(
30            {
31                p.stem
32                for root in self.dirs
33                if self._safe_path(root) and root.is_dir()
34                for p in root.glob("*.md")
35                if SKILL_NAME.fullmatch(p.stem) and self._safe_path(p) and p.is_file()
36            }
37        )
38
39    def load(self, name: str) -> str:
40        if not SKILL_NAME.fullmatch(name):
41            raise ValueError("invalid skill name")
42        for root in self.dirs:
43            path = root / f"{name}.md"
44            if self._safe_path(path) and path.is_file():
45                if path.stat().st_size > MAX_SKILL_BYTES:
46                    raise ValueError("skill exceeds 20 KiB")
47                with path.open("rb") as stream:
48                    data = stream.read(MAX_SKILL_BYTES + 1)
49                if len(data) > MAX_SKILL_BYTES:
50                    raise ValueError("skill exceeds 20 KiB")
51                return data.decode("utf-8")
52        raise FileNotFoundError(name)
