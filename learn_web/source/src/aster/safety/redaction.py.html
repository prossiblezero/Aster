7class Redactor:
8    def __init__(self, secrets: list[str] = ()):
9        self.secrets = sorted({x for x in secrets if len(x) >= 6}, key=len, reverse=True)
10
11    def text(self, text: str) -> str:
12        for secret in self.secrets:
13            text = text.replace(secret, "[REDACTED]")
14        return re.sub(r"(?i)\b(?:sk-[a-z0-9_-]{12,}|bearer\s+[a-z0-9._-]{12,})", "[REDACTED]", text)
15
16    def value(self, value: Any, *, mask_fields: bool = True) -> Any:
17        if isinstance(value, str):
18            return self.text(value)
19        if isinstance(value, dict):
20            return {
21                k: "[REDACTED]"
22                if mask_fields and re.fullmatch(r"(?i)(api_key|password|secret|authorization)", k)
23                else self.value(v, mask_fields=mask_fields)
24                for k, v in value.items()
25            }
26        if isinstance(value, list):
27            return [self.value(v, mask_fields=mask_fields) for v in value]
28        return value
