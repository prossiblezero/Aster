1"""Tiny isolated child launcher: Landlock filesystem rules and seccomp syscalls.
2
3Executed as a separate interpreter, never via preexec_fn in a threaded parent.
4No fallback here: installation failure exits 125 before the command executes.
5"""
6

8import errno
9import os
10import platform
11import resource

14from pathlib import Path

17def checked(value: int, operation: str) -> int:
18    if value < 0:
19        raise OSError(ctypes.get_errno(), operation)
20    return value

23def restrict(root: Path, network: bool, readonly_dependencies: tuple[Path, ...] = (),
24             readable: tuple[Path, ...] = ()) -> None:
25    if sys.platform != "linux" or platform.machine() not in {"x86_64", "aarch64"}:
26        raise RuntimeError("Linux x86_64/aarch64 required")
27    libc = ctypes.CDLL(None, use_errno=True)
28    abi = checked(libc.syscall(444, 0, 0, 1), "landlock ABI")
29    if abi < 3:
30        raise RuntimeError("Landlock ABI >= 3 required (truncate mediation)")
31    handled = (1 << 15) - 1
32    ruleset = ctypes.c_uint64(handled)
33    fd = checked(
34        libc.syscall(444, ctypes.byref(ruleset), ctypes.sizeof(ruleset), 0), "landlock create"
35    )
36
37    class PathRule(ctypes.Structure):
38        _pack_ = 1
39        _fields_ = [("access", ctypes.c_uint64), ("parent_fd", ctypes.c_int)]
40
41    def allow(path: Path, access: int):
42        if not path.exists():
43            return
44        if not path.is_dir():
45            access &= 1 | 2 | 4 | (1 << 14)
46        parent = os.open(path, os.O_PATH | os.O_CLOEXEC)
47        try:
48            rule = PathRule(access, parent)
49            checked(libc.syscall(445, fd, 1, ctypes.byref(rule), 0), "landlock add rule")
50        finally:
51            os.close(parent)
52
53    try:
54        for raw in (
55            "/usr",
56            "/bin",
57            "/lib",
58            "/lib64",
59            sys.prefix,
60            "/etc/ld.so.cache",
61            "/etc/localtime",
62            "/etc/ssl",
63            "/dev/urandom",
64            "/dev/random",
65        ):
66            allow(Path(raw).resolve(), 1 | 4 | 8)
67        allow(Path("/etc/debian_version").resolve(), 4)  # pip's distro metadata, read only.
68        mime_types = Path("/etc/mime.types").resolve()
69        if mime_types.is_file():
70            allow(mime_types, 4)  # Python mimetypes/openpyxl need this exact read-only file.
71        if network:
72            # DNS needs these exact files; resolv.conf may point into /run/systemd.
73            # Resolving the file grants neither its parent directory nor /etc writes.
74            for raw in ("/etc/resolv.conf", "/etc/hosts", "/etc/nsswitch.conf", "/etc/gai.conf"):
75                allow(Path(raw).resolve(), 4)
76        allow(Path("/dev/null"), 2 | 4)
77        for path in (*readonly_dependencies, *readable):
78            allow(path.resolve(strict=True), 1 | 4 | 8)
79        allow(root, handled)
80        checked(libc.prctl(38, 1, 0, 0, 0), "no_new_privs")
81        checked(libc.syscall(446, fd, 0), "landlock restrict")
82    finally:
83        os.close(fd)
84
85    sec = ctypes.CDLL("libseccomp.so.2", use_errno=True)
86    sec.seccomp_init.argtypes = [ctypes.c_uint32]
87    sec.seccomp_init.restype = ctypes.c_void_p
88    sec.seccomp_rule_add.argtypes = [ctypes.c_void_p, ctypes.c_uint32, ctypes.c_int, ctypes.c_uint]
89    sec.seccomp_load.argtypes = [ctypes.c_void_p]
90    sec.seccomp_release.argtypes = [ctypes.c_void_p]
91    sec.seccomp_syscall_resolve_name.argtypes = [ctypes.c_char_p]
92
93    class ArgRule(ctypes.Structure):
94        _fields_ = [
95            ("arg", ctypes.c_uint),
96            ("op", ctypes.c_int),
97            ("value", ctypes.c_uint64),
98            ("unused", ctypes.c_uint64),
99        ]
100
101    sec.seccomp_rule_add_array.argtypes = [
102        ctypes.c_void_p,
103        ctypes.c_uint32,
104        ctypes.c_int,
105        ctypes.c_uint,
106        ctypes.POINTER(ArgRule),
107    ]
108    ctx = sec.seccomp_init(0x7FFF0000)  # SCMP_ACT_ALLOW
109    if not ctx:
110        raise RuntimeError("seccomp initialization failed")
111    denied = [
112        "ptrace",
113        "process_vm_readv",
114        "process_vm_writev",
115        "mount",
116        "umount2",
117        "pivot_root",
118        "unshare",
119        "setns",
120        "bpf",
121        "perf_event_open",
122        "io_uring_setup",
123        "open_by_handle_at",
124        "setsid",
125        "setpgid",
126        "kill",
127        "tkill",
128        "tgkill",
129    ]
130    if readonly_dependencies:
131        # Landlock mediates content writes, not chmod/chown/timestamps/xattrs.
132        # Borrowed dependencies share inodes, so scoped commands must not mutate
133        # their metadata either. Source publication is performed by the parent.
134        denied += [
135            "chmod", "fchmod", "fchmodat", "fchmodat2",
136            "chown", "fchown", "lchown", "fchownat",
137            "utime", "utimes", "futimesat", "utimensat",
138            "setxattr", "lsetxattr", "fsetxattr",
139            "removexattr", "lremovexattr", "fremovexattr",
140        ]
141    if not network:
142        denied += ["socket", "connect", "bind", "sendmsg", "sendmmsg"]
143    try:
144        for name in denied:
145            number = sec.seccomp_syscall_resolve_name(name.encode())
146            if number >= 0:
147                checked(
148                    sec.seccomp_rule_add(ctx, 0x00050000 | errno.EPERM, number, 0), "seccomp rule"
149                )
150        if not network:
151            # asyncio wakes its loop through a local pair and send(...), implemented
152            # by sendto with a NULL destination. The launcher inherits no socket FDs;
153            # socket/connect/bind and sending to an explicit address remain forbidden.
154            for name, argument, value in (("socketpair", 0, socket.AF_UNIX), ("sendto", 4, 0)):
155                number = checked(sec.seccomp_syscall_resolve_name(name.encode()), f"resolve {name}")
156                rule = ArgRule(argument, 1, value, 0)  # SCMP_CMP_NE
157                checked(
158                    sec.seccomp_rule_add_array(
159                        ctx, 0x00050000 | errno.EPERM, number, 1, ctypes.byref(rule)
160                    ),
161                    f"seccomp {name} rule",
162                )
163        checked(sec.seccomp_load(ctx), "seccomp load")
164    finally:
165        sec.seccomp_release(ctx)
166

168def main():
169    root, network, command, *dependencies = sys.argv[1:]
170    # Readable roots follow "--readable": read-only like dependencies, but the parent guards
171    # their metadata itself, so chmod stays available inside the copy.
172    readable = dependencies[dependencies.index("--readable") + 1:] if "--readable" in dependencies else []
173    dependencies = dependencies[:dependencies.index("--readable")] if "--readable" in dependencies else dependencies
174    try:
175        os.chdir(root)
176        resource.setrlimit(resource.RLIMIT_FSIZE, (16 * 1024 * 1024, 16 * 1024 * 1024))
177        resource.setrlimit(resource.RLIMIT_NOFILE, (256, 256))
178        resource.setrlimit(resource.RLIMIT_CPU, (120, 120))
179        restrict(Path(root), network == "on", tuple(Path(path) for path in dependencies),
180                 tuple(Path(path) for path in readable))
181    except Exception as exc:
182        print(f"ASTER_SANDBOX_UNAVAILABLE:{type(exc).__name__}:{exc}", file=sys.stderr)
183        sys.exit(125)
184    os.execvpe("/bin/sh", ["sh", "-c", command], os.environ)

187if __name__ == "__main__":
188    main()
