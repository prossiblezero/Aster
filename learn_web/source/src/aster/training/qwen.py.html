1"""Parse raw Qwen completions into Aster actions, without trusting the server's tool parser.
2
3Rollouts must train on the exact tokens a policy produced, so the harness reads the decoded
4completion itself instead of an OpenAI-compatible server's re-serialized tool_calls. Two
5formats appear in Qwen chat templates:
6
7- hermes JSON (Qwen2.5, Qwen3): <tool_call>{"name": ..., "arguments": {...}}</tool_call>
8- XML parameters (Qwen3-Coder, Qwen3.5): <tool_call><function=NAME><parameter=KEY>
9  VALUE</parameter></function></tool_call>
10
11A malformed call is reported, never guessed: the episode continues with that error as an
12observation, which is what a policy needs to learn from.
13"""
14
15from __future__ import annotations
16
17import json
18import re
19from dataclasses import dataclass, field
20
21THINK = re.compile(r"<think>.*?</think>\s*", re.S)
22TOOL_CALL = re.compile(r"<tool_call>(.*?)(?:</tool_call>|$)", re.S)
23FUNCTION = re.compile(r"<function=([^>\s]+)>(.*?)(?:</function>|$)", re.S)
24PARAMETER = re.compile(r"<parameter=([^>\s]+)>\n?(.*?)\n?</parameter>", re.S)
25
26
27@dataclass
28class ParsedTurn:
29    text: str
30    calls: list[tuple[str, dict]] = field(default_factory=list)
31    errors: list[str] = field(default_factory=list)
32    reasoning: str = ""
33
34
35def _typed(value: str, schema: dict):
36    """XML parameters arrive as text; the tool schema says what they mean."""
37    kind = schema.get("type")
38    if kind == "string" or (kind is None and "enum" not in schema):
39        return value
40    if kind == "boolean":
41        if value.strip().lower() in {"true", "false"}:
42            return value.strip().lower() == "true"
43        raise ValueError(f"expected a boolean, got {value[:40]!r}")
44    if kind == "integer":
45        return int(value.strip())
46    if kind == "number":
47        return float(value.strip())
48    return json.loads(value)
49
50
51def parse_completion(text: str, tools: list[dict] | None = None) -> ParsedTurn:
52    """Split a decoded completion into visible text, reasoning, and tool calls."""
53    if "</think>" in text and "<think>" not in text:
54        # Thinking templates open <think> inside the prompt, so the completion starts inside it.
55        text = "<think>" + text
56    reasoning = "\n".join(m.strip() for m in re.findall(r"<think>(.*?)</think>", text, re.S))
57    body = THINK.sub("", text)
58    if "<think>" in body and "</think>" not in body:
59        return ParsedTurn(text="", reasoning=body, errors=["reasoning was cut off before any answer"])
60    schemas = {}
61    for tool in tools or []:
62        spec = tool.get("function", tool)
63        schemas[spec.get("name")] = spec.get("parameters", {}).get("properties", {})
64    turn = ParsedTurn(text=TOOL_CALL.sub("", body).strip(), reasoning=reasoning)
65    complete = re.findall(r"<tool_call>(.*?)</tool_call>", body, re.S)
66    if "<tool_call>" in (body.rsplit("</tool_call>", 1)[-1] if complete else body):
67        turn.errors.append("a tool call is not closed with </tool_call> (the output may have been cut off)")
68    for index, raw in enumerate(complete):
69        raw = raw.strip()
70        if raw.startswith("{"):
71            try:
72                data = json.loads(raw)
73                name, arguments = data["name"], data.get("arguments", {})
74                if isinstance(arguments, str):
75                    arguments = json.loads(arguments)
76                if not isinstance(arguments, dict):
77                    raise TypeError("arguments must be an object")
78            except (ValueError, KeyError, TypeError) as exc:
79                turn.errors.append(f"tool call {index + 1} is not valid JSON with name and arguments: {exc}")
80                continue
81            turn.calls.append((name, arguments))
82            continue
83        match = FUNCTION.search(raw)
84        if not match:
85            turn.errors.append(f"tool call {index + 1} has neither JSON nor <function=...>")
86            continue
87        name, inner = match.group(1), match.group(2)
88        if not raw.rstrip().endswith("</function>"):
89            turn.errors.append(f"tool call {index + 1} is not closed with </function>")
90            continue
91        arguments, properties = {}, schemas.get(name, {})
92        try:
93            for key, value in PARAMETER.findall(inner):
94                arguments[key] = _typed(value, properties.get(key, {}))
95        except ValueError as exc:
96            turn.errors.append(f"tool call {index + 1} ({name}) has a malformed parameter: {exc}")
97            continue
98        turn.calls.append((name, arguments))
99    return turn
