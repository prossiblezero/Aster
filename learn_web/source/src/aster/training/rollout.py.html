1"""Token-in, token-out rollouts of a local policy through AsterEnv.
2
3The trainer needs, for every assistant turn, the exact prompt token ids the policy saw, the
4token ids it generated, and their log-probabilities under the behavior policy. Decoding and
5re-encoding text would silently change those tokens, so this module:
6
7- renders each request with the policy's own chat template (``tokenizer``),
8- samples token ids from a generation backend (``generate``; see ``VLLMCompletions``),
9- parses the decoded completion itself (``qwen.parse_completion``),
10- records one ``Turn`` per decision and splits the episode into segments wherever the next
11  prompt is not an exact extension of the previous prompt plus completion (context
12  compaction, archived observations, or a chat template that rewrites earlier turns).
13
14Per-turn samples are always exact. Segments say which turns can be concatenated into one
15sequence. Episode labels keep infrastructure failures out of the gradient.
16"""
17

32def training_settings(**overrides) -> Settings:
33    """Product behaviour that would leak into the training distribution is off.
34
35    No memory or background maintenance, no workers, no approval prompts, no completion gate
36    (the reward comes from an independent verifier) and no workflow-state notes. Chat
37    Completions without streaming, which local OpenAI-compatible servers implement.
38    """
39    defaults = dict(
40        protocol="chat", stream=False, auth="none", approval="auto", completion="off",
41        memory_enabled=False, memory_evolution=False, auto_dream=False, max_workers=0,
42        workflow_notes=False, max_recovery=0, oneshot_maintenance_seconds=0,
43    )
44    return Settings(**{**defaults, **overrides})

94def chat_conversation(model_input: list[dict], tools: list[dict]) -> tuple[list[dict], list[dict]]:
95    """Aster's request as chat messages and tools, with arguments as objects for templates."""
96    payload = request_payload(model_input, tools, protocol="chat")
97    messages = json.loads(json.dumps(payload["messages"]))
98    for message in messages:
99        for call in message.get("tool_calls") or []:
100            arguments = call.get("function", {}).get("arguments")
101            if isinstance(arguments, str):
102                try:
103                    call["function"]["arguments"] = json.loads(arguments)
104                except ValueError:
105                    pass
106    return messages, payload["tools"]

109class VLLMCompletions:
110    """vLLM's OpenAI-compatible /v1/completions with a token-id prompt.
111
112    ``return_tokens_as_token_ids`` makes vLLM report each sampled token as ``token_id:N``,
113    so the ids are read back exactly instead of being re-encoded from text.
114    """
115
116    def __init__(self, base_url: str, model: str, *, max_tokens: int = 4096, temperature: float = 1.0,
117                 top_p: float = 1.0, seed: int | None = None, client=None, timeout: float = 600):
118        import httpx
119
120        self.url = base_url.rstrip("/") + "/completions"
121        self.body = {"model": model, "max_tokens": max_tokens, "temperature": temperature, "top_p": top_p,
122                     "logprobs": 0, "return_tokens_as_token_ids": True, "skip_special_tokens": False}
123        if seed is not None:
124            self.body["seed"] = seed
125        self.client = client or httpx.AsyncClient(timeout=timeout)
126
127    async def __call__(self, prompt_ids: list[int]) -> Generation:
128        response = await self.client.post(self.url, json={**self.body, "prompt": prompt_ids})
129        response.raise_for_status()
130        choice = response.json()["choices"][0]
131        tokens = choice["logprobs"]["tokens"]
132        ids = [int(token.removeprefix("token_id:")) for token in tokens]
133        logprobs = [float(value) for value in choice["logprobs"]["token_logprobs"]]
134        if len(ids) != len(logprobs):
135            raise ValueError("token ids and logprobs differ in length")
136        return Generation(ids, logprobs, choice.get("finish_reason") or "stop")

139async def rollout(env: AsterEnv, task: TaskSpec | str, tokenizer: Tokenizer, generate: Generate, *,
140                  task_id: str = "", template_kwargs: dict | None = None) -> Trajectory:
141    """Run one episode; never raises for policy mistakes, only labels them."""
142    observation, info = env.reset(task)
143    turns: list[Turn] = []
144    segment, previous = 0, None
145    reward, terminated, truncated = None, False, False
146    label = "budget_exhausted"
147    try:
148        while True:
149            messages, tools = chat_conversation(observation["model_input"], observation["tools"])
150            prompt_ids = list(tokenizer.apply_chat_template(
151                messages, tools=tools, add_generation_prompt=True, tokenize=True, **(template_kwargs or {})))
152            extends = previous is not None and prompt_ids[:len(previous)] == previous
153            if previous is not None and not extends:
154                segment += 1
155            generation = await generate(prompt_ids)
156            text = tokenizer.decode(generation.token_ids, skip_special_tokens=False)
157            parsed = parse_completion(text, tools)
158            if generation.finish_reason == "length":
159                parsed.errors.append("the output reached the token limit before finishing")
160            turn = Turn(prompt_ids, generation.token_ids, generation.logprobs, generation.finish_reason,
161                        segment, extends, [{"name": n, "arguments": a} for n, a in parsed.calls],
162                        list(parsed.errors), bool(observation.get("context_compacted")))
163            turns.append(turn)
164            previous = prompt_ids + generation.token_ids
165            if parsed.errors:
166                result = await env.step_invalid(text, "; ".join(parsed.errors))
167            elif parsed.calls:
168                result = await env.step_turn(parsed.text, [
169                    ToolCall(call_id=f"call_{len(turns)}_{i}", name=n, arguments=a)
170                    for i, (n, a) in enumerate(parsed.calls)])
171            else:
172                result = await env.step(Final(answer=parsed.text))
173            observation, reward, terminated, truncated, step_info = result
174            if terminated or truncated:
175                info.update(step_info)
176                break
177        verified = (info.get("reward_verification") or {}).get("passed")
178        if turns and turns[-1].parse_errors and not verified:
179            label = "parse_error"
180        elif verified:
181            label = "success"
182        elif terminated:
183            label = "candidate_failure"
184    except Exception as exc:  # generation server, sandbox or verifier crashed: not the policy's fault
185        info["error"] = f"{type(exc).__name__}: {exc}"
186        label, reward = "infrastructure_error", None
187    finally:
188        try:
189            env.close()
190        except RuntimeError:
191            pass
192    trainable = label in {"success", "candidate_failure", "parse_error"} and reward is not None
193    return Trajectory(task_id or str(info.get("run_id", "")), turns, reward, label, trainable,
194                      {key: info.get(key) for key in ("run_id", "stop_reason", "error", "reward_verification")})
