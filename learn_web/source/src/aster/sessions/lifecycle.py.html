18def archive_native_history(
19    evidence: EvidenceStore,
20    run_id: str,
21    history: list[dict],
22    converted: CanonicalHistory,
23    origins: list[dict],
24) -> dict:
25    artifact = ("runs", run_id, f"native-history-{len(origins) + 1}.json")
26    evidence.save(
27        artifact,
28        {"source": converted.source, "structured_history": history},
29        preserve_native_history=True,
30    )
31    return {
32        "artifact": "/".join(artifact),
33        "source": converted.source,
34        "omitted_reasoning": converted.omitted_reasoning,
35    }

38def preview_context(
39    evidence: EvidenceStore,
40    state: RunState,
41    history: list[dict],
42    settings: Settings,
43    sources: dict[str, str],
44    tools: list[dict],
45    *,
46    environment: str = "",
47    skills_available: bool = False,
48) -> ContextReport:
49    """Budget-check detached state before Runtime publishes a session transition."""
50    _, report = ContextBuilder(
51        settings.effective_context_tokens, settings.effective_output_tokens
52    ).build(
53        state,
54        state.task.prompt,
55        structured_history=history,
56        tools=tools,
57        protocol=settings.protocol,
58        model=settings.model,
59        memory_enabled=settings.memory_enabled,
60        skills_available=skills_available,
61        workers_available=settings.max_workers > 0,
62        environment=environment,
63        archive=lambda call_id, tool, content: evidence.archive_observation(
64            state.run_id, call_id, tool, content
65        ),
66        validate_archive=lambda ref: evidence.read_observation(
67            sources.get(ref["id"], state.run_id), ref["id"], limit=1
68        ),
69        redact_history=evidence.redact_history,
70    )
71    return report

74def prepare_continuation(
75    previous: RunState,
76    history: list[dict],
77    verifier: Verifier,
78    *,
79    feedback: str | None,
80    drift: list[str],
81    max_steps: int | None,
82    max_turns: int | None,
83) -> tuple[RunState, list[dict]]:
84    """Copy a conversation into a fresh bounded run without mutating its predecessor."""
85    state = previous.model_copy(deep=True)
86    state.run_id = uuid4().hex
87    state.phase, state.stop_reason, state.final, state.feedback = "running", "", "", ""
88    state.step = state.turn = state.final_revisions = 0
89    state.usage = {}
90    state.team_usage = {}
91    state.pending_handoffs = {}
92    for item in state.todos:
93        if item.get("delivery") == "running":
94            item.update(status="pending", owner="parent", delivery="interrupted")
95    history = deepcopy(history)
96    if feedback is not None:
97        inferred = verifier.infer(feedback)
98        unfinished = previous.phase != "completed"
99        if unfinished:
100            inferred = {
101                key: value or getattr(previous.task, key) for key, value in inferred.items()
102            }
103        if inferred["require_edit"] and previous.task.require_verification:
104            inferred["require_verification"] = True
105        state.task = TaskSpec(
106            prompt=previous.task.prompt,
107            max_steps=previous.task.max_steps,
108            max_turns=previous.task.max_turns,
109            max_tokens=previous.task.max_tokens,
110            required_artifacts=previous.task.required_artifacts
111            if inferred["require_edit"] or unfinished
112            else [],
113            **inferred,
114        )
115        if not unfinished:
116            state.observed, state.changed_paths = False, []
117        history.append({"role": "user", "content": feedback})
118        state.messages.append({"role": "user", "content": feedback, "important": True})
119    limits = {
120        key: value
121        for key, value in {"max_steps": max_steps, "max_turns": max_turns}.items()
122        if value is not None
123    }
124    if limits:
125        state.task = TaskSpec.model_validate({**state.task.model_dump(), **limits})
126    if feedback is not None or drift:
127        state.verification = Verification()
128        state.read_evidence, state.working_memory, state.recalled_ids = {}, {}, []
129    if drift:
130        note = (
131            "User adopted external workspace edits; read current files before editing: "
132            + ", ".join(drift)
133        )
134        history.append({"role": "user", "content": note})
135    return state, history

138def read_session_observation(
139    evidence: EvidenceStore, run_id: str, sources: dict[str, str], args: dict
140) -> Outcome:
141    """Read only this run and explicitly authorized ancestor observation IDs."""
142    try:
143        if args["observation_id"]:
144            page = evidence.read_observation(
145                sources.get(args["observation_id"], run_id),
146                args["observation_id"],
147                offset=args["offset"],
148                limit=args["limit"],
149                query=args["query"],
150            )
151        else:
152            if not sources:
153                page = evidence.list_observations(
154                    run_id,
155                    offset=args["offset"],
156                    limit=min(10, args["limit"]),
157                    query=args["query"],
158                )
159            else:
160                records = {}
161                for source_run in [run_id, *sorted(set(sources.values()))]:
162                    offset = 0
163                    while True:
164                        listing = evidence.list_observations(
165                            source_run, offset=offset, limit=20, query=args["query"]
166                        )
167                        for ref in listing["observations"]:
168                            if (
169                                source_run == run_id
170                                or sources.get(ref["id"]) == source_run
171                            ):
172                                records[ref["id"]] = ref
173                        offset = listing["next_offset"]
174                        if offset is None:
175                            break
176                ordered = [records[key] for key in sorted(records)]
177                start, end = args["offset"], args["offset"] + min(10, args["limit"])
178                page = {
179                    "observations": ordered[start:end],
180                    "offset": start,
181                    "next_offset": end if end < len(ordered) else None,
182                    "total": len(ordered),
183                }
184    except (OSError, ValueError) as exc:
185        return Outcome(status="blocked", reason="observation_unavailable", content=str(exc))
186    # Wire previews need only the locator and paging fields; full provenance
187    # stays in the archive and trace rather than every prompt.
188    if args["observation_id"]:
189        page = {
190            key: value
191            for key, value in page.items()
192            if key not in {"sha256", "tool", "call_id"}
193        }
194    else:
195        page["observations"] = [
196            {key: value for key, value in ref.items() if key != "sha256"}
197            for ref in page["observations"]
198        ]
199    return Outcome(status="success", content=json.dumps(page, ensure_ascii=False))

203def continuation_sources(evidence, previous_run_id, sources):
204    sources = dict(sources)
205    directory = evidence.path("runs", previous_run_id, "observations")
206    for path in directory.glob("*.json"):
207        evidence.read_observation(previous_run_id, path.stem, limit=1)
208        sources[path.stem] = previous_run_id
209    return sources

212def prepare_model_history(evidence, state, history, identity, settings, sources, origins, preview):
213    """Validate and compact a candidate before archiving its native source."""
214    evidence.validate_native_history(history)
215    converted = canonicalize(history, provider=identity["provider"],
216                             protocol=identity["protocol"], model=identity["model"])
217    state = state.model_copy(deep=True)
218    redact_skill_snapshots(state, evidence.redactor)
219    state.messages = evidence.redactor.value(state.messages, mask_fields=False)
220    candidate = evidence.redact_history(converted.messages)
221    report = preview(state, candidate, settings, sources)
222    origin = archive_native_history(evidence, state.run_id, history, converted, origins)
223    return state, candidate, report, origin
