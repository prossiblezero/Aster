17def safety_identity(identity: dict) -> dict:
18    identity = {"read_parallelism": 1, "reuse_environment": False, **identity}
19    model_fields = {
20        "provider",
21        "profile",
22        "model",
23        "protocol",
24        "auth",
25        "stream",
26        "context_tokens",
27        "output_tokens",
28    }
29    return {key: value for key, value in identity.items() if key not in model_fields}

32def redact_skill_snapshots(state: RunState, redactor: Redactor) -> None:
33    # New model credentials may match text selected before the switch.
34    if any(redactor.text(name) != name for name in {*state.active_skills, *state.skill_snapshots}):
35        raise ValueError("skill_name_contains_credential")
36    for name, snapshot in state.skill_snapshots.items():
37        body = redactor.text(snapshot.body)
38        if body != snapshot.body:
39            state.skill_snapshots[name] = SkillSnapshot(
40                body=body, sha256=hashlib.sha256(body.encode()).hexdigest()
41            )

55def load_checkpoint(
56    evidence: EvidenceStore,
57    workspace: Workspace,
58    settings: Settings,
59    run_id: str,
60    *,
61    allow_model_change: bool,
62    adopt_workspace: bool,
63) -> SessionCheckpoint:
64    """Validate persisted identity, history and archive ownership before runtime mutation."""
65    data = json.loads(evidence.path("runs", run_id, "checkpoint.json").read_text())
66    if data.get("checkpoint_version", 1) not in {1, 2}:
67        raise ValueError("unsupported_checkpoint_version: use the Aster version that saved this run")
68    assert_settled_tools(evidence, run_id, data)
69    # Opt-in workers share the workspace. Adopting edits cannot erase a child's
70    # uncertain effects merely because worker_start already returned to the parent.
71    for event in evidence.events(run_id):
72        if event["event"] == "worker_started":
73            child = event["child_run_id"]
74            child_data = json.loads(evidence.path("runs", child, "checkpoint.json").read_text())
75            if child_data["state"].get("parent_run_id") != run_id:
76                raise ValueError("worker_checkpoint_identity_mismatch")
77            assert_settled_tools(evidence, child, child_data)
78    current_identity = settings.identity()
79    # Old checkpoints predate named providers and streaming; their defaults are explicit.
80    saved_identity = {
81        "provider": "openai",
82        "profile": None,
83        "auth": "required",
84        "stream": False,
85        "read_parallelism": 1,
86        "reuse_environment": False,
87        **data["runtime_identity"],
88    }
89    model_changed = saved_identity != current_identity
90    if model_changed and (
91        not allow_model_change
92        or safety_identity(saved_identity) != safety_identity(current_identity)
93    ):
94        differences = ", ".join(
95            key for key in sorted(set(saved_identity) | set(current_identity))
96            if saved_identity.get(key) != current_identity.get(key)
97        )
98        raise ValueError(
99            "runtime_identity_mismatch: " + differences
100            + ". Restore these explicit settings from the saved checkpoint; "
101            "--allow-model-change permits only model settings, not safety or capability changes."
102        )
103    if data["workspace_hash"] != workspace.fingerprint() and not adopt_workspace:
104        raise ValueError("workspace_mismatch")
105    state = RunState.model_validate(data["state"])
106    redact_skill_snapshots(state, evidence.redactor)
107    state.messages = evidence.redactor.value(state.messages, mask_fields=False)
108    if not state.terminal and "initial_text" not in data:
109        raise ValueError("legacy_checkpoint_missing_diff_baseline")
110    history_groups(data.get("structured_history", []))
111    evidence.validate_native_history(data.get("structured_history", []))
112    history = evidence.redact_history(data.get("structured_history", []))
113    sources = data.get("archive_sources", {})
114    if not isinstance(sources, dict):
115        raise ValueError("invalid_archive_sources")
116    for observation_id, source_run in sources.items():
117        source = json.loads(evidence.path("runs", source_run, "checkpoint.json").read_text())
118        if source["state"]["session_id"] != state.session_id or safety_identity(
119            source["runtime_identity"]
120        ) != safety_identity(current_identity):
121            raise ValueError("archive_source_identity_mismatch")
122        evidence.read_observation(source_run, observation_id, limit=1)
123    return SessionCheckpoint(
124        data=data,
125        state=state,
126        history=history,
127        sources=sources,
128        origins=list(data.get("history_origins", [])),
129        saved_identity=saved_identity,
130        model_changed=model_changed,
131    )

134def assert_settled_tools(evidence: EvidenceStore, run_id: str, data: dict) -> None:
135    marker = evidence.path("runs", run_id, "inflight.json")
136    if not marker.exists():
137        return
138    pending = json.loads(marker.read_text())
139    answered = {
140        item.get("call_id")
141        for item in data.get("structured_history", [])
142        if item.get("type") == "function_call_output"
143    }
144    calls = pending.get("calls", [])
145    if not calls or any(call.get("call_id") not in answered for call in calls):
146        raise ValueError(
147            "interrupted_tool_batch: tool effects may have happened without a saved result. "
148            "Inspect the workspace and .aster/runs/" + run_id
149            + "/ events and inflight.json; start a new task with the confirmed state. "
150            "Resume will not replay uncertain actions; --adopt-workspace cannot bypass this."
151        )

154def save_checkpoint(evidence, state, *, workspace_hash, workspace_snapshot, runtime_identity,
155                    structured_history, initial_text, initial_modes, recovery_used, stall_items,
156                    archive_sources, history_origins, continued_from):
157    evidence.save(
158        ("runs", state.run_id, "checkpoint.json"),
159        {
160            "checkpoint_version": 2,
161            "state": state.model_dump(mode="json"),
162            "workspace_hash": workspace_hash,
163            "workspace": workspace_snapshot,
164            "runtime_identity": runtime_identity,
165            "structured_history": structured_history,
166            "initial_text": initial_text,
167            "initial_modes": initial_modes,
168            "recovery_used": recovery_used,
169            "stall_items": list(stall_items),
170            "archive_sources": archive_sources,
171            "history_origins": history_origins,
172            "continued_from": continued_from,
173        },
174        preserve_native_history=True,
175    )
