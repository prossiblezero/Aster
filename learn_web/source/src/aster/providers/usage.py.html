35def _numeric_usage(raw):
36    """Keep known counters, never arbitrary strings or provider-supplied mapping keys."""
37
38    def counts(value):
39        if not isinstance(value, dict):
40            return {}
41        return {
42            key: value[key]
43            for key in sorted(_USAGE_FIELDS)
44            if type(value.get(key)) is int and 0 <= value[key] <= 2**63 - 1
45        }
46
47    result = counts(raw)
48    if not isinstance(raw, dict):
49        return result
50    for key in (
51        "input_tokens_details",
52        "output_tokens_details",
53        "prompt_tokens_details",
54        "completion_tokens_details",
55        "cache_creation",
56    ):
57        if isinstance(raw.get(key), dict):
58            result[key] = counts(raw[key])
59    attribution = raw.get("attribution")
60    if isinstance(attribution, dict):
61        result["attribution"] = attr = {}
62        items = attribution.get("items")
63        if isinstance(items, dict):
64            attr["items_count"] = len(items)
65            # IDs and content are deliberately omitted; top-level item counters suffice.
66            attr["items"] = [counts(value) for value in islice(items.values(), 128)]
67        fields = attribution.get("request_fields")
68        if isinstance(fields, dict):
69            attr["request_fields_count"] = len(fields)
70            attr["request_fields"] = {
71                key: counts(fields[key])
72                for key in ("tools", "instructions", "messages", "input", "system")
73                if key in fields
74            }
75    return result

78def _usage_warnings(raw, usage):
79    flags = []
80    for left, right in (("input_tokens", "prompt_tokens"), ("output_tokens", "completion_tokens")):
81        if left in raw and right in raw and raw[left] != raw[right]:
82            flags.append(left + "_aliases_disagree")
83    if all(key in usage for key in ("input_tokens", "output_tokens", "total_tokens")):
84        if usage["total_tokens"] != usage["input_tokens"] + usage["output_tokens"]:
85            flags.append("total_tokens_disagree")
86    for part, total in (
87        ("cache_read_input_tokens", "input_tokens"),
88        ("reasoning_tokens", "output_tokens"),
89    ):
90        if part in usage and total in usage and usage[part] > usage[total]:
91            flags.append(part + "_exceeds_" + total)
92    attr = raw.get("attribution", {})
93    items, fields = attr.get("items", []), attr.get("request_fields", {})
94    if (
95        items
96        and len(items) == attr.get("items_count")
97        and len(fields) == attr.get("request_fields_count")
98        and all("input_tokens" in part for part in [*items, *fields.values()])
99        and "input_tokens" in usage
100    ):
101        total = sum(part["input_tokens"] for part in [*items, *fields.values()])
102        if total != usage["input_tokens"]:
103            flags.append("input_attribution_disagrees")
104    return flags

107def normalize_usage(raw: dict, protocol: str = "responses") -> dict[str, int]:
108    """Known token counts plus per-metric coverage; absent is unknown, not zero."""
109    raw = raw if isinstance(raw, dict) else {}
110
111    def find(*path):
112        value = raw
113        for key in path:
114            value = value.get(key) if isinstance(value, dict) else None
115        return value if type(value) is int and 0 <= value <= 2**63 - 1 else None
116
117    # Some compatible gateways use Responses names in Chat usage envelopes.
118    inputs, outputs = find("input_tokens"), find("output_tokens")
119    if inputs is None:
120        inputs = find("prompt_tokens")
121    if outputs is None:
122        outputs = find("completion_tokens")
123    counts = {"input_tokens": inputs, "output_tokens": outputs}
124    if protocol == "anthropic":
125        counts.update(
126            uncached_input_tokens=inputs,
127            cache_read_input_tokens=find("cache_read_input_tokens"),
128            cache_creation_input_tokens=find("cache_creation_input_tokens"),
129        )
130        components = [
131            counts[key]
132            for key in (
133                "uncached_input_tokens",
134                "cache_read_input_tokens",
135                "cache_creation_input_tokens",
136            )
137        ]
138        # Anthropic cache tokens are additional input; OpenAI cache tokens are a subset.
139        counts["input_tokens"] = (
140            sum(components) if all(value is not None for value in components) else None
141        )
142    else:
143        input_details = (
144            "prompt_tokens_details" if protocol == "chat" else "input_tokens_details"
145        )
146        output_details = (
147            "completion_tokens_details" if protocol == "chat" else "output_tokens_details"
148        )
149        counts["cache_read_input_tokens"] = find(input_details, "cached_tokens")
150        counts["reasoning_tokens"] = find(output_details, "reasoning_tokens")
151    total = find("total_tokens")
152    if total is None and counts["input_tokens"] is not None and outputs is not None:
153        total = counts["input_tokens"] + outputs
154    counts["total_tokens"] = total
155    usage = {"usage_requests": 1}
156    for name, value in counts.items():
157        if value is not None:
158            usage[name] = value
159            usage[f"{name}_requests"] = 1
160    return usage

164class RequestAccounting:
165    def __init__(self, settings, secrets):
166        self.settings = settings
167        self.redaction_secrets = tuple(secrets)
168        self.usage = {}
169        self.diagnostics = {}
170
171    def record(self, data):
172        usage = normalize_usage(data.get("usage"), self.settings.protocol)
173        for key, value in usage.items():
174            self.usage[key] = self.usage.get(key, 0) + value
175        return usage

178    def diagnose(self, payload, data, response, attempt, started, category, usage, callback):
179        """Only bounded metadata and allowlisted numeric usage leave this boundary."""
180        redactor = Redactor(self.redaction_secrets)
181
182        def label(value, limit=160):
183            if not isinstance(value, str):
184                return ""
185            value = redactor.text(value)
186            value = re.sub(r"https?://\S+", "[provider URL]", value)
187            value = re.sub(
188                r"(?i)\b(password|api[_ -]?key|authorization|secret)\s*[:=]\s*\S+",
189                r"\1=[REDACTED]",
190                value,
191            )
192            return value[:limit]
193
194        data = data if isinstance(data, dict) else {}
195        raw = _numeric_usage(data.get("usage"))
196        returned_model = data.get("model")
197        flags = _usage_warnings(raw, usage)
198        if isinstance(returned_model, str) and returned_model != self.settings.model:
199            flags.append("response_model_mismatch")
200        structure = []
201        terminal_status = data.get("status", data.get("stop_reason", ""))
202        output = data.get("output", data.get("content", []))
203        if self.settings.protocol == "chat":
204            choices = data.get("choices")
205            if isinstance(choices, list) and choices and isinstance(choices[0], dict):
206                terminal_status = choices[0].get("finish_reason", "")
207            output = (
208                [c.get("message") for c in choices[:32] if isinstance(c, dict)]
209                if isinstance(choices, list)
210                else []
211            )
212        if isinstance(output, list):
213            for item in output[:32]:
214                if not isinstance(item, dict):
215                    structure.append({"type": "malformed"})
216                    continue
217                row = {"type": label(item.get("type", item.get("role", "unknown")), 80)}
218                for key in (
219                    "text",
220                    "arguments",
221                    "thinking",
222                    "encrypted_content",
223                    "content",
224                    "reasoning_content",
225                    "reasoning",
226                ):
227                    if isinstance(item.get(key), str):
228                        row[key + "_length"] = len(item[key])
229                tool_calls = item.get("tool_calls")
230                if isinstance(tool_calls, list):
231                    row["tool_call_count"] = len(tool_calls)
232                    row["tool_calls"] = [
233                        {
234                            "type": label(call.get("type"), 80),
235                            "arguments_length": len(call["function"]["arguments"])
236                            if isinstance(call.get("function"), dict)
237                            and isinstance(call["function"].get("arguments"), str)
238                            else 0,
239                        }
240                        for call in tool_calls[:32]
241                        if isinstance(call, dict)
242                    ]
243                if isinstance(item.get("input"), dict):
244                    row["argument_keys_count"] = len(item["input"])
245                content = item.get("content")
246                if isinstance(content, list):
247                    row["content_count"] = len(content)
248                    row["content"] = [
249                        {
250                            "type": label(block.get("type"), 80),
251                            "text_length": len(block["text"])
252                            if isinstance(block.get("text"), str)
253                            else 0,
254                        }
255                        for block in content[:32]
256                        if isinstance(block, dict)
257                    ]
258                structure.append(row)
259        diagnostic = {
260            "requested_model": label(self.settings.model),
261            "response_model": label(returned_model),
262            "protocol": self.settings.protocol,
263            "response_id": label(data.get("id")),
264            "request_ids": {
265                key: label(response.headers[key])
266                for key in ("x-request-id", "request-id", "anthropic-request-id")
267                if response is not None and key in response.headers
268            },
269            # httpx uses this same JSON encoder for the actual json= request body.
270            "payload_sha256": hashlib.sha256(
271                httpx.Request("POST", "https://unused", json=payload).content
272            ).hexdigest(),
273            "attempt": attempt,
274            "elapsed_seconds": round(time.monotonic() - started, 6),
275            "classification": category,
276            "terminal_status": label(terminal_status, 80),
277            "output_count": len(output) if isinstance(output, list) else None,
278            "output_structure": structure,
279            "output_text_length": len(data["output_text"])
280            if isinstance(data.get("output_text"), str)
281            else 0,
282            "raw_usage": raw,
283            "usage": dict(usage),
284            "usage_observed": isinstance(data.get("usage"), dict),
285            "flags": flags,
286        }
287        self.diagnostics = diagnostic
288        if callback is not None:
289            try:
290                callback(deepcopy({"type": "diagnostic", **diagnostic}))
291            except Exception as exc:
292                logging.getLogger(__name__).warning(
293                    "Provider observer failed: %s", type(exc).__name__
294                )
295        return diagnostic
