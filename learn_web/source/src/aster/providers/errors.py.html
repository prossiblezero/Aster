35def transient_error_code(error):
36    """Only known service failures are retryable; error prose is not a contract."""
37    code = error.get("code", error.get("type")) if isinstance(error, dict) else None
38    return code if isinstance(code, str) and code in {
39        "server_error", "overloaded_error", "rate_limit_exceeded", "rate_limit_error",
40    } else ""

43def retry_request(category, attempt, empty_retries):
44    retry = category in {"transport", "transient_http", "transient_provider"}
45    if category == "empty_response" and empty_retries == 0:
46        empty_retries += 1
47        retry = True
48    return retry and attempt < 3, empty_retries

51def retry_delay(attempt, retry_after=0):
52    return min(2, max(0.25 * 2 ** (attempt - 1), retry_after))
