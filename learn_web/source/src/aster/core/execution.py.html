1"""Bounded detached read execution; caller owns authorization and ordered publication."""
2import asyncio
3import time
4
5from ..schema import Outcome
6
7
8async def execute_reads(preparations, parallelism):
9    semaphore = asyncio.Semaphore(parallelism)
10
11    async def execute(prepared):
12        async with semaphore:
13            call, registry = prepared
14
15            def read():
16                started = time.monotonic()
17                try:
18                    outcome = asyncio.run(registry.execute(call))
19                except Exception as exc:
20                    outcome = Outcome(
21                        status="error", reason=type(exc).__name__, content=str(exc)
22                    )
23                return outcome, (time.monotonic() - started) * 1000
24
25            work = asyncio.create_task(asyncio.to_thread(read))
26            try:
27                return await asyncio.shield(work)
28            except asyncio.CancelledError:
29                # Python cannot cancel a running thread. Keep the operation gate
30                # until it exits; cancelled results never enter Runtime state.
31                await work
32                raise
33
34    jobs = [
35        asyncio.create_task(execute(prepared))
36        for prepared in preparations
37    ]
38    group = asyncio.gather(*jobs, return_exceptions=True)
39    try:
40        return await asyncio.shield(group)
41    except asyncio.CancelledError:
42        for job in jobs:
43            job.cancel()
44        while not group.done():
45            try:
46                await asyncio.shield(group)
47            except asyncio.CancelledError:
48                continue
49        raise
