10# Task requirements are guessed from the request with keywords, not with a model. Wrongly
11# demanding an edit is the costly mistake: the completion check would push the agent to change
12# code nobody asked it to change. So English keywords must be whole words ("dispatch" is not
13# "patch"), a negated verb never counts as an edit request, and a negation makes the task
14# read-only only when it targets the workspace itself ("do not modify files", "不要修改代码"),
15# not a property of the solution ("without changing existing values", "不要修改原始数据").
16_EDIT_EN = re.compile(
17    r"\b(?:re)?(?:fix(?:es|ing)?|implement(?:s|ing)?|writ(?:e|es|ing)|edit(?:s|ing)?|"
18    r"modif(?:y|ies|ying)|creat(?:e|es|ing)|patch(?:es|ing)?|updat(?:e|es|ing))\b"
19)
20_EDIT_ZH = re.compile(r"修改|修复|实现|写入|创建|更新")
21_NEGATOR_EN = r"(?:do\s+not|don'?t|never|not|no\s+need\s+to|without)"
22_NEGATOR_ZH = r"(?:不要|不用|无需|不需要|不必|别|请勿|勿|禁止|不准|不许|不得|不能|不)"
23_CHANGE_EN = r"(?:re)?(?:fix|implement|writ|edit|modif|creat|patch|updat|chang|touch|alter)\w*"
24_CHANGE_ZH = r"(?:修改|改动|更改|变更|改变|修复|更新|创建|写入|实现|改)"
25# Mentions that are not requests: negated verbs, and "how was it implemented" questions.
26_NOT_A_REQUEST = re.compile(
27    rf"\b{_NEGATOR_EN}\s+(?:\w+\s+)?{_CHANGE_EN}"
28    rf"|{_NEGATOR_ZH}\s*(?:再|去|做|作|进行)?\s*(?:任何)?{_CHANGE_ZH}"
29    r"|(?:怎么|如何|怎样)[^，。,.!?？！；;\n]{0,12}?实现的"
30)
31_WORKSPACE_EN = r"(?:files?|code(?:base)?|source(?:\s+code)?|repo(?:sitory)?|workspace|project|directory|folder|anything)"
32_WORKSPACE_ZH = r"(?:文件|代码|源码|源代码|仓库|工作区|项目|东西|内容)"
33_CLAUSE_END_EN = r"(?=\s*(?:$|[.,;:!?)\]]|(?:and|but|just|only|please|at\s+all|in\s+any\s+way)\b))"
34_CLAUSE_END_ZH = r"(?=$|[。．.!！?？,，;；、:：)）\s]|只|仅|请|就|即|直接|也)"
35_EXCEPT = r"(?!\s+(?:to|in|outside|other|except|beyond|besides|apart))"
36_READ_ONLY = re.compile(
37    rf"\b{_NEGATOR_EN}\s+(?:\w+\s+)?(?:modif|chang|edit|touch|alter)\w*"
38    rf"(?:\s+(?:any|the|these|this|my|our|existing))?(?:\s+{_WORKSPACE_EN})?{_CLAUSE_END_EN}"
39    rf"|\b(?:make\s+no|(?:do\s+not|don'?t|never)\s+make\s+any|without\s+(?:making\s+)?any)"
40    rf"\s+(?:code\s+|file\s+)?changes\b{_EXCEPT}"
41    rf"|\bno\s+(?:code|file)\s+changes\b{_EXCEPT}"
42    rf"|\bno\s+changes\s+to\s+(?:the\s+|any\s+)?{_WORKSPACE_EN}\b"
43    rf"|\bread[- ]only\b"
44    rf"|{_NEGATOR_ZH}\s*(?:再|去|做|作|进行)?\s*(?:任何)?(?:修改|改动|更改|变更|改变|改)\s*(?:任何)?{_WORKSPACE_ZH}?{_CLAUSE_END_ZH}"
45    rf"|{_NEGATOR_ZH}\s*对\s*(?:任何)?{_WORKSPACE_ZH}\s*(?:做|作|进行)?\s*(?:任何)?(?:修改|改动|更改|变更|改变)"
46    r"|只读"
47)
48_VERIFY = re.compile(r"(?<![a-z])(?:py|unit)?tests?(?:ing|ed|cases?)?(?![a-z])|\bverification\b|验证|测试")
49_OBSERVE = re.compile(r"\b(?:inspect|review|check)|查看|检查|分析")

52class Verifier:
53    def __init__(self, workspace: Workspace):
54        self.workspace = workspace
55
56    def observe(self, state: RunState, outcome: Outcome) -> None:
57        if outcome.status == "success" and outcome.data.get("path"):
58            state.observed = True
59        if outcome.changed_paths:
60            state.changed_paths = sorted(set(state.changed_paths + outcome.changed_paths))
61            state.verification = Verification(
62                state="missing", workspace_hash=self.workspace.fingerprint()
63            )
64        if outcome.data.get("is_verification"):
65            anonymous = outcome.data.get("test_evidence_source") in {"exit_code", "inline_script"}
66            for test, status in outcome.data.get("test_results", {}).items():
67                if status == "failed" and anonymous:
68                    # Evidence without a test identity (an exit code; an assertion script under a
69                    # fixed id) cannot be rerun as "the same test". It fails verification only.
70                    continue
71                if status == "failed":
72                    state.verification_failures[test] = outcome.data.get("command", "")
73                    state.environment_failures.pop(test, None)
74                elif status == "passed":
75                    state.verification_failures.pop(test, None)
76                    state.environment_failures.pop(test, None)
77                elif status == "network_unavailable":
78                    # Cannot pass without network; reported as unverified, never as fixed.
79                    state.verification_failures.pop(test, None)
80                    state.environment_failures[test] = outcome.data.get("command", "")
81            state.baseline_signatures.update(outcome.data.get("baseline_signatures", {}))
82            # Latest evidence wins: a test stays a pre-existing candidate only while it keeps
83            # failing exactly as it does on the original files.
84            identical = set(outcome.data.get("baseline_identical", []))
85            for test, status in outcome.data.get("test_results", {}).items():
86                if status == "failed" and test in identical:
87                    # Pinned to the tested code: any later edit voids it until the test reruns.
88                    state.preexisting_failures[test] = outcome.data.get("tested_workspace_hash", "")
89                else:
90                    state.preexisting_failures.pop(test, None)
91        if (
92            outcome.data.get("is_verification")
93            and (outcome.status == "success" or outcome.data.get("network_only_failures"))
94            and outcome.reason != "no_tests_collected"
95        ):
96            state.verification = Verification(
97                state="failed" if state.verification_failures else "passed",
98                workspace_hash=self.workspace.fingerprint(),
99                evidence_id=outcome.evidence_id,
100                command=outcome.data.get("command", ""),
101            )
102        elif outcome.data.get("is_verification"):
103            state.verification = Verification(
104                state="failed",
105                workspace_hash=self.workspace.fingerprint(),
106                evidence_id=outcome.evidence_id,
107            )
108
109    def accepted_preexisting(self, state: RunState) -> dict[str, str]:
110        """Failures that stop blocking final delivery. All must hold: the agent changed code,
111        was shown this evidence at a completion check at least once, and the test still
112        fails exactly as on the original files when run against the current workspace."""
113        if not state.changed_paths or state.final_revisions < 1:
114            return {}
115        current = self.workspace.fingerprint()
116        return {test: state.verification_failures[test] for test, tested in state.preexisting_failures.items()
117                if test in state.verification_failures and tested == current}
118
119    def final(self, state: RunState) -> VerifierResult:
120        task = state.task
121        if (
122            state.verification.state == "passed"
123            and state.verification.workspace_hash != self.workspace.fingerprint()
124        ):
125            state.verification.state = "missing"
126        if task.require_observation and not state.observed:
127            return VerifierResult(passed=False, reason="observation_missing")
128        if task.require_edit and not state.changed_paths:
129            return VerifierResult(passed=False, reason="edit_missing")
130        accepted = self.accepted_preexisting(state)
131        blocking = {test: command for test, command in state.verification_failures.items()
132                    if test not in accepted}
133        if task.require_verification and blocking:
134            return VerifierResult(passed=False, reason="verification_unresolved_failures",
135                                  details={"failures": blocking})
136        verified = state.verification.state == "passed" or (
137            accepted and state.verification.state == "failed"
138            and state.verification.workspace_hash == self.workspace.fingerprint()
139        )
140        if task.require_verification and not verified:
141            return VerifierResult(passed=False, reason=f"verification_{state.verification.state}")
142        if state.pending_requirements:
143            return VerifierResult(passed=False, reason="requirements_unreviewed",
144                                  details={"todo_ids": [item["id"] for item in state.pending_requirements]})
145        for name in task.required_artifacts:
146            try:
147                if self.workspace.path(name).is_file():
148                    continue
149            except ValueError:
150                pass
151            return VerifierResult(passed=False, reason=f"artifact_missing:{name}")
152        if state.cancelled_requirements:
153            return VerifierResult(passed=False, reason="requirements_cancelled")
154        return VerifierResult(
155            passed=True,
156            reason="requirements_satisfied",
157            details={"changed_paths": state.changed_paths,
158                     **({"environment_failures": state.environment_failures}
159                        if state.environment_failures else {}),
160                     **({"preexisting_failures": accepted} if accepted else {})},
161        )
162
163    @staticmethod
164    def infer(prompt: str) -> dict[str, bool]:
165        text = prompt.casefold()
166        requested = _NOT_A_REQUEST.sub(" ", text)
167        edit = bool(_EDIT_EN.search(requested) or _EDIT_ZH.search(requested)) and not _READ_ONLY.search(text)
168        return {
169            "require_edit": edit,
170            "require_verification": edit and bool(_VERIFY.search(text)),
171            "require_observation": bool(_OBSERVE.search(text)) or edit,
172        }
