3from __future__ import annotations

5from ..schema import Outcome, RunState, Verification, VerifierResult
6from ..storage.workspace import Workspace

9class Verifier:
10    def __init__(self, workspace: Workspace):
11        self.workspace = workspace
12
13    def observe(self, state: RunState, outcome: Outcome) -> None:
14        if outcome.status == "success" and outcome.data.get("path"):
15            state.observed = True
16        if outcome.changed_paths:
17            state.changed_paths = sorted(set(state.changed_paths + outcome.changed_paths))
18            state.verification = Verification(
19                state="missing", workspace_hash=self.workspace.fingerprint()
20            )
21        if outcome.data.get("is_verification"):
22            for test, status in outcome.data.get("test_results", {}).items():
23                if status == "failed":
24                    state.verification_failures[test] = outcome.data.get("command", "")
25                elif status == "passed":
26                    state.verification_failures.pop(test, None)
27        if (
28            outcome.data.get("is_verification")
29            and outcome.status == "success"
30            and outcome.reason != "no_tests_collected"
31        ):
32            state.verification = Verification(
33                state="failed" if state.verification_failures else "passed",
34                workspace_hash=self.workspace.fingerprint(),
35                evidence_id=outcome.evidence_id,
36                command=outcome.data.get("command", ""),
37            )
38        elif outcome.data.get("is_verification"):
39            state.verification = Verification(
40                state="failed",
41                workspace_hash=self.workspace.fingerprint(),
42                evidence_id=outcome.evidence_id,
43            )
44
45    def final(self, state: RunState) -> VerifierResult:
46        task = state.task
47        if (
48            state.verification.state == "passed"
49            and state.verification.workspace_hash != self.workspace.fingerprint()
50        ):
51            state.verification.state = "missing"
52        if task.require_observation and not state.observed:
53            return VerifierResult(passed=False, reason="observation_missing")
54        if task.require_edit and not state.changed_paths:
55            return VerifierResult(passed=False, reason="edit_missing")
56        if task.require_verification and state.verification_failures:
57            return VerifierResult(passed=False, reason="verification_unresolved_failures",
58                                  details={"failures": state.verification_failures})
59        if task.require_verification and state.verification.state != "passed":
60            return VerifierResult(passed=False, reason=f"verification_{state.verification.state}")
61        if state.pending_requirements:
62            return VerifierResult(passed=False, reason="requirements_unreviewed",
63                                  details={"todo_ids": [item["id"] for item in state.pending_requirements]})
64        for name in task.required_artifacts:
65            try:
66                if self.workspace.path(name).is_file():
67                    continue
68            except ValueError:
69                pass
70            return VerifierResult(passed=False, reason=f"artifact_missing:{name}")
71        if state.cancelled_requirements:
72            return VerifierResult(passed=False, reason="requirements_cancelled")
73        return VerifierResult(
74            passed=True,
75            reason="requirements_satisfied",
76            details={"changed_paths": state.changed_paths},
77        )

79    @staticmethod
80    def infer(prompt: str) -> dict[str, bool]:
81        text = prompt.casefold()
82        edit_requested = any(
83            x in text
84            for x in (
85                "fix",
86                "implement",
87                "write",
88                "edit",
89                "modify",
90                "create",
91                "patch",
92                "update",
93                "修改",
94                "修复",
95                "实现",
96                "写入",
97                "创建",
98                "更新",
99            )
100        )
101        edit_negated = any(
102            x in text
103            for x in (
104                "do not modify",
105                "don't modify",
106                "without modifying",
107                "without changing",
108                "read-only",
109                "read only",
110                "不要修改",
111                "无需修改",
112                "只读",
113            )
114        )
115        edit = edit_requested and not edit_negated
116        verify = any(x in text for x in ("test", "tests", "pytest", "verification", "验证", "测试"))
117        observe = any(x in text for x in ("inspect", "review", "check", "查看", "检查", "分析"))
118        return {
119            "require_edit": edit,
120            "require_verification": edit and verify,
121            "require_observation": observe or edit,
122        }
