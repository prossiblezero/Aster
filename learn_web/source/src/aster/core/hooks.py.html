1"""Small injectable lifecycle hooks; hook failures do not widen authority."""
2
3from collections.abc import Iterable
4from dataclasses import dataclass, field
5
6from ..schema import Outcome, RunState, ToolCall
7
8
9@dataclass(frozen=True)
10class HookResult:
11    action: str = "continue"
12    feedback: str = ""
13    metadata: dict = field(default_factory=dict)
14
15
16class Hooks:
17    def __init__(self, plugins: Iterable[object] = ()):
18        self.plugins = list(plugins)
19
20    def add(self, plugin: object) -> None:
21        self.plugins.append(plugin)
22
23    @staticmethod
24    def _safe(plugin, name, *args) -> HookResult:
25        callback = getattr(plugin, name, None)
26        if callback is None:
27            return HookResult()
28        try:
29            result = callback(*args)
30            return result if isinstance(result, HookResult) else HookResult()
31        except Exception as exc:
32            # A failed precondition must not silently authorize a tool.
33            return HookResult(
34                "deny" if name == "before_tool" else "continue",
35                f"hook_error:{name}:{type(exc).__name__}",
36            )
37
38    def before_tool(self, state: RunState, call: ToolCall) -> HookResult:
39        if call.call_id in state.used_call_ids:
40            return HookResult("deny", "duplicate_call_id")
41        for plugin in self.plugins:
42            result = self._safe(plugin, "before_tool", state, call)
43            if result.action in {"deny", "stop"}:
44                return result
45        return HookResult()
46
47    def after_tool(self, state: RunState, call: ToolCall, outcome: Outcome) -> HookResult:
48        return self._run("after_tool", state, call, outcome)
49
50    def after_failure(self, state: RunState, call: ToolCall, outcome: Outcome) -> HookResult:
51        return self._run("after_failure", state, call, outcome)
52
53    def stop(self, state: RunState) -> HookResult:
54        return self._run("stop", state)
55
56    def _run(self, name, *args) -> HookResult:
57        feedback, metadata = [], {}
58        for plugin in self.plugins:
59            result = self._safe(plugin, name, *args)
60            if result.feedback:
61                feedback.append(result.feedback)
62            metadata.update(result.metadata)
63            if result.action == "stop":
64                return HookResult("stop", "\n".join(feedback), metadata)
65        return HookResult("continue", "\n".join(feedback), metadata)
