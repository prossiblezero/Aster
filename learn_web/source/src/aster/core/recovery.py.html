12class StallDetector:
13    def __init__(self, window=4):
14        self.items = deque(maxlen=max(4, window))
15
16    def observe(self, call: ToolCall, outcome: Outcome) -> str:
17        # Empty stdout is normal for successful edits. Progress breaks an old loop.
18        if outcome.changed_paths:
19            self.items.clear()
20            return ""
21        key = (
22            call.name,
23            hashlib.sha256(json.dumps(call.arguments, sort_keys=True).encode()).hexdigest(),
24            hashlib.sha256(
25                json.dumps([outcome.status, outcome.reason, outcome.content]).encode()
26            ).hexdigest(),
27        )
28        self.items.append(key)
29        if len(self.items) >= 2 and self.items[-1] == self.items[-2]:
30            return "repeated_identical"
31        if len(self.items) >= 4:
32            recent = list(self.items)[-4:]
33            if recent[0] == recent[2] and recent[1] == recent[3] and recent[0][:2] != recent[1][:2]:
34                return "oscillation"
35        if (
36            outcome.status != "success"
37            and len(self.items) >= 3
38            and len({x[2] for x in list(self.items)[-3:]}) == 1
39        ):
40            return "output_loop"
41        return ""

44def completion_feedback(state: RunState) -> str:
45    missing = []
46    task = state.task
47    if state.pending_requirements:
48        missing.append("review open requirement todos " + ", ".join(
49            item["id"] for item in state.pending_requirements
50        ) + " via todo_update with the actual acceptance result; a worker handoff is not acceptance")
51    if task.require_observation and not state.observed:
52        missing.append("inspect the workspace")
53    if task.require_edit and not state.changed_paths:
54        missing.append("make the requested edit")
55    if task.require_verification and state.verification_failures:
56        pending = list(state.verification_failures.items())[:3]
57        missing.append(
58            "resolve previously failing tests: "
59            + "; ".join(f"{test} (from {command})" for test, command in pending)
60            + ". Rerun those cases after fixing them; passing a different selection does not "
61            "resolve them. Report remaining failures honestly"
62        )
63    elif task.require_verification and state.verification.state != "passed":
64        missing.append(
65            "run a successful verification command via run_tests "
66            "(run_shell checks do not update formal verification)"
67        )
68    return "Completion requirements missing: " + ", ".join(missing) if missing else ""
