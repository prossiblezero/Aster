69class SkillSnapshot(Record):
70    body: str
71    sha256: str
72
73    @model_validator(mode="after")
74    def validate_hash(self):
75        if self.sha256 != hashlib.sha256(self.body.encode()).hexdigest():
76            raise ValueError("skill_snapshot_hash_mismatch")
77        return self
