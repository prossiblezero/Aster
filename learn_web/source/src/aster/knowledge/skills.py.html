24def _valid_name(name: str) -> bool:
25    return bool(
26        SKILL_NAME.fullmatch(name)
27        and name not in {".", ".."} | CONTROL
28        and not sensitive(Path(name))
29    )

32def _metadata(lines: list[str], name: str) -> dict[str, str]:
33    """A small, documented frontmatter subset; no YAML tags, aliases or objects."""
34    values = {}
35    index = 0
36    while index < len(lines):
37        line = lines[index]
38        index += 1
39        if not line.strip() or line.lstrip().startswith("#"):
40            continue
41        match = re.fullmatch(r"([A-Za-z][A-Za-z0-9_-]*):\s*(.*)", line)
42        if not match:
43            raise ValueError("invalid skill metadata: expected an unindented key: value")
44        key, value = match.groups()
45        block = []
46        while index < len(lines) and (not lines[index].strip() or lines[index].startswith(" ")):
47            block.append(lines[index])
48            index += 1
49        if key not in {"name", "description"}:
50            continue  # Unknown metadata cannot grant permissions or execute anything.
51        if key in values:
52            raise ValueError(f"invalid skill metadata: duplicate {key}")
53        if value in {"|", "|-", "|+", ">", ">-", ">+"}:
54            value_text = textwrap.dedent("\n".join(block)).strip()
55            value = re.sub(r"(?<=\S)\n(?=\S)", " ", value_text) if value[0] == ">" else value_text
56        else:
57            if any(item.strip() for item in block):
58                raise ValueError(f"invalid skill metadata: unsupported multiline {key}")
59            if value.startswith('"'):
60                try:
61                    value = json.loads(value)
62                except json.JSONDecodeError as exc:
63                    raise ValueError(f"invalid skill metadata: quoted {key}") from exc
64            elif value.startswith("'"):
65                if not re.fullmatch(r"'(?:[^']|'')*'", value):
66                    raise ValueError(f"invalid skill metadata: quoted {key}")
67                value = value[1:-1].replace("''", "'")
68            elif (
69                not value
70                or value[0] in "[{&*!"
71                or value.casefold() in {"true", "false", "null", "~"}
72                or re.fullmatch(r"[-+]?\d+(?:\.\d+)?", value)
73            ):
74                raise ValueError(f"invalid skill metadata: {key} must be a string")
75        if not isinstance(value, str) or not value.strip():
76            raise ValueError(f"invalid skill metadata: {key} must be a nonempty string")
77        values[key] = value.strip()
78    if values.get("name", name) != name:
79        raise ValueError("invalid skill metadata: name must match its file or package directory")
80    if len(values.get("description", "")) > 1000:
81        raise ValueError("skill description exceeds 1000 characters")
82    return values

85class Skills:
86    def __init__(self, workspace: Path):
87        self.workspace = workspace.resolve()
88        self.dirs = [self.workspace / ".aster" / "skills", self.workspace / ".skills"]
89
90    def _safe_path(self, path: Path) -> bool:
91        if not path.is_relative_to(self.workspace):
92            return False
93        cursor = self.workspace
94        for part in path.relative_to(self.workspace).parts:
95            cursor /= part
96            if part == ".." or cursor.is_symlink():
97                return False
98        return path.resolve().is_relative_to(self.workspace)
99
100    @contextmanager
101    def _directory(self, path: Path):
102        """Anchor reads/listings so swapped symlink parents cannot redirect them."""
103        if not self._safe_path(path):
104            raise FileNotFoundError("skill directory unavailable")
105        directory = os.open(self.workspace, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
106        try:
107            for part in path.relative_to(self.workspace).parts:
108                child = os.open(
109                    part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory
110                )
111                os.close(directory)
112                directory = child
113            yield directory
114        finally:
115            os.close(directory)
116
117    @contextmanager
118    def _open(self, path: Path):
119        if not self._safe_path(path) or not path.is_file():
120            raise FileNotFoundError("skill file unavailable")
121        if path.stat().st_size > MAX_SKILL_BYTES:
122            raise ValueError("skill exceeds 20 KiB")
123        descriptor = None
124        with self._directory(path.parent) as directory:
125            descriptor = os.open(
126                path.name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=directory
127            )
128            try:
129                info = os.fstat(descriptor)
130                if not stat.S_ISREG(info.st_mode):
131                    raise ValueError("skill must be a regular file")
132                if info.st_size > MAX_SKILL_BYTES:
133                    raise ValueError("skill exceeds 20 KiB")
134                with os.fdopen(descriptor, "rb") as stream:
135                    descriptor = None
136                    yield stream
137            finally:
138                if descriptor is not None:
139                    os.close(descriptor)

141    def _header(self, stream, name: str) -> dict[str, str]:
142        first = stream.readline(5)
143        if first not in {b"---\n", b"---\r\n"}:
144            stream.seek(0)
145            return {}
146        used = len(first)
147        lines = []
148        while True:
149            line = stream.readline(MAX_METADATA_BYTES - used + 1)
150            used += len(line)
151            if used > MAX_METADATA_BYTES:
152                raise ValueError("skill metadata exceeds 4 KiB")
153            if not line:
154                raise ValueError("invalid skill metadata: missing closing ---")
155            if line.rstrip(b"\r\n") == b"---":
156                return _metadata(lines, name)
157            try:
158                lines.append(line.decode("utf-8").rstrip("\r\n"))
159            except UnicodeDecodeError as exc:
160                raise ValueError("skill metadata must be UTF-8") from exc

162    def _find(self, name: str) -> Path:
163        if not _valid_name(name):
164            raise ValueError("invalid skill name")
165        for root in self.dirs:
166            for path in (root / name / "SKILL.md", root / f"{name}.md"):
167                if self._safe_path(path) and path.is_file():
168                    return path
169        raise FileNotFoundError(name)
170
171    def _names(self) -> list[str]:
172        names = set()
173        for root in self.dirs:
174            if not self._safe_path(root) or not root.is_dir():
175                continue
176            with self._directory(root) as directory, os.scandir(directory) as entries:
177                for count, entry in enumerate(entries, 1):
178                    if count > MAX_CATALOG_ENTRIES:
179                        raise ValueError("skill directory exceeds 256 entries")
180                    if entry.is_symlink():
181                        continue
182                    name = (
183                        entry.name if entry.is_dir(follow_symlinks=False) else Path(entry.name).stem
184                    )
185                    if _valid_name(name) and (
186                        entry.is_dir(follow_symlinks=False) or entry.name.endswith(".md")
187                    ):
188                        names.add(name)
189        return sorted(names)
190
191    def _resources(self, base: Path) -> tuple[list[str], bool]:
192        resources, pending = [], [(base, 0)]
193        visited, truncated = 0, False
194        while pending:
195            directory, depth = pending.pop()
196            if not self._safe_path(directory):
197                continue
198            with self._directory(directory) as descriptor, os.scandir(descriptor) as entries:
199                for entry in entries:
200                    visited += 1
201                    if visited > MAX_RESOURCE_ENTRIES or len(resources) >= MAX_RESOURCES:
202                        return sorted(resources), True
203                    path = directory / entry.name
204                    relative = path.relative_to(base)
205                    if (
206                        entry.is_symlink()
207                        or sensitive(relative)
208                        or any(p in CONTROL | IGNORED for p in relative.parts)
209                    ):
210                        continue
211                    if entry.is_dir(follow_symlinks=False):
212                        if depth < MAX_RESOURCE_DEPTH:
213                            pending.append((path, depth + 1))
214                        else:
215                            truncated = True
216                    elif entry.is_file(follow_symlinks=False) and relative != Path("SKILL.md"):
217                        if entry.stat(follow_symlinks=False).st_size <= MAX_SKILL_BYTES:
218                            resources.append(relative.as_posix())
219        return sorted(resources), truncated

221    def catalog(self) -> list[dict]:
222        """Read headers and bounded filenames only; malformed entries remain diagnosable."""
223        records = []
224        for name in self._names():
225            try:
226                path = self._find(name)
227            except FileNotFoundError:
228                continue
229            package = path.name == "SKILL.md"
230            record = {
231                "name": name,
232                "description": "",
233                "path": path.relative_to(self.workspace).as_posix(),
234                "base_dir": path.parent.relative_to(self.workspace).as_posix() if package else None,
235                "resources": [],
236                "resources_truncated": False,
237            }
238            try:
239                with self._open(path) as stream:
240                    metadata = self._header(stream, name)
241                record["description"] = metadata.get("description", "")
242                if package:
243                    record["resources"], record["resources_truncated"] = self._resources(
244                        path.parent
245                    )
246            except (OSError, ValueError) as exc:
247                record["error"] = str(exc)
248            records.append(record)
249        return records
250
251    def list(self) -> list[str]:
252        return [item["name"] for item in self.catalog() if "error" not in item]

254    def load(self, name: str, resource: str = "") -> str:
255        path = self._find(name)
256        with self._open(path) as stream:
257            self._header(stream, name)
258            data = stream.read(MAX_SKILL_BYTES + 1) if not resource else b""
259            if stream.tell() > MAX_SKILL_BYTES:
260                raise ValueError("skill exceeds 20 KiB")
261        if resource:
262            if path.name != "SKILL.md":
263                raise ValueError("legacy skills have no package resources")
264            relative = Path(resource)
265            if (
266                relative.is_absolute()
267                or "\\" in resource
268                or "\x00" in resource
269                or ".." in relative.parts
270                or ":" in resource
271                or sensitive(relative)
272                or any(p in CONTROL | IGNORED for p in relative.parts)
273            ):
274                raise ValueError("invalid skill resource path")
275            path = path.parent / relative
276            with self._open(path) as stream:
277                data = stream.read(MAX_SKILL_BYTES + 1)
278        if len(data) > MAX_SKILL_BYTES:
279            raise ValueError("skill exceeds 20 KiB")
280        try:
281            return data.decode("utf-8")
282        except UnicodeDecodeError as exc:
283            raise ValueError("skill content must be UTF-8") from exc
