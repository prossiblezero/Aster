26def sensitive(path: Path) -> bool:
27    return any(
28        p.startswith(".env")
29        or p in {".ssh", "id_rsa", "id_ed25519"}
30        or p.endswith((".pem", ".key"))
31        for p in path.parts
32    )

35def digest(data: bytes | str) -> str:
36    if isinstance(data, str):
37        data = data.encode()
38    return hashlib.sha256(data).hexdigest()

41class Workspace:
42    def __init__(self, root: Path):
43        root.mkdir(parents=True, exist_ok=True)
44        self.root = root.resolve(strict=True)
45        self._digests: dict[str, tuple[tuple[int, ...], str]] = {}
46        if not self.root.is_dir():
47            raise ValueError("workspace must be a directory")
48        self._git_repository = False
49        result = self._git("rev-parse", "--is-inside-work-tree")
50        self._git_repository = bool(result and result.stdout.strip() == b"true")
51        # Explicitly selecting an ignored directory makes it the ignore boundary.
52        # Its own rules still apply; the parent repository and index stay untouched.
53        self._git_root_ignored = bool(
54            self._git_repository
55            and self._git("check-ignore", "--no-index", "-q", "--", ".").returncode == 0
56        )
57
58    def _git(self, *arguments: str, scoped: bool = False) -> subprocess.CompletedProcess | None:
59        try:
60            with tempfile.TemporaryDirectory(prefix="aster-ignore-") if scoped else nullcontext() as directory:
61                # An inherited GIT_DIR/INDEX_FILE must not redirect workspace checks.
62                env = {k: v for k, v in os.environ.items() if not k.startswith("GIT_")}
63                options = []
64                if scoped:
65                    options = ["--work-tree", str(self.root)]
66                    # No index: parent-relative paths must not exempt local names.
67                    env["GIT_INDEX_FILE"] = str(Path(directory) / "index")
68                result = subprocess.run(
69                    ["git", "--no-optional-locks", "-c", "core.fsmonitor=false", "-C", str(self.root), *options, *arguments],
70                    capture_output=True,
71                    timeout=10,
72                    env=env,
73                )
74        except FileNotFoundError:
75            if not self._git_repository:
76                return None
77            raise BoundaryError("git_ignore_unavailable") from None
78        except (OSError, subprocess.TimeoutExpired) as exc:
79            raise BoundaryError("git_ignore_unavailable") from exc
80        if self._git_repository and result.returncode not in (0, 1):
81            raise BoundaryError("git_ignore_unavailable")
82        return result

84    def path(self, raw: str, *, write: bool = False) -> Path:
85        candidate = Path(raw)
86        candidate = candidate if candidate.is_absolute() else self.root / candidate
87        try:
88            relative = candidate.relative_to(self.root)
89        except ValueError as exc:
90            raise BoundaryError("path_escape") from exc
91        cursor = self.root
92        for part in relative.parts:
93            if part == "..":
94                raise BoundaryError("path_escape")
95            cursor /= part
96            try:
97                kind = cursor.lstat().st_mode
98            except (FileNotFoundError, NotADirectoryError):
99                continue
100            if stat.S_ISLNK(kind):
101                raise BoundaryError("symlink_path")
102            if not (stat.S_ISREG(kind) or stat.S_ISDIR(kind)):
103                raise BoundaryError("special_file")
104        resolved = candidate.resolve()
105        if not resolved.is_relative_to(self.root):
106            raise BoundaryError("path_escape")
107        if any(p in CONTROL for p in relative.parts):
108            raise BoundaryError("protected_metadata")
109        if sensitive(relative):
110            raise BoundaryError("sensitive_path")
111        if write and resolved == self.root:
112            raise BoundaryError("cannot_write_workspace_root")
113        if write and self._git_repository:
114            if self._git_root_ignored:
115                tracked = self._git("ls-files", "--cached", "-z").stdout.split(b"\0")
116                if os.fsencode(relative) in tracked:
117                    return resolved
118            result = self._git("check-ignore", "-q", "--", str(relative), scoped=self._git_root_ignored)
119            if result.returncode == 0:
120                raise BoundaryError("git_ignored_path")
121        return resolved

123    def files(self, raw: str = ".") -> list[Path]:
124        start = self.path(raw)
125        if start.is_file():
126            return [start]
127        ignored = set()
128        tracked = set()
129        tracked_directories = set()
130        if self._git_root_ignored:
131            result = self._git("ls-files", "--cached", "-z")
132            tracked = {self.root / os.fsdecode(p) for p in result.stdout.split(b"\0") if p}
133            tracked_directories = {parent for path in tracked for parent in path.parents}
134        if self._git_repository:
135            result = self._git(
136                "ls-files", "--others", "--ignored", "--exclude-standard", "--directory", "-z",
137                scoped=self._git_root_ignored,
138            )
139            ignored = {self.root / os.fsdecode(p) for p in result.stdout.split(b"\0") if p}
140
141        ignored_names = {os.path.normpath(str(p)) for p in ignored}
142        tracked_names = {os.path.normpath(str(p)) for p in tracked}
143        tracked_directory_names = {os.path.normpath(str(p)) for p in tracked_directories}
144        start_name = os.path.normpath(str(start))
145        # An ignored ancestor marks every descendant; tracked paths stay visible.
146        inherited = any(os.path.normpath(str(p)) in ignored_names for p in (start, *start.parents))
147        if inherited and start_name not in tracked_names and start_name not in tracked_directory_names:
148            return []
149        result = []
150
151        def walk(directory: str, marked: bool):
152            try:
153                entries = sorted(os.scandir(directory), key=lambda entry: entry.name)
154            except OSError:
155                return
156            directories = []
157            for entry in entries:
158                if entry.is_symlink():
159                    continue
160                if entry.is_dir(follow_symlinks=False):
161                    directories.append(entry)
162                    continue
163                name, full = entry.name, entry.path
164                if (
165                    name in CONTROL
166                    or sensitive(Path(name))
167                    or name.endswith(".pyc")
168                    or (full not in tracked_names and (marked or full in ignored_names))
169                ):
170                    continue
171                if not entry.is_file(follow_symlinks=False):
172                    raise BoundaryError("special_file")
173                result.append(Path(full))
174                if len(result) > 10000:
175                    raise BoundaryError("workspace exceeds 10000 files; narrow --cwd")
176            for entry in directories:
177                name, full = entry.name, entry.path
178                hidden = marked or full in ignored_names
179                if (
180                    name in CONTROL | IGNORED
181                    or sensitive(Path(name))
182                    or (hidden and full not in tracked_names and full not in tracked_directory_names)
183                ):
184                    continue
185                walk(full, hidden)
186
187        walk(start_name, inherited)
188        return result

190    def snapshot(self) -> dict[str, str]:
191        """File identity includes permission bits; read evidence separately hashes text.
192
193        Digests are reused while a file's stat identity is unchanged, as git's index does.
194        A file whose mtime or ctime falls inside the racy window is always rehashed, so a
195        second write within one timestamp tick cannot hide behind a cached digest.
196        """
197        result, cache = {}, {}
198        racy_after = time.time_ns() - RACY_WINDOW_NS
199        root = str(self.root) + os.sep
200        for path in self.files():
201            name = str(path)[len(root):]
202            info = path.stat()
203            key = (info.st_mode, info.st_size, info.st_mtime_ns, info.st_ctime_ns, info.st_ino, info.st_dev)
204            cached = self._digests.get(name)
205            if cached is not None and cached[0] == key:
206                digest_value = cached[1]
207            else:
208                h = hashlib.sha256()
209                h.update(f"{stat.S_IMODE(info.st_mode):04o}\0".encode())
210                with path.open("rb") as stream:
211                    for chunk in iter(lambda: stream.read(1024 * 1024), b""):
212                        h.update(chunk)
213                digest_value = h.hexdigest()
214            if max(info.st_mtime_ns, info.st_ctime_ns) < racy_after:
215                cache[name] = (key, digest_value)
216            result[name] = digest_value
217        self._digests = cache
218        return result

220    def mode_snapshot(self) -> dict[str, int]:
221        return {
222            str(path.relative_to(self.root)): stat.S_IMODE(path.stat().st_mode)
223            for path in self.files()
224        }

226    def fingerprint(self, snapshot: dict[str, str] | None = None) -> str:
227        return digest(json.dumps(self.snapshot() if snapshot is None else snapshot, sort_keys=True))

229    def text_snapshot(self) -> dict[str, str]:
230        """Text baseline for a human-readable diff; hashes remain drift evidence."""
231        result = {}
232        for path in self.files():
233            try:
234                name = str(path.relative_to(self.root))
235                result[name] = self.read(name)
236            except (UnicodeError, ValueError):
237                continue
238        return result

240    def read(self, raw: str) -> str:
241        path = self.path(raw)
242        if path.stat().st_size > 2 * 1024 * 1024:
243            raise ValueError("file exceeds 2 MiB text limit")
244        return path.read_text(encoding="utf-8")

246    @staticmethod
247    def changes(before: dict[str, str], after: dict[str, str]) -> list[str]:
248        return sorted(k for k in before.keys() | after.keys() if before.get(k) != after.get(k))

250    def diff(self, originals: dict[str, str], modes: dict[str, int] | None = None) -> str:
251        current = {str(p.relative_to(self.root)): p for p in self.files()}
252        chunks = []
253        for name in sorted(originals.keys() | current.keys()):
254            if modes is not None and name in modes and name in current:
255                mode = stat.S_IMODE(current[name].stat().st_mode)
256                if mode != modes[name]:
257                    chunks.extend(
258                        [
259                            f"diff --git a/{name} b/{name}\n",
260                            f"old mode {stat.S_IFREG | modes[name]:06o}\n",
261                            f"new mode {stat.S_IFREG | mode:06o}\n",
262                        ]
263                    )
264            try:
265                after = current[name].read_text() if name in current else ""
266            except (UnicodeError, OSError):
267                continue
268            chunks.extend(
269                difflib.unified_diff(
270                    originals.get(name, "").splitlines(True),
271                    after.splitlines(True),
272                    f"a/{name}",
273                    f"b/{name}",
274                )
275            )
276        return "".join(chunks)
