1"""Durable operation keys: replay settled results, never repeat uncertain effects."""
2
3from __future__ import annotations
4
5import hashlib
6import json
7
8from ..schema import Outcome
9
10
11class OperationJournal:
12    def __init__(self, evidence, session_id, call, safety_identity):
13        key = call.idempotency_key or call.call_id
14        key_hash = hashlib.sha256(key.encode()).hexdigest()
15        self.evidence = evidence
16        self.parts = ("operations", session_id, key_hash + ".json")
17        self.fingerprint = hashlib.sha256(json.dumps(
18            [call.name, call.arguments, safety_identity], sort_keys=True, ensure_ascii=False,
19        ).encode()).hexdigest()
20
21    def lookup(self):
22        path = self.evidence.path(*self.parts)
23        if not path.exists():
24            return None
25        saved = json.loads(path.read_text())
26        if saved.get("fingerprint") != self.fingerprint:
27            return Outcome(status="blocked", reason="idempotency_conflict",
28                           content="This operation key was already used with different arguments or permissions.")
29        if saved.get("state") != "settled":
30            return Outcome(status="blocked", reason="idempotency_uncertain",
31                           content="This operation may already have taken effect. Inspect its effects before starting a new operation.")
32        outcome = Outcome.model_validate(saved["outcome"])
33        # A receipt describes a past action; it must not manufacture fresh edits or test evidence.
34        outcome.changed_paths, outcome.evidence_id, outcome.duration_ms = [], "", 0
35        outcome.data = {"idempotency_replayed": True, "original_status": outcome.status}
36        return outcome
37
38    def reserve(self):
39        self.evidence.save(self.parts, {"fingerprint": self.fingerprint, "state": "pending"})
40
41    def settle(self, outcome):
42        if outcome.data.get("safe_to_retry") or outcome.status == "blocked":
43            self.evidence.path(*self.parts).unlink(missing_ok=True)
44        elif outcome.status not in {"timeout", "cancelled"} and not outcome.data.get("uncertain"):
45            self.evidence.save(self.parts, {"fingerprint": self.fingerprint, "state": "settled",
46                                            "outcome": outcome.model_dump(mode="json")})
