17class ReadFile(Record):
18    path: str
19    start: int = Field(default=1, ge=1)
20    end: int = Field(default=200, ge=1)
21
22    @model_validator(mode="after")
23    def valid_range(self):
24        if self.end < self.start or self.end - self.start > 2000:
25            raise ValueError("line range must be ordered and at most 2000 lines")
26        return self

48class PatchEdit(Record):

50        min_length=1,

52    )

56class PatchFile(Record):
57    path: str
58    edits: list[PatchEdit] | None = Field(default=None, min_length=1, max_length=100)
59    old_text: str | None = Field(default=None, min_length=1)
60    new_text: str | None = None
61
62    @model_validator(mode="after")
63    def valid_edits(self):
64        if self.edits is not None:
65            if self.old_text is not None or self.new_text is not None:
66                raise ValueError("Use edits or old_text/new_text, not both")
67        elif self.old_text is None or self.new_text is None:
68            raise ValueError("Provide edits or both old_text and new_text")
69        return self

314def definitions(
315    names: list[str] | None = None, *, available: list[str] | None = None
316) -> list[dict]:
317    selected = list(SCHEMAS) if names is None else names
318    result = []
319    for name in selected:
320        model, description, _effect = SCHEMAS[name]
321        schema = model.model_json_schema()
322        if name == "tool_load":
323            schema["properties"]["names"]["items"]["enum"] = sorted(
324                SCHEMAS if available is None else available
325            )
326        schema.setdefault("additionalProperties", False)
327        result.append(
328            {
329                "type": "function",
330                "name": name,
331                "description": description,
332                "parameters": schema,
333                "strict": False,
334            }
335        )
336    return result
