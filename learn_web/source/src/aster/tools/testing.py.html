28# A failure is attributed to missing network access only when the error is an unreachable
29# network or failed name resolution AND the traceback shows a real socket call. Refused
30# connections stay ordinary failures: a local service the task must start may be down.

32    r"\[Errno (?:101|113|-2|-3|-5)\]|Network is unreachable|No route to host"
33    r"|Temporary failure in name resolution|Name or service not known"

43def failure_signature(detail: str) -> str:
44    """Exception type and message of a failure, with run-specific noise removed."""
45    lines = [line.strip() for line in detail.splitlines() if line.strip()]
46    errors = [line[1:].strip() for line in lines if line.startswith("E ")]
47    candidates = errors or [line for line in lines if re.match(r"[\w.]+(Error|Exception|Failed|Exit)\b", line)]
48    text = (candidates[-1] if candidates else (lines[-1] if lines else ""))[:500]
49    text = re.sub(r"0x[0-9a-fA-F]+", "0x?", text)
50    # Temporary paths differ per run: absolute ones, and the copy's own TMPDIR (.tmp/...),
51    # which programs print relative to the working directory.
52    text = re.sub(r"(/tmp|/var/tmp|(?<![\w/.])\.tmp)/[^\s:'\"]+", "<tmp>", text)
53    text = re.sub(r"\btmp[a-z0-9_]{6,}\b", "tmp?", text)
54    return re.sub(r"__aster_[\w-]+", "__aster_?", text)

63def jest_entry(argv: list[str]) -> int | None:
64    """Length of a Jest entry point (jest, npx jest, node <...>/jest/bin/jest.js), else None."""
65    if argv[:1] == ["jest"]:

75def parse_jest(text: str, root: Path | None = None) -> list[dict[str, str]]:
76    """Convert Jest's built-in --json report into the shared case records."""
77    data = json.loads(text)
78    if not isinstance(data, dict) or not isinstance(data.get("testResults"), list):
79        raise ValueError("invalid_jest_report")
80    cases = []
81    for suite in data["testResults"]:
82        name = Path(suite.get("name", ""))
83        try:
84            name = name.relative_to(root) if root is not None else name
85        except ValueError:
86            pass
87        assertions = suite.get("assertionResults") or []
88        if not assertions and suite.get("status") == "failed":
89            # The file failed before declaring tests (syntax error, missing module).
90            cases.append({"name": str(name), "classname": str(name), "status": "error",
91                          "detail": (suite.get("message") or "")[-20000:]})
92        for case in assertions:
93            status = {"passed": "passed", "failed": "failed"}.get(case.get("status"), "skipped")
94            title = case.get("fullName") or " ".join([*case.get("ancestorTitles", []), case.get("title", "")])
95            cases.append({"name": title.strip(), "classname": str(name), "status": status,
96                          "detail": "\n".join(case.get("failureMessages") or [])[-20000:]})
97    return cases

100def parse_junit(text: str) -> list[dict[str, str]]:
101    root = ET.fromstring(text)
102    if root.tag not in {"testsuites", "testsuite"}:
103        raise ValueError("invalid_junit_report")
104    cases = []
105    for case in root.iter("testcase"):
106        if not case.get("name"):
107            raise ValueError("unnamed_junit_test")
108        children = {child.tag for child in case}
109        status = next(
110            (
111                result
112                for tag, result in (
113                    ("error", "error"),
114                    ("failure", "failed"),
115                    ("skipped", "skipped"),
116                )
117                if tag in children
118            ),
119            "passed",
120        )
121        detail = ""
122        for child in case:
123            if child.tag in {"failure", "error"}:
124                detail += (child.get("message") or "") + "\n" + (child.text or "")
125        cases.append(
126            {"name": case.get("name"), "classname": case.get("classname", ""), "status": status,
127             "detail": detail[-20000:]}
128        )
129    return cases

132async def run_tests(
133    workspace: Workspace,
134    sandbox: SandboxRunner,
135    command: str,
136    timeout: int,
137    *,
138    runner: type[SandboxRunner] = SandboxRunner,
139    code: str = "",
140    baseline: dict[str, str] | None = None,
141    known: dict[str, str] | None = None,
142) -> Outcome:
143    """Run tests; failures identical on the task's original files are marked pre-existing.
144
145    The runtime decides from evidence: the same command runs on a copy restored to the
146    original text, and a failure is pre-existing only when it also fails there with the
147    same normalized signature. known maps already-checked ids to their original
148    signature ("" = did not fail there), so each id is rerun at most once.
149    """
150    result = await _run_tests(workspace, sandbox, command, timeout, runner=runner, code=code)
151    signatures = result.data.get("failure_signatures", {})
152    failing = {key for key, status in result.data.get("test_results", {}).items() if status == "failed"}
153    # Exit-code failures owe no rerun (see Verifier.observe), so need no original-file evidence.
154    if baseline is None or code or not failing or result.data.get("test_evidence_source") == "exit_code":
155        return result
156    known = dict(known or {})
157    unchecked = failing - known.keys()
158    if unchecked:
159        try:
160            with tempfile.TemporaryDirectory(prefix="aster-baseline-") as directory:
161                target = Workspace(Path(directory))
162                for source in workspace.files():
163                    copied = Path(directory) / source.relative_to(workspace.root)
164                    copied.parent.mkdir(parents=True, exist_ok=True)
165                    shutil.copy2(source, copied)
166                current = target.text_snapshot()
167                for name in current.keys() - baseline.keys():
168                    (Path(directory) / name).unlink()
169                for name, text in baseline.items():
170                    if current.get(name) != text:
171                        path = Path(directory) / name
172                        path.parent.mkdir(parents=True, exist_ok=True)
173                        path.write_text(text, encoding="utf-8")
174                original = await _run_tests(target, sandbox, command, timeout, runner=runner,
175                                            readable=workspace.root)
176        except (OSError, ValueError, UnicodeError):
177            original = None
178        if original is not None and (original.status == "success" or original.reason == "tests_failed"):
179            original_results = original.data.get("test_results", {})
180            original_signatures = original.data.get("failure_signatures", {})
181            checked = {key: (original_signatures.get(key, "") if original_results.get(key) == "failed" else "")
182                       for key in unchecked}
183            known.update(checked)
184            result.data["baseline_signatures"] = checked
185    identical = sorted(key for key in failing if known.get(key) and known[key] == signatures.get(key))
186    # Evidence only: the verifier decides at final delivery (see Verifier.final).
187    result.data["baseline_identical"] = identical
188    if identical:
189        result.content += (
190            f"\n[Aster: {len(identical)} failing test(s) fail with the same error on the task's original "
191            f"files, so your changes have not affected them: {', '.join(identical[:10])}. If one covers the "
192            "behavior you were asked to change, your fix is not working yet. Otherwise it is a pre-existing "
193            "failure: do not chase it; report it as unverified in your final answer.]"
194        )
195    return result

213def inline_script_as_test(script: str) -> str:
214    """A plain assertion script runs as one test instead of being reported as collecting none."""
215    return (
216        "import os\nimport runpy\nimport unittest as _aster_unittest\n\n\n"
217        "class InlineScript(_aster_unittest.TestCase):\n"
218        "    def test_script(self):\n"
219        "        try:\n"
220        f"            runpy.run_path(os.path.join(os.path.dirname(__file__), {script!r}), run_name='__main__')\n"
221        "        except SystemExit as exit:\n"
222        "            if exit.code not in (None, 0):\n"
223        "                raise AssertionError(f'script exited with {exit.code!r}') from None\n"
224    )

230def inline_script_command(command: str) -> str:
231    """`python -c '...'` or a `python - <<EOF` heredoc that asserts: the script, else ""."""
232    if match := HEREDOC_SCRIPT.fullmatch(command):
233        script = match.group(2)
234    else:
235        try:
236            argv = shlex.split(command)
237        except ValueError:
238            return ""
239        if len(argv) != 3 or argv[0] not in {"python", "python3"} or argv[1] != "-c":
240            return ""
241        script = argv[2]
242    return script if re.search(r"(?:^|;)\s*assert\b", script, re.M) or TEST_DEFINITION.search(script) else ""

245def project_test_command(command: str) -> bool:
246    """A project's own build/test chain, judged by exit code; ad-hoc shell checks are not.
247
248    Every step is a plain command (no pipes, `||`, `;`, loops or redirected input, which would hide
249    or replace the test's exit code); at least one runs a known test tool or a test executable."""
250    if re.search(r"\||;|<|`|\$\(|\n", command):
251        return False
252    tests = False
253    test_builds = set()  # outputs of compiling a test source, which a later step may run
254    for step in command.split("&&"):
255        try:
256            words = shlex.split(step)
257        except ValueError:
258            return False
259        while words and re.fullmatch(r"[A-Za-z_][A-Za-z0-9_]*=.*", words[0]):
260            words = words[1:]
261        if not words or words[0] in {"cd", "mkdir", "true", "rm", "cp", "touch"}:
262            continue  # setup in a disposable copy
263        program = words[0]
264        if program in {"for", "while", "until", "if", "case", "do", "then", "{", "(", "!"}:
265            return False
266        name = Path(program).name
267        if any(NOT_TESTS.search(word) for word in words):
268            continue  # linters, formatters and type checkers may run but do not verify behaviour
269        if program in PACKAGE_RUNNERS:
270            tests = tests or any(re.search(r"(?i)test|jest|vitest|mocha", word) for word in words[1:])
271            continue  # install/build scripts are setup
272        if program in TEST_TOOLS or (name in {"bash", "sh"} and len(words) > 1
273                                     and re.search(r"(?i)test", Path(words[1]).name)) or (
274                re.fullmatch(r"python[0-9.]*", name) and words[1:2] == ["-m"] and words[2:3] in (["pytest"], ["unittest"])):
275            tests = True
276        elif (re.search(r"(?i)test", name) and program not in {"test", "["}) or program in test_builds:
277            tests = True
278        elif program not in COMPILERS and name not in COMPILERS:
279            return False
280        elif "-o" in words[:-1] and any(re.search(r"(?i)test", Path(w).name) for w in words[1:]):
281            output = words[words.index("-o") + 1]
282            test_builds |= {output, "./" + output.removeprefix("./")}
283    return tests

286async def _run_command_tests(
287    workspace: Workspace, sandbox: SandboxRunner, command: str, timeout: int, *,
288    runner: type[SandboxRunner] = SandboxRunner, code: str = "", readable: Path | None = None,
289) -> Outcome:
290    """Run a project's own test command in a disposable copy; the exit code is the evidence."""
291    if code or not project_test_command(command):
292        return Outcome(status="blocked", reason="test_command_not_supported",
293                       content="Use a recognizable test command; other commands belong in run_shell.")
294    if sandbox.settings.sandbox == "off":
295        return Outcome(status="blocked", reason="isolation_required_for_scope")
296    tested_fingerprint = workspace.fingerprint()
297    tested_environment = sandbox.environment_generation
298    with tempfile.TemporaryDirectory(prefix="aster-tests-") as directory:
299        target = Path(directory)
300        for source in workspace.files():
301            copied = target / source.relative_to(workspace.root)
302            copied.parent.mkdir(parents=True, exist_ok=True)
303            shutil.copy2(source, copied)
304        # Build products and caches may be written; they land in this discarded copy only.
305        settings = sandbox.settings.model_copy(update={"read_only": False, "write_scope": None})
306        command_runner = runner(settings, environment=sandbox)
307        command_runner.readable_roots = (readable or workspace.root,)
308        result = await command_runner.run(Workspace(target), command, timeout)
309    key = "command: " + command
310    result.changed_paths, result.partial_success = [], False
311    result.data.update(test_evidence_source="exit_code", tested_workspace_hash=tested_fingerprint,
312                       tested_environment_generation=tested_environment)
313    # This copy is discarded, so files it refuses to publish do not void the exit code.
314    discarded = result.reason == "sandbox_protected_output" and result.exit_code is not None
315    if (result.status not in {"success", "error"} and not discarded) or result.exit_code is None:
316        result.reason = result.reason or f"test_{result.status}"
317        result.data["test_results"] = {}
318        return result
319    passed = result.exit_code == 0
320    result.status, result.reason = ("success", "") if passed else ("error", "tests_failed")
321    result.data["passed_tests"] = int(passed)
322    result.data["test_results"] = {key: "passed" if passed else "failed"}
323    result.data["failure_signatures"] = {} if passed else {key: failure_signature(result.content[-4000:])}
324    result.content = ("[Aster: no per-test report for this command; verification uses its exit code "
325                      f"({result.exit_code}).]\n" + result.content)
326    if passed and (workspace.fingerprint() != tested_fingerprint
327                   or sandbox.environment_generation != tested_environment):
328        result.status, result.reason = "blocked", "workspace_changed_during_verification"
329    return result

332async def _run_tests(
333    workspace: Workspace,
334    sandbox: SandboxRunner,
335    command: str,
336    timeout: int,
337    *,
338    runner: type[SandboxRunner] = SandboxRunner,
339    code: str = "",
340    readable: Path | None = None,
341) -> Outcome:
342    """Run tests in a disposable copy; publish only structured, version-bound evidence.
343
344    readable is the real workspace the copy stands for (editable installs point at it); a
345    copy of the original files must still see that path, or its failures differ."""
346    if not code and (script := inline_script_command(command)):
347        command, code = "pytest -q", script

354        argv = shlex.split(command)
355    except ValueError:

395    if sandbox.settings.sandbox == "off":
396        return Outcome(status="blocked", reason="isolation_required_for_scope")

398    node = argv[:2] == ["node", "--test"]

404                       content="With code, use pytest, python -m pytest, or python -m unittest without discover.")

406        return Outcome(
407            status="blocked", reason="test_reporter_reserved",
408            content="Remove --test-reporter and --test-reporter-destination; run_tests supplies its own structured reporter. Retry with node --test and your test paths.",
409        )
410    if jest and any(option.split("=")[0] in {"--json", "--outputFile", "--reporters"} for option in options):
411        return Outcome(
412            status="blocked", reason="test_reporter_reserved",
413            content="Remove --json, --outputFile and --reporters; run_tests captures Jest's structured results automatically.",
414        )
415    if go and any(option == "-json" or option.startswith("-json=") for option in options):
416        return Outcome(
417            status="blocked", reason="test_reporter_reserved",
418            content="Remove -json; run_tests captures Go's structured test events automatically.",
419        )
420    tested_fingerprint = workspace.fingerprint()
421    tested_environment = sandbox.environment_generation
422    with tempfile.TemporaryDirectory(prefix="aster-tests-") as directory:

425            copied = target / source.relative_to(workspace.root)
426            copied.parent.mkdir(parents=True, exist_ok=True)
427            shutil.copy2(source, copied)
428        if code:
429            # Temporary tests share the existing sandbox/report path, never the user's files.
430            inline_name = "__aster_inline_" + target.name.replace("-", "_") + ".py"
431            # Only scripts that assert something; printing "passed" is not evidence.
432            script = not TEST_DEFINITION.search(code) and re.search(r"(?:^|;)\s*assert\b", code, re.M) is not None
433            if script:
434                # Tracebacks then point at the agent's own lines in a file of their own.
435                script_name = "__aster_script_" + target.name.replace("-", "_") + ".py"
436                (target / script_name).write_text(code, encoding="utf-8")
437            (target / inline_name).write_text(inline_script_as_test(script_name) if script else code,
438                                              encoding="utf-8")
439            options.append(inline_name)
440        report = "__aster_result_" + target.name + (".json" if jest else ".xml")
441        if jest:

446                "--test",
447                "--test-reporter=junit",

489suite = ET.Element('testsuite')
490observed = False
