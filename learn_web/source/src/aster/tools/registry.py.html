14def validate_arguments(call: ToolCall) -> Record | Outcome:
15    if call.name not in SCHEMAS:
16        return Outcome(status="blocked", reason="unknown_tool")
17    try:
18        return SCHEMAS[call.name][0].model_validate(call.arguments)
19    except ValueError as exc:
20        return Outcome(status="blocked", reason="invalid_arguments", content=str(exc))

33class ToolRegistry:
34    def __init__(self, context: ToolContext, *, validated: tuple[ToolCall, Record] | None = None):
35        self.workspace = WorkspaceTools(context)
36        self.workflow = context.workflow
37        self._validated = (validated[0].model_copy(deep=True), validated[1]) if validated else None
38
39    async def execute(self, call: ToolCall) -> Outcome:
40        args = (self._validated[1] if self._validated and call == self._validated[0]
41                else validate_arguments(call))
42        if isinstance(args, Outcome):
43            return args
44        try:
45            if call.name in WORKFLOW_TOOLS:
46                if self.workflow is None:
47                    return Outcome(status="blocked", reason="workflow_not_implemented", content=call.name)
48                return await self.workflow.execute(call.name, args.model_dump())
49            return await self.workspace.execute_validated(call, args)
50        except Exception as exc:
51            return Outcome(status="error", reason=type(exc).__name__, content=str(exc))
