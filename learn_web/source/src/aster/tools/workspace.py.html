49def text_page(text: str, offset: int, limit: int) -> Outcome:
50    if offset > len(text):
51        return Outcome(status="blocked", reason="page_out_of_range", content=f"total={len(text)}")
52    end = min(offset + limit, len(text))
53    page = {
54        "offset": offset,
55        "next_offset": end if end < len(text) else None,
56        "total": len(text),
57    }
58    return Outcome(status="success", content=json.dumps(page) + "\n" + text[offset:end], data=page)

61def record_page(
62    records: Iterable[str], offset: int, limit: int, *, total: int | None = None
63) -> Outcome:
64    page = list(islice(records, offset, offset + limit + 1))
65    truncated = len(page) > limit
66    page = page[:limit]
67    metadata = {
68        "offset": offset,
69        "count": len(page),
70        "truncated": truncated,
71        "next_offset": offset + len(page) if truncated else None,
72    }
73    if total is not None:
74        metadata["total"] = total
75    return Outcome(status="success", content="\n".join(page), data=metadata)

78def grouped_matches(page: list[tuple[str, int, list[tuple[int, str]]]]) -> str:
79    """Search hits under one heading per file, like ripgrep --heading.
80
81    `12:` marks a matching line and `11-` a context line; overlapping context is printed once
82    and `--` marks a gap. Repeating the path on every line, grep -C style, made search results
83    the largest observations Aster carried (36% of resent tool output on fresh30b).
84    """
85    files: dict[str, dict[int, tuple[str, bool]]] = {}
86    for path, index, window in page:
87        lines = files.setdefault(path, {})
88        for number, text in window:
89            lines[number] = (text, lines.get(number, ("", False))[1] or number == index)
90    gaps = any(len(window) > 1 for _, _, window in page)  # plain matches need no separators
91    out = []
92    for path, lines in files.items():
93        out.append(f"{path}:")
94        last = None
95        for number in sorted(lines):
96            if gaps and last is not None and number > last + 1:
97                out.append("--")
98            text, match = lines[number]
99            out.append(f"{number + 1}{':' if match else '-'} {text}")
100            last = number
101    return "\n".join(out)

104def _numbered(lines: list[str], low: int, high: int) -> list[str]:
105    return [f"{n + 1}: {lines[n]}" for n in range(max(0, low), min(len(lines), high))]

108def edit_preview(text: str, regions: list[tuple[int, int]], limit: int = 80) -> str:
109    """Edited lines with two lines of context, so the model need not reread its own edit."""
110    lines, shown, blocks = text.split("\n"), 0, []
111    for start, end in regions:
112        first = text.count("\n", 0, start)
113        last = text.count("\n", 0, max(start, end - 1))
114        low, high = first - 2, last + 3
115        if blocks and low <= blocks[-1][1]:
116            blocks[-1][1] = max(blocks[-1][1], high)
117        else:
118            blocks.append([low, high])
119    out = []
120    for low, high in blocks:
121        chunk = _numbered(lines, low, high)
122        if shown + len(chunk) > limit:
123            out.append(f"[{len(blocks)} edited regions; preview limited to {limit} lines; use read_file for more]")
124            break
125        out.append("\n".join(chunk))
126        shown += len(chunk)
127    return "Edited lines (current file):\n" + "\n--\n".join(out)

130def tolerant_span(text: str, old: str) -> tuple[int, int] | None:
131    """Locate old_text whose whole lines differ only in surrounding whitespace.
132
133    Mirrors the fallback order of Codex's apply_patch seek_sequence (exact, then trailing,
134    then all surrounding whitespace ignored); only a unique match is accepted.
135    """
136    wanted = old.split("\n")
137    trailing_newline = wanted[-1] == ""
138    if trailing_newline:
139        wanted = wanted[:-1]
140    if not wanted or not any(line.strip() for line in wanted):
141        return None
142    lines = text.split("\n")
143    starts = [0]
144    for line in lines:
145        starts.append(starts[-1] + len(line) + 1)
146    for normalize in (str.rstrip, str.strip):
147        target = [normalize(line) for line in wanted]
148        found = [n for n in range(len(lines) - len(target) + 1)
149                 if [normalize(line) for line in lines[n:n + len(target)]] == target]
150        if len(found) == 1:
151            first, last = found[0], found[0] + len(target) - 1
152            end = starts[last] + len(lines[last]) + (1 if trailing_newline and last + 1 < len(lines) else 0)
153            return starts[first], end
154        if found:
155            return None
156    return None

159def closest_lines(text: str, old: str, limit: int = 3) -> str:
160    """Show where an unmatched old_text most likely belongs, with current line numbers."""
161    lines = text.split("\n")
162    wanted = [line.strip() for line in old.split("\n") if line.strip()]
163    if not wanted:
164        return ""
165    stripped = [line.strip() for line in lines]
166    anchors = [n for n, line in enumerate(stripped) if line == wanted[0]][:limit]
167    if not anchors:
168        best = difflib.get_close_matches(wanted[0], stripped, n=limit, cutoff=0.6)
169        anchors = sorted({stripped.index(match) for match in best})
170    if not anchors:
171        return ""
172    span = min(len(old.split("\n")), 12)
173    return "Closest current lines:\n" + "\n--\n".join(
174        "\n".join(_numbered(lines, n - 2, n + span + 2)) for n in anchors
175    )

178def read_edit_source(path: Path, *, normalize: bool = True) -> tuple[str, str]:
179    """Derive normalized edit text and a raw-byte version from the same bounded read."""
180    limit = 2 * 1024 * 1024
181    with path.open("rb") as stream:
182        data = stream.read(limit + 1)
183    if len(data) > limit:
184        raise ValueError("file exceeds 2 MiB text limit")
185    text = data.decode("utf-8")
186    return (text.replace("\r\n", "\n").replace("\r", "\n") if normalize else text), digest(data)

216    async def _read_file(self, args: ReadFile) -> Outcome:
217        text = self.context.workspace.read(args.path)
218        lines = text.splitlines()
219        selected, characters, truncated = [], 0, False
220        for line in lines[args.start - 1:args.end]:
221            size = len(line) + bool(selected)
222            if characters + size > READ_FILE_CHARACTERS:
223                if not selected:
224                    selected.append(line[:READ_FILE_CHARACTERS])
225                truncated = True
226                break
227            selected.append(line)
228            characters += size
229        end = args.start + len(selected) - 1 if selected else min(args.end, len(lines))
230        return Outcome(
231            status="success",
232            content="\n".join(selected),
233            data={
234                "path": args.path,
235                "start": args.start,
236                "end": end,
237                "source_sha256": digest(text),
238                **({"truncated": True, "next_start": end + 1} if truncated else {}),
239            },
240        )

263    async def _search(self, args: Search) -> Outcome:
264        pattern = args.pattern.casefold()
265        try:
266            regex = re.compile(args.pattern, re.IGNORECASE) if args.regex else None
267        except re.error as exc:
268            return Outcome(status="blocked", reason="invalid_regex", content=str(exc))
269
270        def matches():
271            for path in self.context.workspace.files(args.path):
272                try:
273                    if path.stat().st_size > 2 * 1024 * 1024:
274                        continue
275                    relative = path.relative_to(self.context.workspace.root)
276                    lines = path.read_text(encoding="utf-8").splitlines()
277                    for index, line in enumerate(lines):
278                        if not (regex.search(line[:2000]) if regex is not None
279                                else pattern in line.casefold()):
280                            continue
281                        # One record per match keeps offset counting matches.
282                        low, high = max(0, index - args.context), min(len(lines), index + args.context + 1)
283                        yield str(relative), index, [(n, lines[n][:300]) for n in range(low, high)]
284                except UnicodeError:
285                    continue
286
287        page = list(islice(matches(), args.offset, args.offset + args.limit + 1))
288        truncated = len(page) > args.limit
289        page = page[:args.limit]
290        return Outcome(status="success", content=grouped_matches(page), data={
291            "offset": args.offset, "count": len(page), "truncated": truncated,
292            "next_offset": args.offset + len(page) if truncated else None})

308    async def _patch_file(self, args: PatchFile) -> Outcome:
309        path = self.context.workspace.path(args.path, write=True)
310        raw, fingerprint = read_edit_source(path, normalize=False)
311        relative = str(path.relative_to(self.context.workspace.root))
312        if (
313            self.context.expected_files is not None
314            and fingerprint != self.context.expected_files.get(relative)
315        ):
316            return Outcome(
317                status="blocked",
318                reason="fresh_read_required",
319                content=f"Call read_file on {args.path} again, then rebuild the patch from its current content.",
320            )
321        bom = "\ufeff" if raw.startswith("\ufeff") else ""
322        raw = raw[len(bom) :]
323        text = raw.replace("\r\n", "\n").replace("\r", "\n")
324        edits = args.edits or [PatchEdit(old_text=args.old_text, new_text=args.new_text)]
325        spans, notes = [], []
326        for index, edit in enumerate(edits):
327            old = edit.old_text.replace("\r\n", "\n").replace("\r", "\n")
328            anchored = bool(bom and old.startswith(bom))
329            if bom:
330                old = old.removeprefix(bom)
331            if not old:
332                return Outcome(
333                    status="blocked", reason="patch_empty", content="Match text beyond the BOM"
334                )
335            start = (0 if text.startswith(old) else -1) if anchored else text.find(old)
336            if start < 0 and not anchored and (span := tolerant_span(text, old)):
337                notes.append(f"edits[{index}] matched ignoring whitespace differences")
338                spans.append((*span, edit.new_text))
339                continue
340            if start < 0 or (not anchored and text.find(old, start + 1) >= 0):
341                if start < 0:
342                    hint = closest_lines(text, old)
343                else:
344                    found, position = [], start
345                    while position >= 0 and len(found) < 10:
346                        found.append(text.count("\n", 0, position) + 1)
347                        position = text.find(old, position + 1)
348                    hint = ("Occurrences start at lines " + ", ".join(map(str, found))
349                            + "; include more surrounding text to make old_text unique.")
350                return Outcome(
351                    status="blocked",
352                    reason="patch_nonunique",
353                    content=f"edits[{index}] found {max(2, text.count(old)) if start >= 0 else 0} occurrences; must match exactly once in the original file"
354                    + ("\n" + hint if hint else ""),
355                )
356            spans.append((start, start + len(old), edit.new_text))
357        spans.sort()
358        if any(left[1] > right[0] for left, right in zip(spans, spans[1:], strict=False)):
359            return Outcome(
360                status="blocked",
361                reason="patch_overlap",
362                content="Merge overlapping edits into one replacement",
363            )
364        # Map only edit boundaries back to raw offsets, preserving even mixed line endings.
365        boundaries = {position for start, end, _ in spans for position in (start, end)}
366        offsets = {}
367        raw_index = 0
368        for index in range(len(text) + 1):
369            if index in boundaries:
370                offsets[index] = raw_index
371            raw_index += 2 if raw[raw_index : raw_index + 2] == "\r\n" else 1
372        first_newline = next((i for i, char in enumerate(raw) if char in "\r\n"), None)
373        ending = (
374            ("\r\n" if raw[first_newline : first_newline + 2] == "\r\n" else raw[first_newline])
375            if first_newline is not None
376            else "\n"
377        )
378        for start, end, replacement in reversed(spans):
379            replacement = replacement.replace("\r\n", "\n").replace("\r", "\n")
380            if bom and start == 0:
381                replacement = replacement.removeprefix(bom)
382            raw = raw[: offsets[start]] + replacement.replace("\n", ending) + raw[offsets[end] :]
383        new_text = bom + raw
384        command = self._write_command(relative, new_text)
385        result = await self.context.sandbox.run(
386            self.context.workspace, command, expected_files=self.context.expected_files
387        )
388        if result.status == "success":
389            result.data.update(path=args.path, written_sha256=digest(new_text.encode()))
390            pieces, regions, cursor, size = [], [], 0, 0
391            for start, end, replacement in spans:
392                replacement = replacement.replace("\r\n", "\n").replace("\r", "\n")
393                pieces.append(text[cursor:start])
394                size += start - cursor
395                regions.append((size, size + len(replacement)))
396                pieces.append(replacement)
397                size += len(replacement)
398                cursor = end
399            pieces.append(text[cursor:])
400            preview = edit_preview("".join(pieces), regions)
401            result.content = ("\n".join(notes) + "\n" if notes else "") + (
402                result.content + "\n" if result.content else "") + preview
403        return result

415    async def _shell_tests(self, args: RunShell) -> Outcome | None:
416        """A test command sent to run_shell runs as run_tests would, so it is not run twice."""
417        command = SHELL_OUTPUT_TRIM.sub("", args.command).strip()
418        try:
419            argv = shlex.split(command)
420        except ValueError:
421            return None
422        while argv and re.fullmatch(r"[A-Za-z_][A-Za-z0-9_]*=.*", argv[0]):
423            argv = argv[1:]
424        framework = argv[:1] in (["pytest"], ["jest"], ["npx"]) or argv[:3] in (
425            ["python", "-m", "pytest"], ["python", "-m", "unittest"], ["python3", "-m", "pytest"]
426        ) or argv[:2] in (["node", "--test"], ["go", "test"])
427        if not (framework or project_test_command(command)):
428            return None
429        outcome = await run_tests(
430            self.context.workspace, self.context.sandbox, command.replace("python3 -m", "python -m", 1),
431            args.timeout, runner=SandboxRunner,
432            baseline=self.context.baseline_text, known=self.context.baseline_signatures,
433        )
434        if outcome.reason in NOT_TEST_RUNS:
435            return None
436        outcome.data["shell_verification"] = True
437        outcome.content = ("[Aster ran this test command as formal verification in a disposable copy; "
438                           "files the tests write are not kept.]\n" + outcome.content)
439        return outcome

449    async def _service_start(self, args: ServiceStart) -> Outcome:
450        sandbox, workspace = self.context.sandbox, self.context.workspace
451        current = sandbox.services.get(args.name)
452        if current and current["process"].poll() is None:
453            return Outcome(status="blocked", reason="service_running",
454                           content=f"{args.name} is already running; stop it first or use another name.")
455        directory = Path(tempfile.gettempdir()) / f"aster-services-{os.getuid()}"
456        directory.mkdir(mode=0o700, exist_ok=True)
457        log = directory / f"{self.context.run_id}-{args.name}.log"
458        log.write_bytes(b"")
459        before = workspace.snapshot()
460        process = sandbox.start_service(workspace, args.command, log)
461        sandbox.services[args.name] = {"process": process, "command": args.command, "log": log,
462                                       "keep": args.keep_running}
463        await asyncio.sleep(SERVICE_STARTUP_SECONDS)  # long enough to bind a port or fail fast
464        code = process.poll()
465        changed = Workspace.changes(before, workspace.snapshot())
466        state = f"running (pid {process.pid})" if code is None else f"exited with code {code}"
467        return Outcome(status="success" if code is None else "error",
468                       reason="" if code is None else "service_exited", changed_paths=changed,
469                       content=f"{args.name}: {state}\n{self._tail(log)}",
470                       data={"service": args.name, "running": code is None, "keep_running": args.keep_running})

472    async def _service_status(self, args: ServiceName) -> Outcome:
473        services = self.context.sandbox.services
474        names = [args.name] if args.name else sorted(services)
475        if args.name and args.name not in services:
476            return Outcome(status="blocked", reason="service_unknown", content=", ".join(sorted(services)) or "none")
477        lines = []
478        for name in names:
479            service = services[name]
480            code = service["process"].poll()
481            lines.append(f"{name}: {'running' if code is None else f'exited with code {code}'}"
482                         f"{' (kept after the task)' if service['keep'] else ''} — {service['command']}"
483                         + ("\n" + self._tail(service["log"]) if args.name else ""))
484        return Outcome(status="success", content="\n".join(lines) or "No services.")

504    @staticmethod
505    def _write_command(path: str, content: str) -> str:
506        encoded = base64.b64encode(content.encode()).decode()
507        program = (
508            "from pathlib import Path; import base64; "
509            f"p=Path({path!r}); p.parent.mkdir(parents=True,exist_ok=True); "
510            f"p.write_bytes(base64.b64decode({encoded!r}))"
511        )
512        # Keep file contents out of execve arguments, including Python's -c argument.
513        return "python - <<'ASTER_WRITE'\n" + program + "\nASTER_WRITE\n"
