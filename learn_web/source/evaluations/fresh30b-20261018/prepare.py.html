1"""Select the fresh30b held-out set, drawn before any v14 change is evaluated on it. No model calls.
2
3Same composition and seeded-hash ranking as fresh30, with a new seed and every previously used
4task (including all of fresh30) excluded. Nothing about agent outcomes enters the selection.
5Tasks carry exposure 'new30' (the general-50 engine keys its small/library setup on it) and
6pool 'fresh30b' (the v12 runner's --tasks fresh30b selector).
7"""
8import ast
9import hashlib
10import json
11import shutil
12import sys
13from pathlib import Path
14
15HERE = Path(__file__).resolve().parent
16REPO = HERE.parents[1]
17PRIVATE = REPO / 'artifacts/private' / HERE.name
18G50 = REPO / 'artifacts/private/general-coding-50-20261012'
19G10 = REPO / 'artifacts/private/general-coding-20261010'
20DATA = REPO / 'datasets' / HERE.name
21SEED = '20261018b:'
22FRESH30 = REPO / 'datasets/fresh30-20261018'
23SWE_REPOS = ['django/django', 'sympy/sympy', 'sphinx-doc/sphinx', 'pydata/xarray', 'pytest-dev/pytest',
24             'astropy/astropy', 'scikit-learn/scikit-learn', 'matplotlib/matplotlib']

27def save(path, data):
28    path.parent.mkdir(parents=True, exist_ok=True)
29    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
30
31
32def load(path):
33    return json.loads(path.read_text())
34
35
36def rank(name):
37    return hashlib.sha256((SEED + name).encode()).hexdigest()
38
39
40def main():
41    assert not PRIVATE.exists(), 'Already prepared; the selection is frozen'
42    used = load(G50 / 'tasks.json')
43    used_ids = {t['id'] for t in used} | {t.get('source_id') for t in used if t.get('source_id')}
44    previous = load(REPO / 'datasets/general-coding-20261010/selection.json')
45    used_ids |= {s.removeprefix('worker-code-v1__') for s in previous['repository']['excluded_previous_ids']}
46    used_slugs = {t['task_name'] for t in used if t.get('task_name')}
47    first = load(FRESH30 / 'tasks.json')  # every fresh30 task, by id, source id and slug
48    used_ids |= {t['id'] for t in first} | {t['source_id'] for t in first if t.get('source_id')}
49    used_slugs |= {t['task_name'] for t in first if t.get('task_name')}
50    used_ids |= {t['id'] for t in first if t['kind'] == 'terminal'}
51    tasks, selection = [], {'seed': SEED, 'excluded_count': len(used_ids)}
52
53    rows = load(REPO / 'artifacts/private/swebench-verified-20261008/rows.json')
54    selected_rows = []
55    selection['repository'] = {}
56    for repo in SWE_REPOS:
57        candidates = sorted((r for r in rows if r['repo'] == repo and r['instance_id'] not in used_ids
58                             and r['difficulty'] != '>4 hours'), key=lambda r: rank(r['instance_id']))
59        row = candidates[0]
60        selection['repository'][repo] = {'ranked': [r['instance_id'] for r in candidates[:20]], 'selected': row['instance_id']}
61        selected_rows.append(row)
62        tasks.append({'id': row['instance_id'], 'kind': 'repository', 'source': 'SWE-bench Verified',
63                      'image': row['image'], 'workspace': '/testbed', 'base_commit': row['base_commit'],
64                      'repo': row['repo'], 'difficulty': row['difficulty'], 'previously_used': False,
65                      'exposure': 'new30', 'pool': 'fresh30b', 'language': 'Python', 'prompt': row['problem_statement'],
66                      'test_python': '/opt/miniconda3/envs/testbed/bin/python'})
67
68    selection['aider'] = {}
69    for language, count in (('python', 2), ('javascript', 4), ('cpp', 4)):
70        catalog = G50 / 'sources/aider-catalog' / language / 'exercises/practice'
71        candidates = sorted((p for p in catalog.iterdir() if language + '-' + p.name not in used_slugs),
72                            key=lambda p: rank(language + '/' + p.name))
73        selection['aider'][language] = {'ranked': [p.name for p in candidates], 'selected': [p.name for p in candidates[:count]]}
74        for folder in candidates[:count]:
75            name = language + '-' + folder.name
76            shutil.copytree(folder, PRIVATE / 'sources/aider-tasks' / name)
77            prompt = (folder / '.docs/instructions.md').read_text()
78            if (folder / '.docs/instructions.append.md').exists():
79                prompt += '\n\n' + (folder / '.docs/instructions.append.md').read_text()
80            tasks.append({'id': 'aider-' + name, 'kind': 'small', 'source': 'Aider Polyglot',
81                          'image': ('aster/general50-tools-slim:20261012' if language == 'cpp'
82                                    else 'alexgshaw/cancel-async-tasks:20251031'), 'workspace': '/app',
83                          'task_name': name, 'difficulty': 'unlabelled', 'previously_used': False,
84                          'exposure': 'new30', 'pool': 'fresh30b', 'language': {'python': 'Python', 'javascript': 'JavaScript', 'cpp': 'C++'}[language],
85                          'prompt': prompt, 'test_python': '/opt/task-python/bin/python',
86                          'files': load(folder / '.meta/config.json')['files']})
87
88    bigrows = load(G50 / 'sources/bigcodebench.json')
89    allowed = set(sys.stdlib_module_names) | {'numpy', 'pandas', 'scipy', 'sklearn', 'matplotlib', 'seaborn',
90                                             'PIL', 'bs4', 'lxml', 'yaml', 'faker'}
91    blocked = {'requests', 'urllib', 'http', 'ftplib', 'smtplib', 'socket', 'subprocess', 'multiprocessing',
92               'tkinter', 'turtle', 'unittest', 'pytest'}
93    candidates = sorted((r for r in bigrows if r['task_id'] not in used_ids
94                         and set(ast.literal_eval(r['libs'])) <= allowed
95                         and not set(ast.literal_eval(r['libs'])) & blocked), key=lambda r: rank(r['task_id']))
96    selection['bigcodebench'] = {'ranked': [r['task_id'] for r in candidates[:40]], 'selected': [r['task_id'] for r in candidates[:10]]}
97    for row in candidates[:10]:
98        name = 'bigcodebench-' + row['task_id'].split('/')[1]
99        folder = PRIVATE / 'sources/library-tasks' / name
100        folder.mkdir(parents=True)
101        save(folder / 'record.json', row)
102        (folder / 'solution.py').write_text(row['code_prompt'] + '    pass\n')
103        (folder / 'reference.py').write_text(row['complete_prompt'] + row['canonical_solution'])
104        (folder / 'test_solution.py').write_text(row['test'])
105        tasks.append({'id': name, 'kind': 'library', 'source': 'BigCodeBench v0.1.4', 'source_id': row['task_id'],
106                      'image': 'alexgshaw/cancel-async-tasks:20251031', 'workspace': '/app', 'task_name': name,
107                      'difficulty': 'unlabelled', 'previously_used': False, 'exposure': 'new30', 'pool': 'fresh30b', 'language': 'Python',
108                      'prompt': row['instruct_prompt'] + '\nImplement task_func in solution.py.\n' + row['complete_prompt'],
109                      'test_python': '/opt/task-python/bin/python', 'libraries': ast.literal_eval(row['libs'])})
110
111    # Terminal-Bench: rank all unused tasks; preflight later keeps the first two that are ready.
112    catalog = load(G10 / 'sources/terminal-catalog.json')
113    # Same pool as the earlier sets: archived tasks with a prebuilt image and a 900 s agent budget
114    # (the catalog also holds QA fixtures and newer tasks that need hour-long builds).
115    candidates = sorted((r for r in catalog if r['path'].startswith('archive/')
116                         and r['config'].get('environment', {}).get('docker_image')
117                         and r['config'].get('agent', {}).get('timeout_sec', 10**9) <= 900
118                         and 'terminal-' + r['name'] not in used_ids and r['name'] not in used_slugs),
119                        key=lambda r: rank('terminal/' + r['name']))
120    selection['terminal'] = {'ranked': [r['name'] for r in candidates[:12]],
121                             'rule': 'first two in rank order whose image pulls and whose starter fails and reference passes'}
122    save(PRIVATE / 'terminal-candidates.json', candidates[:12])
123
124    for task in tasks:
125        task['prompt_sha256'] = hashlib.sha256(task['prompt'].encode()).hexdigest()
126    save(PRIVATE / 'tasks.pending.json', tasks)
127    save(PRIVATE / 'repository-original.json', selected_rows)
128    save(DATA / 'selection.json', selection)
129    print('Selected', len(tasks), 'non-terminal tasks; terminal candidates ranked; no model calls.')
130    for task in tasks:
131        print(' ', task['id'], task['language'], task.get('difficulty'))
132
133
134if __name__ == '__main__':
135    main()
