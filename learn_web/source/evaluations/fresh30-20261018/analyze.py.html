1"""Paired comparison on fresh30 with bootstrap confidence intervals. Offline only.
2
3Usage: analyze.py label=run[,run...] ...   (several runs per label are repeats; per-task
4values are averaged across repeats before resampling tasks)
5"""
6import json
7import random
8import sys
9from pathlib import Path
10
11HERE = Path(__file__).resolve().parent
12RUNS = HERE.parent / 'harness-v12-20261017' / 'runs'
13METRICS = ('passed', 'normal_delivery_and_passed', 'requests', 'weighted_cost')
14
15
16def per_task(runs):
17    table = {}
18    for run in runs:
19        for row in json.loads((RUNS / run / 'results.json').read_text()):
20            table.setdefault(row['task'], []).append(row)
21    return {task: {m: sum(float(r[m]) for r in rows) / len(rows) for m in METRICS} for task, rows in table.items()}
22
23
24def bootstrap(values, rounds=10000, seed=20261018):
25    rng = random.Random(seed)
26    n = len(values)
27    means = sorted(sum(values[rng.randrange(n)] for _ in range(n)) / n for _ in range(rounds))
28    return means[int(0.025 * rounds)], means[int(0.975 * rounds)]
29
30
31def main():
32    labels = dict(arg.split('=', 1) for arg in sys.argv[1:])
33    data = {label: per_task(runs.split(',')) for label, runs in labels.items()}
34    tasks = sorted(set.intersection(*(set(d) for d in data.values())))
35    report = {'tasks': len(tasks), 'agents': {}, 'paired': {}}
36    for label, table in data.items():
37        report['agents'][label] = {}
38        for m in METRICS:
39            values = [table[t][m] for t in tasks]
40            total = sum(values)
41            low, high = bootstrap(values)
42            report['agents'][label][m] = {'total': round(total, 3), 'ci95_total': [round(low * len(tasks), 3), round(high * len(tasks), 3)]}
43    names = list(data)
44    for i, a in enumerate(names):
45        for b in names[i + 1:]:
46            report['paired'][f'{a} - {b}'] = {}
47            for m in METRICS:
48                diffs = [data[a][t][m] - data[b][t][m] for t in tasks]
49                low, high = bootstrap(diffs)
50                report['paired'][f'{a} - {b}'][m] = {'difference': round(sum(diffs), 3),
51                                                     'ci95': [round(low * len(tasks), 3), round(high * len(tasks), 3)]}
52    print(json.dumps(report, indent=2))
53    return report
54
55
56if __name__ == '__main__':
57    main()
