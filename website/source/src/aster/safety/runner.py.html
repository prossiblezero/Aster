36    async def run(
37        self,
38        workspace: Workspace,
39        command: str,
40        timeout: float = 30,
41        *,
42        readonly: bool = False,
43        expected_files: dict[str, str | None] | None = None,
44    ) -> Outcome:
45        readonly = readonly or self.settings.read_only
46        constrained = readonly or self.settings.write_scope is not None
47        if conflict := self._check_expected(workspace, expected_files):
48            return conflict
49        if self.settings.sandbox == "off":
50            if constrained:
51                return Outcome(status="blocked", reason="isolation_required_for_scope")
52            before = workspace.snapshot()
53            result = await self._process(["/bin/sh", "-c", command], workspace.root, timeout)
54            result.changed_paths = Workspace.changes(before, workspace.snapshot())
55            result.partial_success = bool(result.changed_paths) and result.status != "success"
56            result.data["isolation"] = "off"
57            return result
58        with tempfile.TemporaryDirectory(prefix="aster-exec-") as directory:
59            root = Path(directory)
60            before = workspace.snapshot()
61            for path in workspace.files():
62                target = root / path.relative_to(workspace.root)
63                target.parent.mkdir(parents=True, exist_ok=True)
64                shutil.copy2(path, target)
65            if conflict := self._check_expected(Workspace(root), expected_files):
66                return conflict
67            (root / ".tmp").mkdir(exist_ok=True)
68            launcher = Path(__file__).with_name("linux.py")
69            result = await self._process(
70                [
71                    sys.executable,
72                    "-I",
73                    str(launcher),
74                    str(root),
75                    "on" if self.settings.network else "off",
76                    command,
77                ],
78                root,
79                timeout,
80            )
81            if result.exit_code == 125 and "ASTER_SANDBOX_UNAVAILABLE" in result.content:
82                if self.settings.sandbox == "best_effort" and not constrained:
83                    if conflict := self._check_expected(workspace, expected_files):
84                        return conflict
85                    before = workspace.snapshot()
86                    fallback = await self._process(
87                        ["/bin/sh", "-c", command], workspace.root, timeout
88                    )
89                    fallback.data.update(isolation="off", degraded=True)
90                    fallback.changed_paths = Workspace.changes(before, workspace.snapshot())
91                    fallback.partial_success = (
92                        bool(fallback.changed_paths) and fallback.status != "success"
93                    )
94                    return fallback
95                return Outcome(
96                    status="blocked",
97                    reason="sandbox_unavailable",
98                    content=result.content,
99                    data={"isolation": "unavailable"},
100                )
101            result.data["isolation"] = "landlock+seccomp"
102            temporary = root / ".tmp"
103            if temporary.is_symlink() or not temporary.is_dir():
104                temporary.unlink(missing_ok=True)
105            else:
106                shutil.rmtree(temporary)
107            try:
108                # Validate entries before opening them: FIFO reads have no process timeout.
109                violations = [
110                    str(p.relative_to(root))
111                    for p in root.rglob("*")
112                    if p.is_symlink()
113                    or not (p.is_file() or p.is_dir())
114                    or sensitive(p.relative_to(root))
115                    or any(x in CONTROL for x in p.relative_to(root).parts)
116                ]
117                if violations:
118                    result.status = "blocked"
119                    result.reason = "sandbox_protected_output"
120                    result.data["rejected_paths"] = violations[:20]
121                    return result
122                after = Workspace(root).snapshot()
123                changed = Workspace.changes(before, after)
124                if readonly and changed:
125                    return Outcome(
126                        status="blocked",
127                        reason="read_only",
128                        content=result.content,
129                        data={"isolation": "landlock+seccomp"},
130                    )
131                return self._publish(workspace, root, before, after, result)
132            finally:
133                shutil.rmtree(temporary, ignore_errors=True)

135    @staticmethod
136    def _check_expected(
137        workspace: Workspace, expected_files: dict[str, str | None] | None
138    ) -> Outcome | None:
139        for name, expected in (expected_files or {}).items():
140            try:
141                path = workspace.path(name)
142                current = None
143                if path.exists():
144                    with path.open("rb") as stream:
145                        current = hashlib.file_digest(stream, "sha256").hexdigest()
146                if current == expected:
147                    continue
148            except (OSError, ValueError):
149                pass  # Unreadable or replaced with a directory is also stale evidence.
150            return Outcome(
151                status="blocked",
152                reason="fresh_read_required",
153                content=f"Read {name} again: it changed while the edit was pending.",
154            )
155        return None

157    def _publish(
158        self,
159        workspace: Workspace,
160        staged: Path,
161        before: dict[str, str],
162        after: dict[str, str],
163        result: Outcome,
164    ) -> Outcome:
165        changed = Workspace.changes(before, after)
166        current = workspace.snapshot()
167        if any(current.get(name) != before.get(name) for name in changed):
168            result.status, result.reason = "error", "concurrent_workspace_change"
169            return result
170        removed = before.keys() - after.keys()
171        directories = set()
172        try:
173            # Plan the complete write set before changing anything, including file↔directory
174            # replacements. Ignored or newly created user files must never be removed with it.
175            for name in changed:
176                target = workspace.path(name, write=True)
177                if self.settings.write_scope is not None and not any(
178                    name == p or name.startswith(p.rstrip("/") + "/")
179                    for p in self.settings.write_scope
180                ):
181                    raise BoundaryError("write_scope")
182                if name in after and target.is_dir():
183                    for child in [target, *target.rglob("*")]:
184                        relative = child.relative_to(workspace.root)
185                        if (
186                            child.is_symlink()
187                            or sensitive(relative)
188                            or any(part in CONTROL for part in relative.parts)
189                            or (not child.is_dir() and str(relative) not in removed)
190                        ):
191                            raise BoundaryError("directory_not_empty")
192                        if child.is_dir():
193                            directories.add(child)
194        except BoundaryError as exc:
195            result.status, result.reason = "blocked", str(exc)
196            return result
197
198        # ponytail: synchronous publication serializes this process's workers. Each file is
199        # atomic; the batch is not. On disk errors keep/report applied changes, never roll
200        # back over an external editor's work. Separate Aster processes need an OS lock.
201        try:
202            for name in sorted(removed, key=lambda name: name.count("/"), reverse=True):
203                workspace.path(name, write=True).unlink(missing_ok=True)
204            for directory in sorted(directories, key=lambda path: len(path.parts), reverse=True):
205                directory.rmdir()
206            for name in changed:
207                if name not in after:
208                    continue
209                target = workspace.path(name, write=True)
210                target.parent.mkdir(parents=True, exist_ok=True)
211                fd, pending = tempfile.mkstemp(dir=target.parent)
212                os.close(fd)
213                try:
214                    shutil.copy2(staged / name, pending)
215                    os.replace(pending, target)
216                finally:
217                    Path(pending).unlink(missing_ok=True)
218        except (OSError, BoundaryError) as exc:
219            result.status, result.reason = "error", "workspace_publish_failed"
220            result.data["publish_error"] = type(exc).__name__
221        result.changed_paths = Workspace.changes(current, workspace.snapshot())
222        result.partial_success = bool(result.changed_paths) and result.status != "success"
223        return result

225    async def _process(self, argv: list[str], cwd: Path, timeout: float) -> Outcome:
226        started = time.monotonic()
227        process = await asyncio.create_subprocess_exec(
228            *argv,
229            cwd=cwd,
230            env=self.environment(cwd),
231            stdin=asyncio.subprocess.DEVNULL,
232            stdout=asyncio.subprocess.PIPE,
233            stderr=asyncio.subprocess.STDOUT,
234            start_new_session=True,
235        )
236        chunks = bytearray()
237
238        async def drain():
239            while chunk := await process.stdout.read(8192):
240                if len(chunks) < 256 * 1024:
241                    chunks.extend(chunk[: 256 * 1024 - len(chunks)])
242            await process.wait()
243
244        reader = asyncio.create_task(drain())
245        status = "success"
246        try:
247            await asyncio.wait_for(asyncio.shield(reader), timeout)
248            if process.returncode:
249                status = "error"
250        except (TimeoutError, asyncio.CancelledError) as exc:
251            status = "timeout" if isinstance(exc, TimeoutError) else "cancelled"
252            try:
253                os.killpg(process.pid, signal.SIGKILL)
254            except ProcessLookupError:
255                pass
256            await reader
257            if status == "cancelled":
258                raise
259        finally:
260            # Prevent a detached background process surviving its shell.
261            try:
262                os.killpg(process.pid, signal.SIGKILL)
263            except ProcessLookupError:
264                pass
265        return Outcome(
266            status=status,
267            content=chunks.decode("utf-8", errors="replace"),
268            exit_code=process.returncode,
269            duration_ms=(time.monotonic() - started) * 1000,
270        )
